     1→//! Quartz Tauri 壳层。
     2→//!
     3→//! 架构契约（方案 A）：UI（WebView/JS）只调用本文件注册的 command；
     4→//! command 是 quartz-core 能力的薄封装，不含业务逻辑。
     5→//!
     6→//! 命令契约（v3 蓝图 + M0 启动接入）：
     7→//! - `offline_login(name)`        离线登录：MD5("OfflinePlayer:"+name) -> accounts.json
     8→//! - `get_account()`              当前选中账号
     9→//! - `list_accounts()`            全部账号 + 选中标记
    10→//! - `select_account(uuid)`       切换选中账号
    11→//! - `remove_account(uuid)`       删除账号
    12→//! - `list_versions()`            真实 manifest 版本列表（release 优先）
    13→//! - `list_installed_versions()`  本地 versions/ 已安装版本
    14→//! - `download_game(version_id)`  后台 spawn 完整下载管线，
    15→//!                                经 `download://progress` 事件上报 {stage,current,total,percent,file}
    16→//! - `get_game_dir()`             标准游戏根目录绝对路径（UI 常显）
    17→//! - `get_env_info()`             存储诊断：根目录 + 可写性 + 存储模式 + 完整文件权限
    18→//! - `get_permission_state()`     存储权限状态 {granted}
    19→//! - `request_storage_permission()` 触发系统权限弹窗（授权后经 focus/visibility 重检）
    20→//! - `reinit_storage()`           权限变化后重解析存储根 + 后台迁移旧数据
    21→//!                                （`quartz://migrate` 事件上报迁移进度）
    22→//! - `launch_preflight(version_id)` 启动前自检（版本/账号/Java 运行时）
    23→//! - `get_runtime_status(java_major)` JVM pack 状态（installed/version/path）
    24→//! - `download_runtime(java_major)` 下载 JVM 运行时 pack（`runtime://progress` 事件）
    25→//! - `launch_game(version_id)`    M0 已接入：JRE pack + LaunchPlan 子进程启动，
    26→//!                                `launch://log` / `launch://exit` 事件实时上报

    52→#[cfg(target_os = "android")]
    53→fn with_jni<T>(
    54→    f: impl FnOnce(&mut jni::JNIEnv, &jni::objects::JObject) -> Option<T>,
    55→) -> Option<T> {
    56→    std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
    57→        let ctx = ndk_context::android_context();
    58→        let vm = unsafe { jni::JavaVM::from_raw(ctx.vm().cast()) }.ok()?;
    59→        let mut env = vm.attach_current_thread().ok()?;
    60→        let context = unsafe { jni::objects::JObject::from_raw(ctx.context().cast()) };
    61→        f(&mut env, &context)
    62→    }))
    63→    .ok()
    64→    .flatten()
    65→}

    67→#[cfg(target_os = "android")]
    68→fn jni_file_path(env: &mut jni::JNIEnv, file: &jni::objects::JObject) -> Option<PathBuf> {
    69→    let path = env
    70→        .call_method(file, "getAbsolutePath", "()Ljava/lang/String;", &[])
    71→        .ok()?
    72→        .l()
    73→        .ok()?;
    74→    let s: String = env.get_string((&path).into()).ok()?.into();
    75→    if s.is_empty() {
    76→        None
    77→    } else {
    78→        Some(PathBuf::from(s))
    79→    }
    80→}

    82→#[cfg(target_os = "android")]
    83→fn jni_external_files_dir(env: &mut jni::JNIEnv, context: &jni::objects::JObject) -> Option<PathBuf> {
    84→    use jni::objects::JValue;
    85→    let file = env
    86→        .call_method(
    87→            context,
    88→            "getExternalFilesDir",
    89→            "(Ljava/lang/String;)Ljava/io/File;",
    90→            &[JValue::Object(&jni::objects::JObject::null())],
    91→        )
    92→        .ok()?
    93→        .l()
    94→        .ok()?;
    95→    if file.is_null() {
    96→        return None;
    97→    }
    98→    jni_file_path(env, &file)
    99→}

   114→// ---------------------------------------------------------------------------
   115→// 存储权限（完整文件权限）
   116→// ---------------------------------------------------------------------------
   117→//
   118→// 设计：APK 固定 targetSdk 28（Android 10 前的存储模型 + 允许 exec() 应用目录
   119→// 内的 JVM —— Pojav/Termux 同款方案）。
   120→//
   121→// 授权通道按系统版本分流（MainActivity.requestStorageAccess）:
   122→// - Android 11+（API 30+）：MANAGE_EXTERNAL_STORAGE「所有文件访问」系统设置页。
   123→//   不走 WRITE 弹窗的原因：targetSdk<=29 应用在 Android 11+ 经弹窗授予存储权限时，
   124→//   系统会杀掉应用进程刷新 FUSE 挂载视图 —— 用户视角即「授权后闪退」。
   125→//   所有文件访问页授权不杀进程，返回后 focus 重检即可切换标准目录。
   126→// - Android 10 及以下（API <= 29）：READ/WRITE_EXTERNAL_STORAGE 运行时弹窗
   127→//   （旧存储模型，授权即时生效不杀进程）。
   128→// 游戏文件即可放在文件管理器可见的标准目录 /storage/emulated/0/Quartz/files。
   129→
   130→/// Android 端存储权限是否已授权。
   131→/// 优先走壳层 Kotlin 判定（Android 11+ = isExternalStorageManager，
   132→/// 10- = WRITE_EXTERNAL_STORAGE 已 GRANTED）；JNI 不可用时回退纯 checkSelfPermission。
   133→#[cfg(target_os = "android")]
   134→fn android_storage_permission_granted() -> bool {
   135→    // 1) MainActivity.isStorageAccessGranted()（覆盖 MANAGE_EXTERNAL_STORAGE）
   136→    if let Some(granted) = with_jni(|env, ctx| {
   137→        Some(
   138→            env.call_method(ctx, "isStorageAccessGranted", "()Z", &[])
   139→                .ok()?
   140→                .z()
   141→                .ok()?,
   142→        )
   143→    }) {
   144→        return granted;
   145→    }
   146→    // 2) 回退：WRITE checkSelfPermission（仅覆盖旧模型，JNI 异常时尽力而为）
   147→    use jni::objects::JValue;
   148→    with_jni(|env, ctx| {
   149→        let perm = env
   150→            .new_string("android.permission.WRITE_EXTERNAL_STORAGE")
   151→            .ok()?;
   152→        let granted = env
   153→            .call_method(
   154→                ctx,
   155→                "checkSelfPermission",
   156→                "(Ljava/lang/String;)I",
   157→                &[JValue::Object(&perm.into())],
   158→            )
   159→            .ok()?
   160→            .i()
   161→            .ok()?;
   162→        // PackageManager.PERMISSION_GRANTED == 0
   163→        Some(granted == 0)
   164→    })
   165→    .unwrap_or(false)
   166→}
   167→
   168→/// 存储权限是否已授权（桌面端恒真）。
   169→fn storage_permission_granted() -> bool {
   170→    #[cfg(target_os = "android")]
   171→    {
   172→        android_storage_permission_granted()
   173→    }
   174→    #[cfg(not(target_os = "android"))]
   175→    {
   176→        true
   177→    }
   178→}

   186→/// 申请完整文件权限：Android 11+ 跳系统「所有文件访问」设置页（授权不杀进程），
   187→/// Android 10- 触发运行时弹窗。结果不通过本命令返回：UI 在 focus/visibilitychange
   188→/// 时重新查询状态（回到应用后自动重检）。
   189→#[tauri::command]
   190→fn request_storage_permission() -> Result<bool, String> {
   191→    #[cfg(target_os = "android")]
   192→    {
   193→        // 1) MainActivity.requestStorageAccess()：按系统版本分流（11+ 设置页 / 10- 弹窗）
   194→        let via_kotlin = with_jni(|env, ctx| {
   195→            env.call_method(ctx, "requestStorageAccess", "()V", &[]).ok()?;
   196→            Some(())
   197→        });
   198→        if via_kotlin.is_some() {
   199→            return Ok(true);
   200→        }
   201→        // 2) 回退：直接 requestPermissions 弹窗（旧路径，仅 WRITE/READ）
   202→        use jni::objects::JValue;
   203→        with_jni(|env, ctx| {
   204→            let write = env.new_string("android.permission.WRITE_EXTERNAL_STORAGE").ok()?;
   205→            let read = env.new_string("android.permission.READ_EXTERNAL_STORAGE").ok()?;
   206→            let write_obj: jni::objects::JObject = write.into();
   207→            let arr = env
   208→                .new_object_array(2, "java/lang/String", write_obj)
   209→                .ok()?;
   210→            let read_obj: jni::objects::JObject = read.into();
   211→            env.set_object_array_element(&arr, 1, read_obj).ok()?;
   212→            env.call_method(
   213→                ctx,
   214→                "requestPermissions",
   215→                "([Ljava/lang/String;I)V",
   216→                &[JValue::Object(&arr), JValue::Int(42)],
   217→            )
   218→            .ok()?;
   219→            Some(())
   220→        });
   221→        Ok(true)
   222→    }
   223→    #[cfg(not(target_os = "android"))]
   224→    {
   225→        Ok(true)
   226→    }
   227→}

   242→static STORAGE_DIAG: Mutex<Option<StorageDiag>> = Mutex::new(None);

   244→/// 迁移互斥：焦点/可见性事件可能连续触发多次 reinit_storage，防止并发重复拷贝。
   245→static MIGRATING: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
   246→
   247→/// Android 平台解析真实可写目录；ndk-context 未初始化等异常一律回退硬编码路径。
   248→///
   249→/// 优先级（与授权状态联动）：
   250→/// 1. 已授权完整文件权限 -> 公共标准目录 /storage/emulated/0/Quartz/files（可见、卸载不丢）
   251→/// 2. 应用外部私有目录（getExternalFilesDir，可浏览，卸载丢失）
   252→/// 3. 应用内部私有目录（getFilesDir，必定可写）
   253→#[cfg(target_os = "android")]
   254→fn resolve_android_root() -> (PathBuf, &'static str) {
   255→    if storage_permission_granted() {
   256→        let public = PathBuf::from("/storage/emulated/0/Quartz/files");
   257→        if std::fs::create_dir_all(&public).is_ok() {
   258→            return (public, "public");
   259→        }
   260→        eprintln!("[quartz] 公共标准目录创建失败，回退应用私有目录");
   261→    }
   262→    let result = with_jni(|env, context| {
   263→        // 外部私有目录（用户可用文件管理器查看 .minecraft）
   264→        if let Some(p) = jni_external_files_dir(env, context) {
   265→            return Some((p, "external"));
   266→        }
   267→        // 内部私有目录（必定可写）
   268→        jni_internal_files_dir(env, context).map(|p| (p, "internal"))
   269→    });
   270→    result.unwrap_or_else(|| {
   271→        (
   272→            PathBuf::from(format!(
   273→                "/storage/emulated/0/Android/data/{}/files",
   274→                paths::ANDROID_PACKAGE_ID
   275→            )),
   276→            "fallback",
   277→        )
   278→    })
   279→}

   281→/// Android 内部私有目录（Context.getFilesDir()，任何设备必定可写；exec 可用）。
   282→#[cfg(target_os = "android")]
   283→fn resolve_android_internal() -> Option<PathBuf> {
   284→    with_jni(|env, context| jni_internal_files_dir(env, context))
   285→}

   287→/// 初始化游戏根目录与 JVM pack 目录并创建标准布局；返回诊断信息。
   288→fn init_storage() -> StorageDiag {
   289→    #[cfg(target_os = "android")]
   290→    let (root, mode) = resolve_android_root();
   291→    #[cfg(not(target_os = "android"))]
   292→    let (root, mode) = (paths::game_root(), "desktop");
   293→
   294→    let _ = paths::set_runtime_root(root.clone());
   295→    let mut layout = paths::ensure_layout();
   296→
   297→    // 外部目录不可写（EACCES / os error 13）时回退内部私有目录，
   298→    // 保证离线登录与下载在任何设备上永远可用。
   299→    #[cfg(target_os = "android")]
   300→    let mut mode = mode;
   301→    #[cfg(target_os = "android")]
   302→    if layout.is_err() && mode != "internal" {
   303→        eprintln!(
   304→            "[quartz] 当前根目录不可写（{:?}），回退内部存储",
   305→            layout.as_ref().err()
   306→        );
   307→        if let Some(internal) = resolve_android_internal() {
   308→            let _ = paths::set_runtime_root(internal.clone());
   309→            layout = paths::ensure_layout();
   310→            mode = "internal";
   311→        }
   312→    }
   313→
   314→    // JVM pack 必须放在可 exec 的内部目录（FUSE noexec + targetSdk>=29 禁 exec 的约束）
   315→    #[cfg(target_os = "android")]
   316→    if let Some(internal) = resolve_android_internal() {
   317→        paths::set_runtime_jres_root(internal.join("jres"));
   318→    }
   319→
   320→    let diag = StorageDiag {
   321→        mode,
   322→        root: paths::game_root(),
   323→        layout_ok: layout.is_ok(),
   324→        layout_error: layout.err().map(|e| format!("{e}")),
   325→    };
   326→    eprintln!(
   327→        "[quartz] 存储初始化：mode={} root={} jres={}",
   328→        diag.mode,
   329→        diag.root.display(),
   330→        paths::runtime_dir().display()
   331→    );
   332→    diag
   333→}

   345→/// 递归收集「源存在且目标缺失/不一致」的文件（不覆盖目标已有的同名同尺寸文件）。
   346→fn collect_migrate_files(src: &Path, dst: &Path, out: &mut Vec<(PathBuf, PathBuf, u64)>) {
   347→    let Ok(entries) = std::fs::read_dir(src) else {
   348→        return;
   349→    };
   350→    for entry in entries.flatten() {
   351→        let sp = entry.path();
   352→        let dp = dst.join(entry.file_name());
   353→        if sp.is_dir() {
   354→            collect_migrate_files(&sp, &dp, out);
   355→        } else if sp.is_file() {
   356→            let size = entry.metadata().map(|m| m.len()).unwrap_or(0);
   357→            let dst_size = std::fs::metadata(&dp).map(|m| m.len()).unwrap_or(u64::MAX);
   358→            if dst_size != size {
   359→                out.push((sp, dp, size));
   360→            }
   361→        }
   362→    }
   363→}

   365→/// 后台迁移：把旧根（私有目录）的数据搬到新根（公共标准目录），
   366→/// 经 `quartz://migrate` 事件上报 {percent,done,error}。
   367→fn migrate_root(from: PathBuf, to: PathBuf, app: AppHandle) {
   368→    let mut plan: Vec<(PathBuf, PathBuf, u64)> = Vec::new();
   369→    collect_migrate_files(&from.join(".minecraft"), &to.join(".minecraft"), &mut plan);
   370→    let old_accounts = from.join("accounts.json");
   371→    let new_accounts = to.join("accounts.json");
   372→    let migrate_accounts = old_accounts.exists() && !new_accounts.exists();
   373→    if migrate_accounts {
   374→        let size = std::fs::metadata(&old_accounts).map(|m| m.len()).unwrap_or(0);
   375→        plan.push((old_accounts, new_accounts, size));
   376→    }
   377→
   378→    if plan.is_empty() {
   379→        let _ = app.emit(
   380→            "quartz://migrate",
   381→            &json!({ "percent": 100.0, "done": true, "error": null }),
   382→        );
   383→        return;
   384→    }
   385→
   386→    let total: u64 = plan.iter().map(|(_, _, s)| *s).sum::<u64>().max(1);
   387→    eprintln!("[quartz] 开始迁移存储：{} 个文件（{} 字节）", plan.len(), total);
   388→    let mut copied: u64 = 0;
   389→    let mut last = std::time::Instant::now() - std::time::Duration::from_secs(1);
   390→    for (src, dst, size) in plan {
   391→        if let Some(parent) = dst.parent() {
   392→            if std::fs::create_dir_all(parent).is_err() {
   393→                let _ = app.emit(
   394→                    "quartz://migrate",
   395→                    &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false,
   396→                             "error": format!("无法创建目录 {}", parent.display()) }),
   397→                );
   398→                return;
   399→            }
   400→        }
   401→        if std::fs::copy(&src, &dst).is_err() {
   402→            let _ = app.emit(
   403→                "quartz://migrate",
   404→                &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false,
   405→                         "error": format!("复制失败：{} -> {}", src.display(), dst.display()) }),
   406→            );
   407→            return;
   408→        }
   409→        copied += size;
   410→        let now = std::time::Instant::now();
   411→        if now.duration_since(last) >= std::time::Duration::from_millis(150) {
   412→            last = now;
   413→            let _ = app.emit(
   414→                "quartz://migrate",
   415→                &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false, "error": null }),
   416→            );
   417→        }
   418→    }
   419→    eprintln!("[quartz] 存储迁移完成");
   420→    let _ = app.emit(
   421→        "quartz://migrate",
   422→        &json!({ "percent": 100.0, "done": true, "error": null }),
   423→    );
   424→}

   426→/// 权限变化后重解析存储根；根变化时后台迁移旧数据（幂等，跳过目标已有文件）。
   427→#[tauri::command]
   428→fn reinit_storage(app: AppHandle, state: State<'_, AppState>) -> Result<serde_json::Value, String> {
   429→    let old_root = paths::game_root();
   430→    let diag = init_storage();
   431→    *STORAGE_DIAG.lock().map_err(|_| "内部状态锁异常")? = Some(diag.clone());
   432→    let new_root = paths::game_root();
   433→    // 游戏运行中不迁移（避免拷贝正在被 JVM 读写的文件）；同时防并发迁移。
   434→    let game_running = state
   435→        .game_pid
   436→        .lock()
   437→        .map_err(|_| "内部状态锁异常")?
   438→        .is_some();
   439→    let mut do_migrate = old_root != new_root && old_root.exists() && !game_running;
   440→    if do_migrate {
   441→        do_migrate = !MIGRATING.swap(true, std::sync::atomic::Ordering::SeqCst);
   442→    }
   443→    if do_migrate {
   444→        let app2 = app.clone();
   445→        std::thread::spawn(move || {
   446→            migrate_root(old_root, new_root, app2);

   619→/// 拉取真实 version_manifest_v2 并排序：release -> snapshot -> old_beta -> old_alpha。
   620→#[tauri::command]
   621→async fn list_versions() -> Result<Vec<VersionInfo>, String> {
   622→    let client = meta::http_client();
   623→    let manifest = meta::fetch_version_manifest(&client)
   624→        .await
   625→        .map_err(|e| format!("版本清单拉取失败：{e}"))?;
   626→    const KIND_ORDER: [&str; 4] = ["release", "snapshot", "old_beta", "old_alpha"];
   627→    let mut out = Vec::with_capacity(manifest.versions.len());
   628→    for kind in KIND_ORDER {
   629→        for v in manifest.versions.iter().filter(|v| v.kind == kind) {
   630→            out.push(VersionInfo {
   631→                id: v.id.clone(),
   632→                version_type: v.kind.clone(),
   633→                url: v.url.clone(),
   634→            });
   635→        }
   636→    }
   637→    for v in manifest
   638→        .versions
   639→        .iter()
   640→        .filter(|v| !KIND_ORDER.contains(&v.kind.as_str()))
   641→    {
   642→        out.push(VersionInfo {
   643→            id: v.id.clone(),
   644→            version_type: v.kind.clone(),
   645→            url: v.url.clone(),
   646→        });
   647→    }
   648→    Ok(out)
   649→}

   672→/// 后台启动完整下载管线（8 并发 + SHA-1 校验 + 3 次重试 + 原子落盘），
   673→/// 进度经 `download://progress` 事件上报 {stage,current,total,percent,file,done,error}。
   674→#[tauri::command]
   675→async fn download_game(
   676→    version_id: String,
   677→    app: AppHandle,
   678→    state: State<'_, AppState>,
   679→) -> Result<(), String> {
   680→    {
   681→        let mut busy = state.downloading.lock().map_err(|_| "内部状态锁异常")?;
   682→        if *busy {
   683→            return Err("已有下载任务正在进行，请等待完成后再试".to_string());
   684→        }
   685→        *busy = true;
   686→    }
   687→    let busy = state.downloading.clone();
   688→    let app = app.clone();
   689→
   690→    tauri::async_runtime::spawn(async move {
   691→        let (tx, mut rx) = tokio::sync::mpsc::channel::<instance::DownloadProgress>(128);
   692→        let forwarder_app = app.clone();
   693→        let forwarder = tokio::spawn(async move {
   694→            while let Some(ev) = rx.recv().await {
   695→                let _ = forwarder_app.emit("download://progress", &ev);
   696→            }
   697→        });
   698→
   699→        let client = meta::http_client();
   700→        let result = instance::install_standard(&client, &version_id, tx).await;
   701→        if let Err(e) = &result {
   702→            // 失败也走事件通道（UI 红字报错 + 可重试），绝不静默。
   703→            let _ = app.emit(
   704→                "download://progress",
   705→                &instance::DownloadProgress {
   706→                    stage: "error".to_string(),
   707→                    current: 0,
   708→                    total: 0,
   709→                    percent: 0.0,
   710→                    file: String::new(),
   711→                    done: false,
   712→                    error: Some(e.to_string()),
   713→                },
   714→            );
   715→        }
   716→        drop(result);
   717→        if let Ok(mut busy) = busy.lock() {
   718→            *busy = false;
   719→        }
   720→        // forwarder 在 tx 关闭后自然退出
   721→        let _ = forwarder.await;
   722→    });
   723→
   724→    Ok(())
   725→}

   752→        .runtime_busy
   753→        .swap(true, std::sync::atomic::Ordering::SeqCst)

   762→        let forwarder_app = app.clone();

   764→            while let Some(ev) = rx.recv().await {

   767→        });

   769→        let client = meta::http_client();

   771→        if let Err(e) = &result {
   772→            let _ = app.emit(
   773→                "runtime://progress",
   774→                &DownloadProgress {
   775→                    stage: "error".to_string(),
   776→                    current: 0,
   777→                    total: 0,
   778→                    percent: 0.0,
   779→                    file: String::new(),
   780→                    done: false,
   781→                    error: Some(e.to_string()),
   782→                },
   783→            );
   784→        }
   785→        drop(result);
   786→        busy.store(false, std::sync::atomic::Ordering::SeqCst);
   787→        let _ = forwarder.await;
   788→    });

   790→    Ok(())
   791→}

   882→/// 拉起 JVM 子进程并托管日志/退出（`launch://log` / `launch://exit` 事件）。
   883→///
   884→/// 前置：JVM pack 已安装（Android 内部目录，targetSdk 28 允许 exec）。
   885→/// smoke test 先跑 `java -version` 把 JRE 版本写进控制台，失败时给出明确报错。
   886→fn spawn_game(
   887→    version_id: String,
   888→    plan: jvmops::LaunchPlan,
   889→    major: u32,
   890→    game_pid: Arc<Mutex<Option<u32>>>,
   891→    app: AppHandle,
   892→) -> Result<(), String> {
   893→    use std::io::BufRead;
   894→    use std::process::{Command, Stdio};
   895→
   896→    let java = runtime::java_bin(major);
   897→    let jre_home = runtime::install_dir(major);
   898→    let game_dir = paths::minecraft_dir();
   899→
   900→    // ---- JVM 动态库搜索路径（必须先于 smoke test 计算，两处共用） ----
   901→    // bionic linker 不认 RUNPATH，bin/java 找 libjli.so 完全依赖 LD_LIBRARY_PATH。
   902→    // 覆盖包内全部可能落点：lib（libjli.so 实测在此）、lib/jli、lib/server、
   903→    // lib/client（arm32 client VM）、natives 目录（游戏 so）。
   904→    let ld_library_path = format!(
   905→        "{}:{}:{}:{}:{}",
   906→        jre_home.join("lib").display(),
   907→        jre_home.join("lib/jli").display(),
   908→        jre_home.join("lib/server").display(),
   909→        jre_home.join("lib/client").display(),
   910→        plan.natives_dir.display()
   911→    );
   912→
   913→    // ---- smoke test：JRE 可执行性（与真实启动完全同环境）----
   914→    // 必须带 LD_LIBRARY_PATH/JAVA_HOME：Pojav android-openjdk 的 bin/java 无 RUNPATH，
   915→    // 裸跑必报 CANNOT LINK。自检失败直接终止启动并回显全部输出（可诊断，不静默）。
   916→    let _ = app.emit(
   917→        "launch://log",
   918→        &LaunchLog {
   919→            source: "quartz".into(),
   920→            line: format!("[quartz] JRE {major}：{}", jre_home.display()),
   921→        },
   922→    );
   923→    match Command::new(&java)
   924→        .arg("-version")
   925→        .env("JAVA_HOME", &jre_home)
   926→        .env("LD_LIBRARY_PATH", &ld_library_path)
   927→        .output()
   928→    {
   929→        Ok(out) if out.status.success() => {
   930→            let text = String::from_utf8_lossy(&out.stderr);
   931→            for line in text.lines().filter(|l| !l.trim().is_empty()).take(2) {
   932→                let _ = app.emit(
   933→                    "launch://log",
   934→                    &LaunchLog {
   935→                        source: "quartz".into(),
   936→                        line: format!("[quartz] {line}"),
   937→                    },
   938→                );
   939→            }
   940→        }
   941→        Ok(out) => {
   942→            // 自检退出码非 0：把完整输出带回给 UI（linker 报错 / JVM 初始化错误都在这里）
   943→            let mut text = String::from_utf8_lossy(&out.stdout).into_owned();
   944→            text.push_str(&String::from_utf8_lossy(&out.stderr));
   945→            let tail: String = text
   946→                .lines()
   947→                .filter(|l| !l.trim().is_empty())
   948→                .collect::<Vec<_>>()
   949→                .join(" | ");
   950→            return Err(format!(
   951→                "JVM 自检失败（退出码 {}，LD_LIBRARY_PATH={}）：{}",
   952→                out.status.code().unwrap_or(-1),
   953→                ld_library_path,
   954→                if tail.is_empty() { "无输出（进程可能被信号杀死）" } else { &tail }
   955→            ));
   956→        }
   957→        Err(e) => {
   958→            return Err(format!(
   959→                "无法执行 JVM（{e}）。JVM pack 需位于可执行目录：{}",
   960→                jre_home.display()
   961→            ))
   962→        }
   963→    }
   964→
   965→    // ---- 组装命令：jvm_args + main_class + game_args ----
   966→    let mut cmd = Command::new(&java);
   967→    cmd.args(&plan.jvm_args)
   968→        .arg(&plan.main_class)
   969→        .args(&plan.game_args)
   970→        .current_dir(&plan.working_dir)
   971→        .env("HOME", &game_dir)
   972→        .env("JAVA_HOME", &jre_home)
   973→        .env("LD_LIBRARY_PATH", &ld_library_path)
   974→        .stdin(Stdio::null())
   975→        .stdout(Stdio::piped())
   976→        .stderr(Stdio::piped());
   977→
   978→    let mut child = cmd
   979→        .spawn()
   980→        .map_err(|e| format!("游戏进程启动失败：{e}"))?;
   981→    let pid = child.id();
   982→    if let Ok(mut slot) = game_pid.lock() {
   983→        *slot = Some(pid);
   984→    }
   985→    {
   986→        let app2 = app.clone();
   987→        let _ = app2.emit(
   988→            "launch://log",
   989→            &LaunchLog {
   990→                source: "quartz".into(),
   991→                line: format!("[quartz] 游戏进程已启动 (pid {pid})，工作目录：{}", plan.working_dir.display()),
   992→            },
   993→        );
   994→    }
   995→
   996→    // ---- 日志泵：stdout / stderr -> launch://log ----
   997→    if let Some(stdout) = child.stdout.take() {
   998→        let app2 = app.clone();
   999→        std::thread::spawn(move || {
  1000→            for line in std::io::BufReader::new(stdout).lines().map_while(Result::ok) {
  1001→                let _ = app2.emit("launch://log", &LaunchLog { source: "stdout".into(), line });
  1002→            }
  1003→        });
  1004→    }
  1005→    if let Some(stderr) = child.stderr.take() {
  1006→        let app2 = app.clone();
  1007→        std::thread::spawn(move || {
  1008→            for line in std::io::BufReader::new(stderr).lines().map_while(Result::ok) {
  1009→                let _ = app2.emit("launch://log", &LaunchLog { source: "stderr".into(), line });
  1010→            }
  1011→        });
  1012→    }
  1013→
  1014→    // ---- 退出看护：wait -> launch://exit -> 清理 pid 槽位 ----
  1015→    // 退出码语义：正常退出 = code；信号死亡 = 128+信号号（137=SIGKILL，134=SIGABRT，
  1016→    // 之前恒报 -1 无法区分 linker abort / SIGSEGV / 被系统杀死）。
  1017→    let game_pid_exit = game_pid;
  1018→    std::thread::spawn(move || {
  1019→        let code = child.wait().map(|st| exit_status_code(&st)).unwrap_or(-1);
  1020→        if let Ok(mut slot) = game_pid_exit.lock() {
  1021→            // stop_game 已 take 时这里已是 None，避免覆盖后续新进程
  1022→            if slot.is_some() {
  1023→                *slot = None;
  1024→            }
  1025→        }
  1026→        // 常见信号死亡的提示行（帮助用户/开发者定位）
  1027→        if code == 137 {
  1028→            let _ = app.emit(
  1029→                "launch://log",
  1030→                &LaunchLog {
  1031→                    source: "quartz".into(),
  1032→                    line: "[quartz] 退出码 137 = SIGKILL：常见于系统低内存回收（尝试调小内存）或 Android 12+ 幽灵进程限制".into(),
  1033→                },
  1034→            );
  1035→        } else if code == 134 {
  1036→            let _ = app.emit(
  1037→                "launch://log",
  1038→                &LaunchLog {
  1039→                    source: "quartz".into(),
  1040→                    line: "[quartz] 退出码 134 = SIGABRT：常见于动态链接失败或 JVM fatal error，见上方 stderr 输出".into(),
  1041→                },
  1042→            );
  1043→        }
  1044→        let _ = app.emit(
  1045→            "launch://exit",
  1046→            &LaunchExit { code, version: version_id },
  1047→        );
  1048→    });
  1049→    Ok(())
  1050→}
  1051→
  1052→/// ExitStatus -> 退出码：正常退出取 code()；信号死亡取 128+signal。
  1053→fn exit_status_code(st: &std::process::ExitStatus) -> i32 {
  1054→    #[cfg(unix)]
  1055→    {
  1056→        use std::os::unix::process::ExitStatusExt;
  1057→        st.code()
  1058→            .or_else(|| st.signal().map(|sig| 128 + sig))
  1059→            .unwrap_or(-1)
  1060→    }
  1061→    #[cfg(not(unix))]
  1062→    {
  1063→        st.code().unwrap_or(-1)
  1064→    }
  1065→}
  1066→
  1067→/// 真实启动：版本 JSON -> LaunchPlan -> JVM 子进程（M0 已接入）。
  1068→/// JVM pack 缺失时返回 `RUNTIME_MISSING:<major>`，UI 自动触发运行时下载。
  1069→#[tauri::command]
  1070→fn launch_game(version_id: String, app: AppHandle, state: State<'_, AppState>) -> Result<(), String> {
  1071→    if state.game_pid.lock().map_err(|_| "内部状态锁异常")?.is_some() {
  1072→        return Err("已有游戏进程在运行，请先停止后再启动".to_string());
  1073→    }
  1074→    let (_version, plan, major) = build_plan_for(&version_id)?;
  1075→    if !runtime::is_installed(major) {
  1076→        return Err(format!("RUNTIME_MISSING:{major}"));
  1077→    }
  1078→    std::fs::create_dir_all(&plan.working_dir).ok();
  1079→    std::fs::create_dir_all(&plan.natives_dir).ok();
  1080→    spawn_game(version_id.clone(), plan, major, state.game_pid.clone(), app.clone())?;
  1081→    Ok(())
  1082→}

  1084→/// 终止运行中的游戏进程（SIGKILL）。
  1085→#[tauri::command]
  1086→fn stop_game(state: State<'_, AppState>) -> Result<bool, String> {
  1087→    let pid = state
  1088→        .game_pid
  1089→        .lock()
  1090→        .map_err(|_| "内部状态锁异常")?
  1091→        .take();
  1092→    match pid {
  1093→        Some(pid) => {
  1094→            #[cfg(unix)]
  1095→            unsafe {
  1096→                libc::kill(pid as libc::pid_t, libc::SIGKILL);
  1097→            }
  1098→            Ok(true)
  1099→        }
  1100→        None => Ok(false),
  1101→    }
  1102→}
