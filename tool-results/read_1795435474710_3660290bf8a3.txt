    99→void pojavPumpEvents(void* window) {
   100→    if(pojav_environ->shouldUpdateMouse) {
   101→        pojav_environ->GLFW_invoke_CursorPos(window, floor(pojav_environ->cursorX),
   102→                                             floor(pojav_environ->cursorY));
   103→    }
   104→    if(pojav_environ->shouldUpdateMonitorSize) {
   105→        updateWindowSize(window);
   106→    }
   107→
   108→    size_t index = pojav_environ->outEventIndex;
   109→    size_t targetIndex = pojav_environ->outTargetIndex;
   110→
   111→    while (targetIndex != index) {
   112→        GLFWInputEvent event = pojav_environ->events[index];
   113→        switch (event.type) {
   114→            case EVENT_TYPE_CHAR:
   115→                if(pojav_environ->GLFW_invoke_Char) pojav_environ->GLFW_invoke_Char(window, event.i1);
   116→                break;
   117→            case EVENT_TYPE_CHAR_MODS:
   118→                if(pojav_environ->GLFW_invoke_CharMods) pojav_environ->GLFW_invoke_CharMods(window, event.i1, event.i2);
   119→                break;
   120→            case EVENT_TYPE_KEY:
   121→                if(pojav_environ->GLFW_invoke_Key) pojav_environ->GLFW_invoke_Key(window, event.i1, event.i2, event.i3, event.i4);
   122→                break;
   123→            case EVENT_TYPE_MOUSE_BUTTON:
   124→                if(pojav_environ->GLFW_invoke_MouseButton) pojav_environ->GLFW_invoke_MouseButton(window, event.i1, event.i2, event.i3);
   125→                break;
   126→            case EVENT_TYPE_SCROLL:
   127→                if(pojav_environ->GLFW_invoke_Scroll) pojav_environ->GLFW_invoke_Scroll(window, event.i1, event.i2);
   128→                break;
   129→        }
   130→
   131→        index++;
   132→        if (index >= EVENT_WINDOW_SIZE)
   133→            index -= EVENT_WINDOW_SIZE;
   134→    }
   135→
   136→    // The out target index is updated by the rewinder
   137→}

   139→/** Prepare the library for sending out callbacks to all windows */
   140→void pojavStartPumping() {
   141→    size_t counter = atomic_load_explicit(&pojav_environ->eventCounter, memory_order_acquire);
   142→    size_t index = pojav_environ->outEventIndex;
   143→
   144→    unsigned targetIndex = index + counter;
   145→    if (targetIndex >= EVENT_WINDOW_SIZE)
   146→        targetIndex -= EVENT_WINDOW_SIZE;
   147→
   148→    // Only accessed by one unique thread, no need for atomic store
   149→    pojav_environ->inEventCount = counter;
   150→    pojav_environ->outTargetIndex = targetIndex;
   151→
   152→    //PumpEvents is called for every window, so this logic should be there in order to correctly distribute events to all windows.
   153→    if((pojav_environ->cLastX != pojav_environ->cursorX || pojav_environ->cLastY != pojav_environ->cursorY) && pojav_environ->GLFW_invoke_CursorPos) {
   154→        pojav_environ->cLastX = pojav_environ->cursorX;
   155→        pojav_environ->cLastY = pojav_environ->cursorY;
   156→        pojav_environ->shouldUpdateMouse = true;
   157→    }
   158→    if(pojav_environ->shouldUpdateMonitorSize) {
   159→        // Perform a monitor size update here to avoid doing it on every single window
   160→        updateMonitorSize(pojav_environ->savedWidth, pojav_environ->savedHeight);
   161→        // Mark the monitor size as consumed (since GLFW was made aware of it)
   162→        pojav_environ->monitorSizeConsumed = true;
   163→    }
   164→}
   165→
   166→/** Prepare the library for the next round of new events */
   167→void pojavStopPumping() {
   168→    pojav_environ->outEventIndex = pojav_environ->outTargetIndex;
   169→
   170→    // New events may have arrived while pumping, so remove only the difference before the start and end of execution
   171→    atomic_fetch_sub_explicit(&pojav_environ->eventCounter, pojav_environ->inEventCount, memory_order_acquire);
   172→    // Make sure the next frame won't send mouse or monitor updates if it's unnecessary
   173→    pojav_environ->shouldUpdateMouse = false;
   174→    // Only reset the update flag if the monitor size was consumed by pojavStartPumping. This
   175→    // will delay the update to next frame if it had occured between pojavStartPumping and pojavStopPumping,
   176→    // but it's better than not having it apply at all
   177→    if(pojav_environ->shouldUpdateMonitorSize && pojav_environ->monitorSizeConsumed) {
   178→        pojav_environ->shouldUpdateMonitorSize = false;
   179→        pojav_environ->monitorSizeConsumed = false;
   180→    }
   181→
   182→}

   217→void sendData(int type, int i1, int i2, int i3, int i4) {
   218→    GLFWInputEvent *event = &pojav_environ->events[pojav_environ->inEventIndex];
   219→    event->type = type;
   220→    event->i1 = i1;
   221→    event->i2 = i2;
   222→    event->i3 = i3;
   223→    event->i4 = i4;
   224→
   225→    if (++pojav_environ->inEventIndex >= EVENT_WINDOW_SIZE)
   226→        pojav_environ->inEventIndex -= EVENT_WINDOW_SIZE;
   227→
   228→    atomic_fetch_add_explicit(&pojav_environ->eventCounter, 1, memory_order_acquire);
   229→}

   231→/**
   232→ * This function is meant as a substitute for SharedLibraryUtil.getLibraryPath() that just returns 0
   233→ * (thus making the parent Java function return null). This is done to avoid using the LWJGL's default function,
   234→ * which will hang the crappy EMUI linker by dlopen()ing inside of dl_iterate_phdr().
   235→ * @return 0, to make the parent Java function return null immediately.
   236→ * For reference: https://github.com/PojavLauncherTeam/lwjgl3/blob/fix_huawei_hang/modules/lwjgl/core/src/main/java/org/lwjgl/system/SharedLibraryUtil.java
   237→ */
   238→jint getLibraryPath_fix(__attribute__((unused)) JNIEnv *env,
   239→                        __attribute__((unused)) jclass class,
   240→                        __attribute__((unused)) jlong pLibAddress,
   241→                        __attribute__((unused)) jlong sOutAddress,
   242→                        __attribute__((unused)) jint bufSize){
   243→    return 0;
   244→}

   246→/**
   247→ * Install the linker hang mitigation that is meant to prevent linker hangs on old EMUI firmware.
   248→ */
   249→void installEMUIIteratorMititgation() {
   250→    if(getenv("POJAV_EMUI_ITERATOR_MITIGATE") == NULL) return;
   251→    __android_log_print(ANDROID_LOG_INFO, "EMUIIteratorFix", "Installing...");
   252→    JNIEnv* env = pojav_environ->runtimeJNIEnvPtr_JRE;
   253→    jclass sharedLibraryUtil = (*env)->FindClass(env, "org/lwjgl/system/SharedLibraryUtil");
   254→    if(sharedLibraryUtil == NULL) {
   255→        __android_log_print(ANDROID_LOG_ERROR, "EMUIIteratorFix", "Failed to find the target class");
   256→        (*env)->ExceptionClear(env);
   257→        return;
   258→    }
   259→    JNINativeMethod getLibraryPathMethod[] = {
   260→            {"getLibraryPath", "(JJI)I", &getLibraryPath_fix}
   261→    };
   262→    if((*env)->RegisterNatives(env, sharedLibraryUtil, getLibraryPathMethod, 1) != 0) {
   263→        __android_log_print(ANDROID_LOG_ERROR, "EMUIIteratorFix", "Failed to register the mitigation method");
   264→        (*env)->ExceptionClear(env);
   265→    }
   266→}

   276→JNIEXPORT jstring JNICALL Java_org_lwjgl_glfw_CallbackBridge_nativeClipboard(JNIEnv* env, __attribute__((unused)) jclass clazz, jint action, jbyteArray copySrc) {
   277→#ifdef DEBUG
   278→    LOGD("Debug: Clipboard access is going on\n", pojav_environ->isUseStackQueueCall);
   279→#endif
   280→
   281→    JNIEnv *dalvikEnv;
   282→    (*pojav_environ->dalvikJavaVMPtr)->AttachCurrentThread(pojav_environ->dalvikJavaVMPtr, &dalvikEnv, NULL);
   283→    assert(dalvikEnv != NULL);
   284→    assert(pojav_environ->bridgeClazz != NULL);
   285→    
   286→    LOGD("Clipboard: Converting string\n");
   287→    char *copySrcC;
   288→    jstring copyDst = NULL;
   289→    if (copySrc) {
   290→        copySrcC = (char *)((*env)->GetByteArrayElements(env, copySrc, NULL));
   291→        copyDst = (*dalvikEnv)->NewStringUTF(dalvikEnv, copySrcC);
   292→    }
   293→
   294→    LOGD("Clipboard: Calling 2nd\n");
   295→    jstring pasteDst = convertStringJVM(dalvikEnv, env, (jstring) (*dalvikEnv)->CallStaticObjectMethod(dalvikEnv, pojav_environ->bridgeClazz, pojav_environ->method_accessAndroidClipboard, action, copyDst));
   296→
   297→    if (copySrc) {
   298→        (*dalvikEnv)->DeleteLocalRef(dalvikEnv, copyDst);    
   299→        (*env)->ReleaseByteArrayElements(env, copySrc, (jbyte *)copySrcC, 0);
   300→    }
   301→    (*pojav_environ->dalvikJavaVMPtr)->DetachCurrentThread(pojav_environ->dalvikJavaVMPtr);
   302→    return pasteDst;
   303→}

   365→void critical_send_cursor_pos(jfloat x, jfloat y) {
   366→#ifdef DEBUG
   367→    LOGD("Sending cursor position \n");
   368→#endif
   369→    if (pojav_environ->GLFW_invoke_CursorPos && pojav_environ->isInputReady) {
   370→#ifdef DEBUG
   371→        LOGD("pojav_environ->GLFW_invoke_CursorPos && pojav_environ->isInputReady \n");
   372→#endif
   373→        if (!pojav_environ->isCursorEntered) {
   374→            if (pojav_environ->GLFW_invoke_CursorEnter) {
   375→                pojav_environ->isCursorEntered = true;
   376→                if (pojav_environ->isUseStackQueueCall) {
   377→                    sendData(EVENT_TYPE_CURSOR_ENTER, 1, 0, 0, 0);
   378→                } else {
   379→                    pojav_environ->GLFW_invoke_CursorEnter((void*) pojav_environ->showingWindow, 1);
   380→                }
   381→            } else if (pojav_environ->isGrabbing) {
   382→                // Some Minecraft versions does not use GLFWCursorEnterCallback
   383→                // This is a smart check, as Minecraft will not in grab mode if already not.
   384→                pojav_environ->isCursorEntered = true;
   385→            }
   386→        }
   387→
   388→        if (!pojav_environ->isUseStackQueueCall) {
   389→            pojav_environ->GLFW_invoke_CursorPos((void*) pojav_environ->showingWindow, (double) (x), (double) (y));
   390→        } else {
   391→            pojav_environ->cursorX = x;
   392→            pojav_environ->cursorY = y;
   393→        }
   394→    }
   395→}

   471→    // Check for stack queue no longer necessary here as the JVM crash's origin is resolved
   472→    if (!pojav_environ->showingWindow) {
   473→        // If the window is not shown, there is nothing to do yet.
   474→        return;
   475→    }
   476→
   477→    // We cannot use pojav_environ->runtimeJNIEnvPtr_JRE here because that environment is attached
   478→    // on the thread that loaded pojavexec (which is the thread that first references the GLFW class)
   479→    // But this method is only called from the Android UI thread
   480→
   481→    // Technically the better solution would be to have a permanently attached env pointer stored
   482→    // in environ for the Android UI thread but this is the only place that uses it
   483→    // (very rarely, only in lifecycle callbacks) so i dont care
   484→
   485→    JavaVM* jvm = pojav_environ->runtimeJavaVMPtr;
   486→    JNIEnv *jvm_env = NULL;
   487→    jint env_result = (*jvm)->GetEnv(jvm, (void**)&jvm_env, JNI_VERSION_1_4);
   488→    if(env_result == JNI_EDETACHED) {
   489→        env_result = (*jvm)->AttachCurrentThread(jvm, &jvm_env, NULL);
   490→    }

   492→        printf("input_bridge nativeSetWindowAttrib() JNI call failed: %i\n", env_result);

   497→            jvm_env, pojav_environ->vmGlfwClass,
   498→            pojav_environ->method_glftSetWindowAttrib,
   499→            (jlong) pojav_environ->showingWindow, attrib, value
   500→    );

   539→static bool tryCriticalNative(JNIEnv *env) {
   540→    static const JNINativeMethod testJNIMethod[] = {
   541→            { "testCriticalNative", "(II)V", dvm_testCriticalNative}
   542→    };
   543→    jclass criticalNativeTest = (*env)->FindClass(env, "net/kdt/pojavlaunch/CriticalNativeTest");
   544→    if(criticalNativeTest == NULL) {
   545→        LOGD("No CriticalNativeTest class found !");
   546→        (*env)->ExceptionClear(env);
   547→        return false;
   548→    }
   549→    jmethodID criticalNativeTestMethod = (*env)->GetStaticMethodID(env, criticalNativeTest, "invokeTest", "()V");
   550→    (*env)->RegisterNatives(env, criticalNativeTest, testJNIMethod, 1);
   551→    (*env)->CallStaticVoidMethod(env, criticalNativeTest, criticalNativeTestMethod);
   552→    (*env)->UnregisterNatives(env, criticalNativeTest);
   553→    return criticalNativeAvailable;
   554→}

   556→static void registerFunctions(JNIEnv *env) {
   557→    bool use_critical_cc = tryCriticalNative(env);
   558→    jclass bridge_class = (*env)->FindClass(env, "org/lwjgl/glfw/CallbackBridge");
   559→    if(use_critical_cc) {
   560→        __android_log_print(ANDROID_LOG_INFO, "pojavexec", "CriticalNative is available. Enjoy the 4.6x times faster input!");
   561→    }else{
   562→        __android_log_print(ANDROID_LOG_INFO, "pojavexec", "CriticalNative is not available. Upgrade, maybe?");
   563→    }
   564→    (*env)->RegisterNatives(env,
   565→                            bridge_class,
   566→                            use_critical_cc ? critical_fcns : noncritical_fcns,
   567→                            sizeof(critical_fcns)/sizeof(critical_fcns[0]));
   568→}
