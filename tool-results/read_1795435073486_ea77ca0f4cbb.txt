     1→//! Quartz Tauri 壳层。
     2→//!
     3→//! 架构契约（方案 A）：UI（WebView/JS）只调用本文件注册的 command；

     5→//!
     6→//! 命令契约（v3 蓝图 + M0 启动接入）：
     7→//! - `offline_login(name)`        离线登录：MD5("OfflinePlayer:"+name) -> accounts.json
     8→//! - `get_account()`              当前选中账号
     9→//! - `list_accounts()`            全部账号 + 选中标记
    10→//! - `select_account(uuid)`       切换选中账号
    11→//! - `remove_account(uuid)`       删除账号
    12→//! - `list_versions()`            真实 manifest 版本列表（release 优先）
    13→//! - `list_installed_versions()`  本地 versions/ 已安装版本
    14→//! - `download_game(version_id)`  后台 spawn 完整下载管线，
    15→//!                                经 `download://progress` 事件上报 {stage,current,total,percent,file}
    16→//! - `get_game_dir()`             标准游戏根目录绝对路径（UI 常显）
    17→//! - `get_env_info()`             存储诊断：根目录 + 可写性 + 存储模式 + 完整文件权限
    18→//! - `get_permission_state()`     存储权限状态 {granted}

    21→//!                                （`quartz://migrate` 事件上报迁移进度）
    22→//! - `launch_preflight(version_id)` 启动前自检（版本/账号/Java 运行时）

    56→#[cfg(target_os = "android")]
    57→fn with_jni<T>(
    58→    f: impl FnOnce(&mut jni::JNIEnv, &jni::objects::JObject) -> Option<T>,
    59→) -> Option<T> {
    60→    std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
    61→        let ctx = ndk_context::android_context();
    62→        let vm = unsafe { jni::JavaVM::from_raw(ctx.vm().cast()) }.ok()?;
    63→        let mut env = vm.attach_current_thread().ok()?;
    64→        let context = unsafe { jni::objects::JObject::from_raw(ctx.context().cast()) };
    65→        // 防御：清掉上游遗留的挂起异常。JNIEnv 按线程复用，任何未清理的异常都会让
    66→        // 本闭包内的下一个 JNI 调用（甚至 Tauri 内部调用）在 CheckJNI 下直接 abort。
    67→        if env.exception_check().unwrap_or(false) {
    68→            let _ = env.exception_clear();
    69→        }
    70→        let result = f(&mut env, &context);
    71→        // 关键：闭包内任何 JNI Err（如 NoSuchMethodError）都会把异常留在挂起态，
    72→        // 若不清掉，下一次 NewStringUTF 等调用即触发
    73→        // "JNI NewStringUTF called with pending exception" abort 闪退。
    74→        if env.exception_check().unwrap_or(false) {
    75→            let _ = env.exception_clear();
    76→        }
    77→        result
    78→    }))
    79→    .ok()
    80→    .flatten()
    81→}

    83→#[cfg(target_os = "android")]
    84→fn jni_file_path(env: &mut jni::JNIEnv, file: &jni::objects::JObject) -> Option<PathBuf> {
    85→    let path = env
    86→        .call_method(file, "getAbsolutePath", "()Ljava/lang/String;", &[])
    87→        .ok()?
    88→        .l()
    89→        .ok()?;
    90→    let s: String = env.get_string((&path).into()).ok()?.into();
    91→    if s.is_empty() {
    92→        None
    93→    } else {
    94→        Some(PathBuf::from(s))
    95→    }
    96→}

    98→#[cfg(target_os = "android")]
    99→fn jni_external_files_dir(env: &mut jni::JNIEnv, context: &jni::objects::JObject) -> Option<PathBuf> {
   100→    use jni::objects::JValue;
   101→    let file = env
   102→        .call_method(
   103→            context,
   104→            "getExternalFilesDir",
   105→            "(Ljava/lang/String;)Ljava/io/File;",
   106→            &[JValue::Object(&jni::objects::JObject::null())],
   107→        )
   108→        .ok()?
   109→        .l()
   110→        .ok()?;
   111→    if file.is_null() {
   112→        return None;
   113→    }
   114→    jni_file_path(env, &file)
   115→}

   123→        .ok()?;
   124→    if file.is_null() {

   130→// ---------------------------------------------------------------------------
   131→// 存储权限（完整文件权限）
   132→// ---------------------------------------------------------------------------
   133→//
   134→// 设计：APK 固定 targetSdk 28（Android 10 前的存储模型 + 允许 exec() 应用目录
   135→// 内的 JVM —— Pojav/Termux 同款方案）。
   136→//
   137→// 授权通道按系统版本分流（MainActivity.requestStorageAccess）:
   138→// - Android 11+（API 30+）：MANAGE_EXTERNAL_STORAGE「所有文件访问」系统设置页。
   139→//   不走 WRITE 弹窗的原因：targetSdk<=29 应用在 Android 11+ 经弹窗授予存储权限时，
   140→//   系统会杀掉应用进程刷新 FUSE 挂载视图 —— 用户视角即「授权后闪退」。
   141→//   所有文件访问页授权不杀进程，返回后 focus 重检即可切换标准目录。
   142→// - Android 10 及以下（API <= 29）：READ/WRITE_EXTERNAL_STORAGE 运行时弹窗
   143→//   （旧存储模型，授权即时生效不杀进程）。
   144→// 游戏文件即可放在文件管理器可见的标准目录 /storage/emulated/0/Quartz/files。
   145→
   146→/// Android 端存储权限是否已授权（纯系统 API，不依赖 Activity 实例）。
   147→///
   148→/// 注意：ndk-context 给到的 context 是 **Application** 对象（不是 MainActivity），
   149→/// 其上不存在任何自定义方法 —— 在它上面 GetMethodID("isStorageAccessGranted")
   150→/// 必抛 NoSuchMethodError。故全部改用静态方法 / Context 自有方法：
   151→/// - Android 11+（API 30+）：静态 Environment.isExternalStorageManager()
   152→///   （MANAGE_EXTERNAL_STORAGE「所有文件访问」）
   153→/// - Android 10 及以下：Context.checkSelfPermission(WRITE_EXTERNAL_STORAGE)
   154→///   （targetSdk 28 旧存储模型，Application 即 Context，可直接调）
   155→#[cfg(target_os = "android")]
   156→fn android_storage_permission_granted() -> bool {
   157→    use jni::objects::JValue;
   158→    with_jni(|env, ctx| {
   159→        let sdk = env
   160→            .get_static_field("android/os/Build$VERSION", "SDK_INT", "I")
   161→            .ok()?
   162→            .i()
   163→            .ok()?;
   164→        if sdk >= 30 {
   165→            let granted = env
   166→                .call_static_method(
   167→                    "android/os/Environment",
   168→                    "isExternalStorageManager",
   169→                    "()Z",
   170→                    &[],
   171→                )
   172→                .ok()?
   173→                .z()
   174→                .ok()?;
   175→            Some(granted)
   176→        } else {
   177→            let perm = env
   178→                .new_string("android.permission.WRITE_EXTERNAL_STORAGE")
   179→                .ok()?;
   180→            let granted = env
   181→                .call_method(
   182→                    ctx,
   183→                    "checkSelfPermission",
   184→                    "(Ljava/lang/String;)I",
   185→                    &[JValue::Object(&perm.into())],
   186→                )
   187→                .ok()?
   188→                .i()
   189→                .ok()?;
   190→            // PackageManager.PERMISSION_GRANTED == 0
   191→            Some(granted == 0)
   192→        }
   193→    })
   194→    .unwrap_or(false)
   195→}
   196→
   197→/// 存储权限是否已授权（桌面端恒真）。
   198→fn storage_permission_granted() -> bool {
   199→    #[cfg(target_os = "android")]
   200→    {
   201→        android_storage_permission_granted()
   202→    }
   203→    #[cfg(not(target_os = "android"))]
   204→    {
   205→        true
   206→    }
   207→}

   215→/// 申请完整文件权限：Android 11+ 跳系统「所有文件访问」设置页（授权不杀进程），
   216→/// Android 10- 触发运行时弹窗。结果不通过本命令返回：UI 在 focus/visibilitychange
   217→/// 时重新查询状态（回到应用后自动重检）。
   218→#[tauri::command]
   219→fn request_storage_permission() -> Result<bool, String> {
   220→    #[cfg(target_os = "android")]
   221→    {
   222→        // MainActivity 伴生对象静态方法（GetStaticMethodID，不依赖接收对象类型）：
   223→        // 内部持有当前 Activity 实例，按系统版本分流（11+「所有文件访问」设置页 /
   224→        // 10- 运行时弹窗）。此前直接在 ndk-context 的 Application 对象上调实例方法
   225→        // requestStorageAccess —— 其上查无此方法，NoSuchMethodError 未清理即 abort。
   226→        let via_kotlin = with_jni(|env, ctx| {
   227→            use jni::objects::JClass;
   228→            use jni::objects::JValue;
   229→            // 注意：FindClass 在 AttachCurrentThread 的原生线程上走系统类加载器，
   230→            // 解析不到 APK 内的应用类；必须经 context.getClassLoader().loadClass()
   231→            // 显式解析壳层 MainActivity（Application 即 Context，链路稳定）。
   232→            let loader = env
   233→                .call_method(ctx, "getClassLoader", "()Ljava/lang/ClassLoader;", &[])
   234→                .ok()?
   235→                .l()
   236→                .ok()?;
   237→            let name = env
   238→                .new_string("com.quartz.launcher.MainActivity")
   239→                .ok()?;
   240→            let cls_obj = env
   241→                .call_method(
   242→                    &loader,
   243→                    "loadClass",
   244→                    "(Ljava/lang/String;)Ljava/lang/Class;",
   245→                    &[JValue::Object(&name.into())],
   246→                )
   247→                .ok()?
   248→                .l()
   249→                .ok()?;
   250→            let cls = JClass::from(cls_obj);
   251→            env.call_static_method(&cls, "requestStorageAccessStatic", "()V", &[])
   252→                .ok()?;
   253→            Some(())
   254→        });
   255→        if via_kotlin.is_some() {
   256→            return Ok(true);
   257→        }
   258→        // 到这里说明 JNI 链路本身异常（with_jni 已清异常）；给用户明确的手动指引。
   259→        Err("无法唤起系统权限界面。请手动授予：系统设置 → 应用 → Quartz → 权限 → 所有文件访问权限".to_string())
   260→    }
   261→    #[cfg(not(target_os = "android"))]
   262→    {
   263→        Ok(true)
   264→    }
   265→}

   267→/// Android 方向切换：进入游戏 → 横屏，游戏退出 → 竖屏。
   268→/// 与权限申请同链路（MainActivity 伴生对象静态方法转发到当前 Activity 实例；
   269→/// ndk-context 的 context 是 Application，其上无实例方法）。失败静默：
   270→/// 方向切换绝不阻塞/中止游戏启动。
   271→#[cfg(target_os = "android")]
   272→fn android_set_orientation(landscape: bool) {
   273→    let ok = with_jni(|env, ctx| {
   274→        use jni::objects::{JClass, JValue};
   275→        let loader = env
   276→            .call_method(ctx, "getClassLoader", "()Ljava/lang/ClassLoader;", &[])
   277→            .ok()?
   278→            .l()
   279→            .ok()?;
   280→        let name = env
   281→            .new_string("com.quartz.launcher.MainActivity")
   282→            .ok()?;
   283→        let cls_obj = env
   284→            .call_method(
   285→                &loader,
   286→                "loadClass",
   287→                "(Ljava/lang/String;)Ljava/lang/Class;",
   288→                &[JValue::Object(&name.into())],
   289→            )
   290→            .ok()?
   291→            .l()
   292→            .ok()?;
   293→        let cls = JClass::from(cls_obj);
   294→        env.call_static_method(
   295→            &cls,
   296→            "setGameOrientationStatic",
   297→            "(Z)V",
   298→            &[JValue::Bool(u8::from(landscape))],
   299→        )
   300→        .ok()?;
   301→        Some(())
   302→    });
   303→    if ok.is_none() {
   304→        eprintln!("[quartz] 方向切换 JNI 调用失败（忽略，不影响启动）");
   305→    }
   306→}

   326→/// Android 平台解析真实可写目录；ndk-context 未初始化等异常一律回退硬编码路径。
   327→///
   328→/// 优先级（与授权状态联动）：
   329→/// 1. 已授权完整文件权限 -> 公共标准目录 /storage/emulated/0/Quartz/files（可见、卸载不丢）
   330→/// 2. 应用外部私有目录（getExternalFilesDir，可浏览，卸载丢失）
   331→/// 3. 应用内部私有目录（getFilesDir，必定可写）
   332→#[cfg(target_os = "android")]
   333→fn resolve_android_root() -> (PathBuf, &'static str) {
   334→    if storage_permission_granted() {
   335→        let public = PathBuf::from("/storage/emulated/0/Quartz/files");
   336→        if std::fs::create_dir_all(&public).is_ok() {
   337→            return (public, "public");
   338→        }
   339→        eprintln!("[quartz] 公共标准目录创建失败，回退应用私有目录");
   340→    }
   341→    let result = with_jni(|env, context| {
   342→        // 外部私有目录（用户可用文件管理器查看 .minecraft）
   343→        if let Some(p) = jni_external_files_dir(env, context) {
   344→            return Some((p, "external"));
   345→        }
   346→        // 内部私有目录（必定可写）
   347→        jni_internal_files_dir(env, context).map(|p| (p, "internal"))
   348→    });
   349→    result.unwrap_or_else(|| {
   350→        (
   351→            PathBuf::from(format!(
   352→                "/storage/emulated/0/Android/data/{}/files",
   353→                paths::ANDROID_PACKAGE_ID
   354→            )),
   355→            "fallback",
   356→        )
   357→    })
   358→}

   360→/// Android 内部私有目录（Context.getFilesDir()，任何设备必定可写；exec 可用）。
   361→#[cfg(target_os = "android")]
   362→fn resolve_android_internal() -> Option<PathBuf> {
   363→    with_jni(|env, context| jni_internal_files_dir(env, context))
   364→}

   366→/// 初始化游戏根目录与 JVM pack 目录并创建标准布局；返回诊断信息。
   367→fn init_storage() -> StorageDiag {
   368→    #[cfg(target_os = "android")]
   369→    let (root, mode) = resolve_android_root();
   370→    #[cfg(not(target_os = "android"))]
   371→    let (root, mode) = (paths::game_root(), "desktop");
   372→
   373→    let _ = paths::set_runtime_root(root.clone());
   374→    let mut layout = paths::ensure_layout();
   375→
   376→    // 外部目录不可写（EACCES / os error 13）时回退内部私有目录，
   377→    // 保证离线登录与下载在任何设备上永远可用。
   378→    #[cfg(target_os = "android")]
   379→    let mut mode = mode;
   380→    #[cfg(target_os = "android")]
   381→    if layout.is_err() && mode != "internal" {
   382→        eprintln!(
   383→            "[quartz] 当前根目录不可写（{:?}），回退内部存储",
   384→            layout.as_ref().err()
   385→        );
   386→        if let Some(internal) = resolve_android_internal() {
   387→            let _ = paths::set_runtime_root(internal.clone());
   388→            layout = paths::ensure_layout();
   389→            mode = "internal";
   390→        }
   391→    }
   392→
   393→    // JVM pack 必须放在可 exec 的内部目录（FUSE noexec + targetSdk>=29 禁 exec 的约束）
   394→    #[cfg(target_os = "android")]
   395→    if let Some(internal) = resolve_android_internal() {
   396→        paths::set_runtime_jres_root(internal.join("jres"));
   397→    }
   398→
   399→    let diag = StorageDiag {
   400→        mode,
   401→        root: paths::game_root(),
   402→        layout_ok: layout.is_ok(),
   403→        layout_error: layout.err().map(|e| format!("{e}")),
   404→    };
   405→    eprintln!(
   406→        "[quartz] 存储初始化：mode={} root={} jres={}",
   407→        diag.mode,
   408→        diag.root.display(),
   409→        paths::runtime_dir().display()
   410→    );
   411→    diag
   412→}

   426→// 崩溃证据落盘 + 启动自愈（v8「一打开就闪退」加固）
   427→// ---------------------------------------------------------------------------
   428→//
   429→// 「闪退」的三类根因都在这里落证据 / 自愈：
   430→// 1. Rust panic（任意线程）：panic hook 追加标记到唯一崩溃日志，下次启动 UI 横幅
   431→//    直接展示原因 —— 不再是黑盒闪退；
   432→// 2. Java/Kotlin 未捕获异常：QuartzApp（gen/android）同一文件追加 JAVA 标记；
   433→// 3. WebView 状态损坏导致的打开即崩：QuartzApp 检测 2 分钟内 ≥2 次崩溃，
   434→//    在 WebView 创建前清理 app_webview（Chromium profile）自愈。
   435→//
   436→// 另：失败下载每次重试都留下大文件（.qz-part/.part/.staging，PoJav APK 154MB /

   441→/// 崩溃日志唯一权威路径（run() 最早阶段解析；panic hook 与 get_diagnostics 共用）。
   442→/// Android 用内部私有目录（必定可写，无需任何权限）。
   443→static CRASH_LOG: std::sync::OnceLock<PathBuf> = std::sync::OnceLock::new();
   444→
   445→fn crash_log_path() -> PathBuf {
   446→    #[cfg(target_os = "android")]
   447→    {
   448→        if let Some(p) = resolve_android_internal() {
   449→            return p.join("crash").join("quartz-crash.log");
   450→        }
   451→        // JNI 不可用时的壳层回退（应用私有目录固定路径）
   452→        PathBuf::from("/data/data/com.quartz.launcher/files/crash/quartz-crash.log")
   453→    }
   454→    #[cfg(not(target_os = "android"))]
   455→    {
   456→        paths::game_root().join("logs").join("quartz-crash.log")
   457→    }

   460→/// 追加一条崩溃标记。格式（Rust/Java 两端约定，QuartzApp.kt 同款解析）：
   461→/// `==== CRASH @<epoch秒> <KIND> ====` 换行正文。失败静默（panic hook 内不能报错）。
   462→fn append_crash_marker(kind: &str, body: &str) {
   463→    let Some(path) = CRASH_LOG.get() else { return };
   464→    if let Some(parent) = path.parent() {
   465→        let _ = std::fs::create_dir_all(parent);
   466→    }
   467→    let epoch = std::time::SystemTime::now()
   468→        .duration_since(std::time::UNIX_EPOCH)
   469→        .map(|d| d.as_secs())
   470→        .unwrap_or(0);
   471→    use std::io::Write as _;
   472→    let text = format!("==== CRASH @{epoch} {kind} ====\n{body}\n");
   473→    if let Ok(mut f) = std::fs::OpenOptions::new().create(true).append(true).open(path) {
   474→        let _ = writeln!(f, "{text}");
   475→        let _ = f.flush();
   476→    }
   477→    // 授权时同步一份到公共目录：用户可用文件管理器直接取出日志（不依赖 UI）
   478→    #[cfg(target_os = "android")]
   479→    {
   480→        if storage_permission_granted() {
   481→            let public = paths::game_root().join("logs").join("quartz-crash.log");
   482→            if let Some(parent) = public.parent() {
   483→                let _ = std::fs::create_dir_all(parent);
   484→            }
   485→            if let Ok(mut f) = std::fs::OpenOptions::new().create(true).append(true).open(public) {
   486→                let _ = writeln!(f, "{text}");
   487→            }
   488→        }
   489→    }
   490→}

   492→/// 安装 panic hook：Rust panic（任意线程）先落盘再交回默认打印（logcat 保留）。
   493→/// 必须在 tauri::Builder 构建前调用 —— setup 钩子/主线程 panic 由此才有证据。
   494→fn install_panic_hook() {
   495→    let _ = CRASH_LOG.set(crash_log_path());
   496→    let default_hook = std::panic::take_hook();
   497→    std::panic::set_hook(Box::new(move |info| {
   498→        let location = info
   499→            .location()
   500→            .map(|l| format!("{}:{}:{}", l.file(), l.line(), l.column()))
   501→            .unwrap_or_else(|| "未知位置".into());
   502→        let msg = info
   503→            .payload()
   504→            .downcast_ref::<&str>()
   505→            .map(|s| s.to_string())
   506→            .or_else(|| info.payload().downcast_ref::<String>().cloned())
   507→            .unwrap_or_else(|| "非字符串 panic 载荷".into());
   508→        let thread = std::thread::current()
   509→            .name()
   510→            .unwrap_or("<unnamed>")
   511→            .to_string();
   512→        let bt = std::backtrace::Backtrace::force_capture().to_string();
   513→        let bt_tail = bt
   514→            .lines()
   515→            .filter(|l| !l.trim().is_empty())
   516→            .take(40)
   517→            .collect::<Vec<_>>()
   518→            .join("\n");
   519→        append_crash_marker(
   520→            "RUST-PANIC",
   521→            &format!(
   522→                "thread: {thread}\nlocation: {location}\nmessage: {msg}\nbacktrace(40):\n{bt_tail}"
   523→            ),
   524→        );
   525→        eprintln!("[quartz-panic] {msg} @ {location}");
   526→        default_hook(info);
   527→    }));
   528→}

   530→/// 清理失败下载残留的临时文件（*.qz-part / *.part / *.tmp 与 .*.staging 目录）。
   531→/// 只动顶层明确的临时名，绝不碰正常数据。返回回收字节数（eprintln 汇报）。
   532→fn cleanup_stale_partials() {
   533→    let roots = [paths::runtime_dir(), paths::game_root()];
   534→    let mut reclaimed: u64 = 0;
   535→    for root in roots {
   536→        for entry in std::fs::read_dir(&root).into_iter().flatten().flatten() {
   537→            let p = entry.path();
   538→            let name = entry.file_name().to_string_lossy().into_owned();
   539→            let is_dir = p.is_dir();
   540→            let stale_part = !is_dir
   541→                && (name.ends_with(".qz-part")
   542→                    || name.ends_with(".part")
   543→                    || name.ends_with(".tmp"));
   544→            let stale_staging = is_dir && name.starts_with('.') && name.ends_with(".staging");
   545→            if stale_part || stale_staging {
   546→                let size = entry
   547→                    .metadata()
   548→                    .map(|m| if m.is_dir() { 0 } else { m.len() })
   549→                    .unwrap_or(0);
   550→                let ok = if is_dir {
   551→                    std::fs::remove_dir_all(&p).is_ok()
   552→                } else {
   553→                    std::fs::remove_file(&p).is_ok()
   554→                };
   555→                if ok {
   556→                    reclaimed += size;
   557→                    eprintln!("[quartz] 已清理残留下载缓存：{}（{} 字节）", p.display(), size);
   558→                }
   559→            }
   560→        }
   561→    }
   562→    if reclaimed > 0 {
   563→        eprintln!("[quartz] 共回收残留下载缓存 {} 字节", reclaimed);
   564→    }
   565→}

   567→/// 下载类命令的可用空间下限（精确需求在 core 各下载函数内按实际字节数校验）。
   568→const DOWNLOAD_FLOOR: u64 = 200 * 1024 * 1024;

   570→/// 下载前底线检查：可用空间 < 200MB 直接拦（ENOSPC 在 WebView/系统层面后果严重）。
   571→fn check_download_floor(target: &Path) -> Result<(), String> {
   572→    match quartz_core::space::free_space(target) {
   573→        Some(free) if free < DOWNLOAD_FLOOR => Err(quartz_core::space::insufficient(
   574→            DOWNLOAD_FLOOR,
   575→            free,
   576→            "执行下载",
   577→        )),
   578→        _ => Ok(()),
   579→    }
   580→}

   582→/// 字符安全截断（按 char 而非字节，避免切断 UTF-8）。
   583→fn clamp_chars(s: &str, max: usize) -> String {
   584→    if s.chars().count() <= max {
   585→        return s.to_string();
   586→    }
   587→    s.chars().take(max).collect()
   588→}

   590→/// 从崩溃日志提取最后一个标记块（含标记行，最多 60 行）。
   591→fn last_crash_summary() -> Option<String> {
   592→    let path = CRASH_LOG.get()?;
   593→    let text = std::fs::read_to_string(path).ok()?;
   594→    let mut blocks: Vec<Vec<&str>> = Vec::new();
   595→    for line in text.lines() {
   596→        if line.starts_with("==== CRASH @") {
   597→            blocks.push(vec![line]);
   598→        } else if let Some(last) = blocks.last_mut() {
   599→            last.push(line);
   600→        }
   601→    }
   602→    let last = blocks.last()?;
   603→    let body = last
   604→        .iter()
   605→        .take(60)
   606→        .copied()
   607→        .collect::<Vec<&str>>()
   608→        .join("\n");
   609→    Some(clamp_chars(&body, 3000))
   610→}

   612→/// 最近一次游戏启动日志尾部（SIGABRT/退出码诊断在 UI 上直达，无需文件管理器）。
   613→fn last_launch_tail() -> Option<String> {
   614→    let dir = paths::game_root().join("logs");
   615→    let mut best: Option<(PathBuf, std::time::SystemTime)> = None;
   616→    for e in std::fs::read_dir(&dir).into_iter().flatten().flatten() {
   617→        let name = e.file_name().to_string_lossy().into_owned();
   618→        if name.starts_with("launch-") && name.ends_with(".log") {
   619→            let t = e
   620→                .metadata()
   621→                .and_then(|m| m.modified())
   622→                .unwrap_or(std::time::SystemTime::UNIX_EPOCH);
   623→            if best.as_ref().map(|(_, bt)| t > *bt).unwrap_or(true) {
   624→                best = Some((e.path(), t));
   625→            }
   626→        }
   627→    }
   628→    let (p, _) = best?;
   629→    let text = std::fs::read_to_string(&p).ok()?;
   630→    let lines: Vec<&str> = text.lines().collect();
   631→    let tail = lines.len().saturating_sub(60);
   632→    Some(clamp_chars(&lines[tail..].join("\n"), 3000))
   633→}

   646→/// 递归收集「源存在且目标缺失/不一致」的文件（不覆盖目标已有的同名同尺寸文件）。
   647→fn collect_migrate_files(src: &Path, dst: &Path, out: &mut Vec<(PathBuf, PathBuf, u64)>) {
   648→    let Ok(entries) = std::fs::read_dir(src) else {
   649→        return;
   650→    };
   651→    for entry in entries.flatten() {
   652→        let sp = entry.path();
   653→        let dp = dst.join(entry.file_name());
   654→        if sp.is_dir() {
   655→            collect_migrate_files(&sp, &dp, out);
   656→        } else if sp.is_file() {
   657→            let size = entry.metadata().map(|m| m.len()).unwrap_or(0);
   658→            let dst_size = std::fs::metadata(&dp).map(|m| m.len()).unwrap_or(u64::MAX);
   659→            if dst_size != size {
   660→                out.push((sp, dp, size));
   661→            }
   662→        }
   663→    }
   664→}

   666→/// 后台迁移：把旧根（私有目录）的数据搬到新根（公共标准目录），
   667→/// 经 `quartz://migrate` 事件上报 {percent,done,error}。
   668→fn migrate_root(from: PathBuf, to: PathBuf, app: AppHandle) {
   669→    let mut plan: Vec<(PathBuf, PathBuf, u64)> = Vec::new();
   670→    collect_migrate_files(&from.join(".minecraft"), &to.join(".minecraft"), &mut plan);
   671→    let old_accounts = from.join("accounts.json");
   672→    let new_accounts = to.join("accounts.json");
   673→    let migrate_accounts = old_accounts.exists() && !new_accounts.exists();
   674→    if migrate_accounts {
   675→        let size = std::fs::metadata(&old_accounts).map(|m| m.len()).unwrap_or(0);
   676→        plan.push((old_accounts, new_accounts, size));
   677→    }
   678→
   679→    if plan.is_empty() {
   680→        let _ = app.emit(
   681→            "quartz://migrate",
   682→            &json!({ "percent": 100.0, "done": true, "error": null }),
   683→        );
   684→        return;
   685→    }
   686→
   687→    let total: u64 = plan.iter().map(|(_, _, s)| *s).sum::<u64>().max(1);
   688→    eprintln!("[quartz] 开始迁移存储：{} 个文件（{} 字节）", plan.len(), total);
   689→    let mut copied: u64 = 0;
   690→    let mut last = std::time::Instant::now() - std::time::Duration::from_secs(1);
   691→    for (src, dst, size) in plan {
   692→        if let Some(parent) = dst.parent() {
   693→            if std::fs::create_dir_all(parent).is_err() {
   694→                let _ = app.emit(
   695→                    "quartz://migrate",
   696→                    &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false,
   697→                             "error": format!("无法创建目录 {}", parent.display()) }),
   698→                );
   699→                return;
   700→            }
   701→        }
   702→        if std::fs::copy(&src, &dst).is_err() {
   703→            let _ = app.emit(
   704→                "quartz://migrate",
   705→                &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false,
   706→                         "error": format!("复制失败：{} -> {}", src.display(), dst.display()) }),
   707→            );
   708→            return;
   709→        }
   710→        copied += size;
   711→        let now = std::time::Instant::now();
   712→        if now.duration_since(last) >= std::time::Duration::from_millis(150) {
   713→            last = now;
   714→            let _ = app.emit(
   715→                "quartz://migrate",
   716→                &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false, "error": null }),
   717→            );
   718→        }
   719→    }
   720→    eprintln!("[quartz] 存储迁移完成");
   721→    let _ = app.emit(
   722→        "quartz://migrate",
   723→        &json!({ "percent": 100.0, "done": true, "error": null }),
   724→    );
   725→}

   727→/// 权限变化后重解析存储根；根变化时后台迁移旧数据（幂等，跳过目标已有文件）。
   728→#[tauri::command]
   729→fn reinit_storage(app: AppHandle, state: State<'_, AppState>) -> Result<serde_json::Value, String> {
   730→    let old_root = paths::game_root();
   731→    let diag = init_storage();
   732→    *STORAGE_DIAG.lock().map_err(|_| "内部状态锁异常")? = Some(diag.clone());
   733→    let new_root = paths::game_root();
   734→    // 游戏运行中不迁移（避免拷贝正在被 JVM 读写的文件）；同时防并发迁移。
   735→    let game_running = state
   736→        .game_pid
   737→        .lock()
   738→        .map_err(|_| "内部状态锁异常")?
   739→        .is_some();
   740→    let mut do_migrate = old_root != new_root && old_root.exists() && !game_running;
   741→    if do_migrate {
   742→        do_migrate = !MIGRATING.swap(true, std::sync::atomic::Ordering::SeqCst);
   743→    }
   744→    if do_migrate {
   745→        let app2 = app.clone();
   746→        std::thread::spawn(move || {
   747→            migrate_root(old_root, new_root, app2);
   748→            MIGRATING.store(false, std::sync::atomic::Ordering::SeqCst);
   749→        });
   750→    }
   751→    Ok(storage_json(&diag))
   752→}

   920→/// 拉取真实 version_manifest_v2 并排序：release -> snapshot -> old_beta -> old_alpha。
   921→#[tauri::command]
   922→async fn list_versions() -> Result<Vec<VersionInfo>, String> {
   923→    let client = meta::http_client();
   924→    let manifest = meta::fetch_version_manifest(&client)
   925→        .await
   926→        .map_err(|e| format!("版本清单拉取失败：{e}"))?;
   927→    const KIND_ORDER: [&str; 4] = ["release", "snapshot", "old_beta", "old_alpha"];
   928→    let mut out = Vec::with_capacity(manifest.versions.len());
   929→    for kind in KIND_ORDER {
   930→        for v in manifest.versions.iter().filter(|v| v.kind == kind) {
   931→            out.push(VersionInfo {
   932→                id: v.id.clone(),
   933→                version_type: v.kind.clone(),
   934→                url: v.url.clone(),
   935→            });
   936→        }
   937→    }
   938→    for v in manifest
   939→        .versions
   940→        .iter()
   941→        .filter(|v| !KIND_ORDER.contains(&v.kind.as_str()))
   942→    {
   943→        out.push(VersionInfo {
   944→            id: v.id.clone(),
   945→            version_type: v.kind.clone(),
   946→            url: v.url.clone(),
   947→        });
   948→    }
   949→    Ok(out)

   974→/// 进度经 `download://progress` 事件上报 {stage,current,total,percent,file,done,error}。

   977→    version_id: String,

   983→        if *busy {

   985→        }

   995→        let forwarder_app = app.clone();
   996→        let forwarder = tokio::spawn(async move {
   997→            while let Some(ev) = rx.recv().await {

  1006→            let _ = app.emit(

  1008→                &instance::DownloadProgress {

  1010→                    current: 0,

  1017→            );
