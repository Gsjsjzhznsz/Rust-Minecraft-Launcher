    56→    56→#[cfg(target_os = "android")]
    57→    57→fn with_jni<T>(
    58→    58→    f: impl FnOnce(&mut jni::JNIEnv, &jni::objects::JObject) -> Option<T>,
    59→    59→) -> Option<T> {
    60→    60→    std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
    61→    61→        let ctx = ndk_context::android_context();
    62→    62→        let vm = unsafe { jni::JavaVM::from_raw(ctx.vm().cast()) }.ok()?;
    63→    63→        let mut env = vm.attach_current_thread().ok()?;
    64→    64→        let context = unsafe { jni::objects::JObject::from_raw(ctx.context().cast()) };
    65→    65→        // 防御：清掉上游遗留的挂起异常。JNIEnv 按线程复用，任何未清理的异常都会让
    66→    66→        // 本闭包内的下一个 JNI 调用（甚至 Tauri 内部调用）在 CheckJNI 下直接 abort。
    67→    67→        if env.exception_check().unwrap_or(false) {
    68→    68→            let _ = env.exception_clear();
    69→    69→        }
    70→    70→        let result = f(&mut env, &context);
    71→    71→        // 关键：闭包内任何 JNI Err（如 NoSuchMethodError）都会把异常留在挂起态，
    72→    72→        // 若不清掉，下一次 NewStringUTF 等调用即触发
    73→    73→        // "JNI NewStringUTF called with pending exception" abort 闪退。
    74→    74→        if env.exception_check().unwrap_or(false) {
    75→    75→            let _ = env.exception_clear();
    76→    76→        }
    77→    77→        result
    78→    78→    }))
    79→    79→    .ok()
    80→    80→    .flatten()
    81→    81→}

    84→    84→fn jni_file_path(env: &mut jni::JNIEnv, file: &jni::objects::JObject) -> Option<PathBuf> {
    85→    85→    let path = env
    86→    86→        .call_method(file, "getAbsolutePath", "()Ljava/lang/String;", &[])
    87→    87→        .ok()?
    88→    88→        .l()
    89→    89→        .ok()?;
    90→    90→    let s: String = env.get_string((&path).into()).ok()?.into();
    91→    91→    if s.is_empty() {
    92→    92→        None
    93→    93→    } else {
    94→    94→        Some(PathBuf::from(s))
    95→    95→    }
    96→    96→}

   131→   131→// 存储权限（完整文件权限）
   132→   132→// ---------------------------------------------------------------------------
   133→   133→//
   134→   134→// 设计：APK 固定 targetSdk 28（Android 10 前的存储模型 + 允许 exec() 应用目录
   135→   135→// 内的 JVM —— Pojav/Termux 同款方案）。
   136→   136→//
   137→   137→// 授权通道按系统版本分流（MainActivity.requestStorageAccess）:
   138→   138→// - Android 11+（API 30+）：MANAGE_EXTERNAL_STORAGE「所有文件访问」系统设置页。
   139→   139→//   不走 WRITE 弹窗的原因：targetSdk<=29 应用在 Android 11+ 经弹窗授予存储权限时，
   140→   140→//   系统会杀掉应用进程刷新 FUSE 挂载视图 —— 用户视角即「授权后闪退」。
   141→   141→//   所有文件访问页授权不杀进程，返回后 focus 重检即可切换标准目录。
   142→   142→// - Android 10 及以下（API <= 29）：READ/WRITE_EXTERNAL_STORAGE 运行时弹窗
   143→   143→//   （旧存储模型，授权即时生效不杀进程）。
   144→   144→// 游戏文件即可放在文件管理器可见的标准目录 /storage/emulated/0/Quartz/files。
   145→   145→
   146→   146→/// Android 端存储权限是否已授权（纯系统 API，不依赖 Activity 实例）。
   147→   147→///
   148→   148→/// 注意：ndk-context 给到的 context 是 **Application** 对象（不是 MainActivity），
   149→   149→/// 其上不存在任何自定义方法 —— 在它上面 GetMethodID("isStorageAccessGranted")
   150→   150→/// 必抛 NoSuchMethodError。故全部改用静态方法 / Context 自有方法：
   151→   151→/// - Android 11+（API 30+）：静态 Environment.isExternalStorageManager()
   152→   152→///   （MANAGE_EXTERNAL_STORAGE「所有文件访问」）
   153→   153→/// - Android 10 及以下：Context.checkSelfPermission(WRITE_EXTERNAL_STORAGE)
   154→   154→///   （targetSdk 28 旧存储模型，Application 即 Context，可直接调）
   155→   155→#[cfg(target_os = "android")]
   156→   156→fn android_storage_permission_granted() -> bool {
   157→   157→    use jni::objects::JValue;
   158→   158→    with_jni(|env, ctx| {
   159→   159→        let sdk = env
   160→   160→            .get_static_field("android/os/Build$VERSION", "SDK_INT", "I")
   161→   161→            .ok()?
   162→   162→            .i()
   163→   163→            .ok()?;
   164→   164→        if sdk >= 30 {
   165→   165→            let granted = env
   166→   166→                .call_static_method(
   167→   167→                    "android/os/Environment",
   168→   168→                    "isExternalStorageManager",
   169→   169→                    "()Z",
   170→   170→                    &[],
   171→   171→                )
   172→   172→                .ok()?
   173→   173→                .z()
   174→   174→                .ok()?;
   175→   175→            Some(granted)
   176→   176→        } else {
   177→   177→            let perm = env
   178→   178→                .new_string("android.permission.WRITE_EXTERNAL_STORAGE")
   179→   179→                .ok()?;
   180→   180→            let granted = env
   181→   181→                .call_method(
   182→   182→                    ctx,
   183→   183→                    "checkSelfPermission",
   184→   184→                    "(Ljava/lang/String;)I",
   185→   185→                    &[JValue::Object(&perm.into())],
   186→   186→                )
   187→   187→                .ok()?
   188→   188→                .i()
   189→   189→                .ok()?;
   190→   190→            // PackageManager.PERMISSION_GRANTED == 0
   191→   191→            Some(granted == 0)
   192→   192→        }
   193→   193→    })
   194→   194→    .unwrap_or(false)
   195→   195→}

   197→   197→/// 存储权限是否已授权（桌面端恒真）。
   198→   198→fn storage_permission_granted() -> bool {
   199→   199→    #[cfg(target_os = "android")]
   200→   200→    {
   201→   201→        android_storage_permission_granted()

   203→   203→    #[cfg(not(target_os = "android"))]
   204→   204→    {
   205→   205→        true
   206→   206→    }
   207→   207→}

   215→   215→/// 申请完整文件权限：Android 11+ 跳系统「所有文件访问」设置页（授权不杀进程），
   216→   216→/// Android 10- 触发运行时弹窗。结果不通过本命令返回：UI 在 focus/visibilitychange
   217→   217→/// 时重新查询状态（回到应用后自动重检）。
   218→   218→#[tauri::command]
   219→   219→fn request_storage_permission() -> Result<bool, String> {
   220→   220→    #[cfg(target_os = "android")]
   221→   221→    {
   222→   222→        // MainActivity 伴生对象静态方法（GetStaticMethodID，不依赖接收对象类型）：
   223→   223→        // 内部持有当前 Activity 实例，按系统版本分流（11+「所有文件访问」设置页 /
   224→   224→        // 10- 运行时弹窗）。此前直接在 ndk-context 的 Application 对象上调实例方法
   225→   225→        // requestStorageAccess —— 其上查无此方法，NoSuchMethodError 未清理即 abort。
   226→   226→        let via_kotlin = with_jni(|env, ctx| {
   227→   227→            use jni::objects::JClass;
   228→   228→            use jni::objects::JValue;
   229→   229→            // 注意：FindClass 在 AttachCurrentThread 的原生线程上走系统类加载器，
   230→   230→            // 解析不到 APK 内的应用类；必须经 context.getClassLoader().loadClass()
   231→   231→            // 显式解析壳层 MainActivity（Application 即 Context，链路稳定）。
   232→   232→            let loader = env
   233→   233→                .call_method(ctx, "getClassLoader", "()Ljava/lang/ClassLoader;", &[])
   234→   234→                .ok()?
   235→   235→                .l()
   236→   236→                .ok()?;
   237→   237→            let name = env
   238→   238→                .new_string("com.quartz.launcher.MainActivity")
   239→   239→                .ok()?;
   240→   240→            let cls_obj = env
   241→   241→                .call_method(
   242→   242→                    &loader,
   243→   243→                    "loadClass",
   244→   244→                    "(Ljava/lang/String;)Ljava/lang/Class;",
   245→   245→                    &[JValue::Object(&name.into())],
   246→   246→                )
   247→   247→                .ok()?
   248→   248→                .l()
   249→   249→                .ok()?;
   250→   250→            let cls = JClass::from(cls_obj);
   251→   251→            env.call_static_method(&cls, "requestStorageAccessStatic", "()V", &[])
   252→   252→                .ok()?;
   253→   253→            Some(())
   254→   254→        });
   255→   255→        if via_kotlin.is_some() {
   256→   256→            return Ok(true);
   257→   257→        }
   258→   258→        // 到这里说明 JNI 链路本身异常（with_jni 已清异常）；给用户明确的手动指引。
   259→   259→        Err("无法唤起系统权限界面。请手动授予：系统设置 → 应用 → Quartz → 权限 → 所有文件访问权限".to_string())
   260→   260→    }
   261→   261→    #[cfg(not(target_os = "android"))]
   262→   262→    {
   263→   263→        Ok(true)
   264→   264→    }
   265→   265→}

   285→   285→/// Android 平台解析真实可写目录；ndk-context 未初始化等异常一律回退硬编码路径。
   286→   286→///
   287→   287→/// 优先级（与授权状态联动）：
   288→   288→/// 1. 已授权完整文件权限 -> 公共标准目录 /storage/emulated/0/Quartz/files（可见、卸载不丢）
   289→   289→/// 2. 应用外部私有目录（getExternalFilesDir，可浏览，卸载丢失）
   290→   290→/// 3. 应用内部私有目录（getFilesDir，必定可写）
   291→   291→#[cfg(target_os = "android")]
   292→   292→fn resolve_android_root() -> (PathBuf, &'static str) {
   293→   293→    if storage_permission_granted() {
   294→   294→        let public = PathBuf::from("/storage/emulated/0/Quartz/files");
   295→   295→        if std::fs::create_dir_all(&public).is_ok() {
   296→   296→            return (public, "public");
   297→   297→        }
   298→   298→        eprintln!("[quartz] 公共标准目录创建失败，回退应用私有目录");
   299→   299→    }
   300→   300→    let result = with_jni(|env, context| {
   301→   301→        // 外部私有目录（用户可用文件管理器查看 .minecraft）
   302→   302→        if let Some(p) = jni_external_files_dir(env, context) {
   303→   303→            return Some((p, "external"));
   304→   304→        }
   305→   305→        // 内部私有目录（必定可写）
   306→   306→        jni_internal_files_dir(env, context).map(|p| (p, "internal"))
   307→   307→    });
   308→   308→    result.unwrap_or_else(|| {
   309→   309→        (
   310→   310→            PathBuf::from(format!(
   311→   311→                "/storage/emulated/0/Android/data/{}/files",
   312→   312→                paths::ANDROID_PACKAGE_ID
   313→   313→            )),
   314→   314→            "fallback",
   315→   315→        )
   316→   316→    })
   317→   317→}

   319→   319→/// Android 内部私有目录（Context.getFilesDir()，任何设备必定可写；exec 可用）。
   320→   320→#[cfg(target_os = "android")]
   321→   321→fn resolve_android_internal() -> Option<PathBuf> {
   322→   322→    with_jni(|env, context| jni_internal_files_dir(env, context))
   323→   323→}

   325→   325→/// 初始化游戏根目录与 JVM pack 目录并创建标准布局；返回诊断信息。
   326→   326→fn init_storage() -> StorageDiag {
   327→   327→    #[cfg(target_os = "android")]
   328→   328→    let (root, mode) = resolve_android_root();
   329→   329→    #[cfg(not(target_os = "android"))]
   330→   330→    let (root, mode) = (paths::game_root(), "desktop");
   331→   331→
   332→   332→    let _ = paths::set_runtime_root(root.clone());
   333→   333→    let mut layout = paths::ensure_layout();
   334→   334→
   335→   335→    // 外部目录不可写（EACCES / os error 13）时回退内部私有目录，
   336→   336→    // 保证离线登录与下载在任何设备上永远可用。
   337→   337→    #[cfg(target_os = "android")]
   338→   338→    let mut mode = mode;
   339→   339→    #[cfg(target_os = "android")]
   340→   340→    if layout.is_err() && mode != "internal" {
   341→   341→        eprintln!(
   342→   342→            "[quartz] 当前根目录不可写（{:?}），回退内部存储",
   343→   343→            layout.as_ref().err()
   344→   344→        );
   345→   345→        if let Some(internal) = resolve_android_internal() {
   346→   346→            let _ = paths::set_runtime_root(internal.clone());
   347→   347→            layout = paths::ensure_layout();
   348→   348→            mode = "internal";
   349→   349→        }
   350→   350→    }
   351→   351→
   352→   352→    // JVM pack 必须放在可 exec 的内部目录（FUSE noexec + targetSdk>=29 禁 exec 的约束）
   353→   353→    #[cfg(target_os = "android")]
   354→   354→    if let Some(internal) = resolve_android_internal() {
   355→   355→        paths::set_runtime_jres_root(internal.join("jres"));
   356→   356→    }
   357→   357→
   358→   358→    let diag = StorageDiag {
   359→   359→        mode,
   360→   360→        root: paths::game_root(),
   361→   361→        layout_ok: layout.is_ok(),
   362→   362→        layout_error: layout.err().map(|e| format!("{e}")),
   363→   363→    };
   364→   364→    eprintln!(
   365→   365→        "[quartz] 存储初始化：mode={} root={} jres={}",
   366→   366→        diag.mode,
   367→   367→        diag.root.display(),
   368→   368→        paths::runtime_dir().display()
   369→   369→    );
   370→   370→    diag
   371→   371→}

   373→   373→fn storage_json(diag: &StorageDiag) -> serde_json::Value {
   374→   374→    json!({
   375→   375→        "gameDir": diag.root.to_string_lossy(),
   376→   376→        "storageMode": diag.mode,
   377→   377→        "layoutOk": diag.layout_ok,
   378→   378→        "layoutError": diag.layout_error,
   379→   379→        "fullAccess": storage_permission_granted(),
   380→   380→    })

   383→   383→/// 递归收集「源存在且目标缺失/不一致」的文件（不覆盖目标已有的同名同尺寸文件）。
   384→   384→fn collect_migrate_files(src: &Path, dst: &Path, out: &mut Vec<(PathBuf, PathBuf, u64)>) {
   385→   385→    let Ok(entries) = std::fs::read_dir(src) else {
   386→   386→        return;
   387→   387→    };
   388→   388→    for entry in entries.flatten() {
   389→   389→        let sp = entry.path();
   390→   390→        let dp = dst.join(entry.file_name());
   391→   391→        if sp.is_dir() {
   392→   392→            collect_migrate_files(&sp, &dp, out);
   393→   393→        } else if sp.is_file() {
   394→   394→            let size = entry.metadata().map(|m| m.len()).unwrap_or(0);
   395→   395→            let dst_size = std::fs::metadata(&dp).map(|m| m.len()).unwrap_or(u64::MAX);
   396→   396→            if dst_size != size {
   397→   397→                out.push((sp, dp, size));
   398→   398→            }
   399→   399→        }
   400→   400→    }
   401→   401→}

   403→   403→/// 后台迁移：把旧根（私有目录）的数据搬到新根（公共标准目录），
   404→   404→/// 经 `quartz://migrate` 事件上报 {percent,done,error}。
   405→   405→fn migrate_root(from: PathBuf, to: PathBuf, app: AppHandle) {
   406→   406→    let mut plan: Vec<(PathBuf, PathBuf, u64)> = Vec::new();
   407→   407→    collect_migrate_files(&from.join(".minecraft"), &to.join(".minecraft"), &mut plan);
   408→   408→    let old_accounts = from.join("accounts.json");
   409→   409→    let new_accounts = to.join("accounts.json");
   410→   410→    let migrate_accounts = old_accounts.exists() && !new_accounts.exists();
   411→   411→    if migrate_accounts {
   412→   412→        let size = std::fs::metadata(&old_accounts).map(|m| m.len()).unwrap_or(0);
   413→   413→        plan.push((old_accounts, new_accounts, size));
   414→   414→    }
   415→   415→
   416→   416→    if plan.is_empty() {
   417→   417→        let _ = app.emit(
   418→   418→            "quartz://migrate",
   419→   419→            &json!({ "percent": 100.0, "done": true, "error": null }),
   420→   420→        );
   421→   421→        return;
   422→   422→    }
   423→   423→
   424→   424→    let total: u64 = plan.iter().map(|(_, _, s)| *s).sum::<u64>().max(1);
   425→   425→    eprintln!("[quartz] 开始迁移存储：{} 个文件（{} 字节）", plan.len(), total);
   426→   426→    let mut copied: u64 = 0;
   427→   427→    let mut last = std::time::Instant::now() - std::time::Duration::from_secs(1);
   428→   428→    for (src, dst, size) in plan {
   429→   429→        if let Some(parent) = dst.parent() {
   430→   430→            if std::fs::create_dir_all(parent).is_err() {
   431→   431→                let _ = app.emit(
   432→   432→                    "quartz://migrate",
   433→   433→                    &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false,
   434→   434→                             "error": format!("无法创建目录 {}", parent.display()) }),
   435→   435→                );
   436→   436→                return;
   437→   437→            }
   438→   438→        }
   439→   439→        if std::fs::copy(&src, &dst).is_err() {
   440→   440→            let _ = app.emit(
   441→   441→                "quartz://migrate",
   442→   442→                &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false,
   443→   443→                         "error": format!("复制失败：{} -> {}", src.display(), dst.display()) }),
   444→   444→            );
   445→   445→            return;
   446→   446→        }
   447→   447→        copied += size;
   448→   448→        let now = std::time::Instant::now();
   449→   449→        if now.duration_since(last) >= std::time::Duration::from_millis(150) {
   450→   450→            last = now;
   451→   451→            let _ = app.emit(
   452→   452→                "quartz://migrate",
   453→   453→                &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false, "error": null }),
   454→   454→            );
   455→   455→        }
   456→   456→    }
   457→   457→    eprintln!("[quartz] 存储迁移完成");
   458→   458→    let _ = app.emit(
   459→   459→        "quartz://migrate",
   460→   460→        &json!({ "percent": 100.0, "done": true, "error": null }),
   461→   461→    );
   462→   462→}

   464→   464→/// 权限变化后重解析存储根；根变化时后台迁移旧数据（幂等，跳过目标已有文件）。
   465→   465→#[tauri::command]
   466→   466→fn reinit_storage(app: AppHandle, state: State<'_, AppState>) -> Result<serde_json::Value, String> {
   467→   467→    let old_root = paths::game_root();
   468→   468→    let diag = init_storage();
   469→   469→    *STORAGE_DIAG.lock().map_err(|_| "内部状态锁异常")? = Some(diag.clone());
   470→   470→    let new_root = paths::game_root();
   471→   471→    // 游戏运行中不迁移（避免拷贝正在被 JVM 读写的文件）；同时防并发迁移。
   472→   472→    let game_running = state
   473→   473→        .game_pid
   474→   474→        .lock()
   475→   475→        .map_err(|_| "内部状态锁异常")?
   476→   476→        .is_some();
   477→   477→    let mut do_migrate = old_root != new_root && old_root.exists() && !game_running;
   478→   478→    if do_migrate {
   479→   479→        do_migrate = !MIGRATING.swap(true, std::sync::atomic::Ordering::SeqCst);
   480→   480→    }
   481→   481→    if do_migrate {
   482→   482→        let app2 = app.clone();
   483→   483→        std::thread::spawn(move || {
   484→   484→            migrate_root(old_root, new_root, app2);
   485→   485→            MIGRATING.store(false, std::sync::atomic::Ordering::SeqCst);
   486→   486→        });
   487→   487→    }
   488→   488→    Ok(storage_json(&diag))
   489→   489→}

   678→   678→        .filter(|v| !KIND_ORDER.contains(&v.kind.as_str()))
   679→   679→    {
   680→   680→        out.push(VersionInfo {

   682→   682→            version_type: v.kind.clone(),
   683→   683→            url: v.url.clone(),

   710→   710→/// 后台启动完整下载管线（8 并发 + SHA-1 校验 + 3 次重试 + 原子落盘），
   711→   711→/// 进度经 `download://progress` 事件上报 {stage,current,total,percent,file,done,error}。
   712→   712→#[tauri::command]
   713→   713→async fn download_game(
   714→   714→    version_id: String,
   715→   715→    app: AppHandle,
   716→   716→    state: State<'_, AppState>,
   717→   717→) -> Result<(), String> {
   718→   718→    {
   719→   719→        let mut busy = state.downloading.lock().map_err(|_| "内部状态锁异常")?;
   720→   720→        if *busy {
   721→   721→            return Err("已有下载任务正在进行，请等待完成后再试".to_string());
   722→   722→        }
   723→   723→        *busy = true;
   724→   724→    }
   725→   725→    let busy = state.downloading.clone();
   726→   726→    let app = app.clone();
   727→   727→
   728→   728→    tauri::async_runtime::spawn(async move {
   729→   729→        let (tx, mut rx) = tokio::sync::mpsc::channel::<instance::DownloadProgress>(128);
   730→   730→        let forwarder_app = app.clone();
   731→   731→        let forwarder = tokio::spawn(async move {
   732→   732→            while let Some(ev) = rx.recv().await {
   733→   733→                let _ = forwarder_app.emit("download://progress", &ev);
   734→   734→            }
   735→   735→        });
   736→   736→
   737→   737→        let client = meta::http_client();
   738→   738→        let result = instance::install_standard(&client, &version_id, tx).await;
   739→   739→        if let Err(e) = &result {
   740→   740→            // 失败也走事件通道（UI 红字报错 + 可重试），绝不静默。
   741→   741→            let _ = app.emit(
   742→   742→                "download://progress",
   743→   743→                &instance::DownloadProgress {
   744→   744→                    stage: "error".to_string(),
   745→   745→                    current: 0,
   746→   746→                    total: 0,
   747→   747→                    percent: 0.0,
   748→   748→                    file: String::new(),
   749→   749→                    done: false,
   750→   750→                    error: Some(e.to_string()),
   751→   751→                },
   752→   752→            );
   753→   753→        }
   754→   754→        drop(result);
   755→   755→        if let Ok(mut busy) = busy.lock() {
   756→   756→            *busy = false;
   757→   757→        }
   758→   758→        // forwarder 在 tx 关闭后自然退出
   759→   759→        let _ = forwarder.await;
   760→   760→    });
   761→   761→
   762→   762→    Ok(())
   763→   763→}

   786→   786→/// 后台下载 JVM 运行时 pack，进度经 `runtime://progress` 上报（与下载游戏事件同构）。
   787→   787→#[tauri::command]
   788→   788→async fn download_runtime(java_major: u32, app: AppHandle, state: State<'_, AppState>) -> Result<(), String> {
   789→   789→    if state
   790→   790→        .runtime_busy
   791→   791→        .swap(true, std::sync::atomic::Ordering::SeqCst)
   792→   792→    {
   793→   793→        return Err("已有运行时下载任务正在进行".to_string());
   794→   794→    }
   795→   795→    let busy = state.runtime_busy.clone();

  1064→  1064→/// 拉起 JVM 子进程并托管日志/退出（`launch://log` / `launch://exit` 事件）。
  1065→  1065→///
  1066→  1066→/// 前置：JVM pack 已安装（Android 内部目录，targetSdk 28 允许 exec）。
  1067→  1067→/// smoke test 先跑 `java -version` 把 JRE 版本写进控制台，失败时给出明确报错。
  1068→  1068→fn spawn_game(
  1069→  1069→    version_id: String,
  1070→  1070→    plan: jvmops::LaunchPlan,
  1071→  1071→    major: u32,
  1072→  1072→    game_pid: Arc<Mutex<Option<u32>>>,
  1073→  1073→    app: AppHandle,
  1074→  1074→) -> Result<(), String> {
  1075→  1075→    use std::io::BufRead;
  1076→  1076→    use std::io::Write;
  1077→  1077→    use std::process::{Command, Stdio};
  1078→  1078→
  1079→  1079→    let java = runtime::java_bin(major);
  1080→  1080→    let jre_home = runtime::install_dir(major);
  1081→  1081→    let game_dir = paths::minecraft_dir();
  1082→  1082→
  1083→  1083→    // ---- 持久化启动日志（完整证据链）----
  1084→  1084→    // stdout/stderr/quartz 行 + 完整命令行/环境 全部落盘 <game_root>/logs/launch-*.log。
  1085→  1085→    // 原生崩溃（SIGABRT 等）的 JVM 报告只写 hs_err 文件 / logcat，不会进 stderr ——
  1086→  1086→    // 因此异常退出时自动摘录 hs_err 到本日志，下次运行即可拿到全部诊断信息。
  1087→  1087→    let logs_dir = paths::game_root().join("logs");
  1088→  1088→    let _ = std::fs::create_dir_all(&logs_dir);
  1089→  1089→    let log_path = logs_dir.join(format!(
  1090→  1090→        "launch-{}.log",
  1091→  1091→        std::time::SystemTime::now()
  1092→  1092→            .duration_since(std::time::UNIX_EPOCH)
  1093→  1093→            .map(|d| d.as_secs())
  1094→  1094→            .unwrap_or(0)
  1095→  1095→    ));
  1096→  1096→    let log_file: Arc<Mutex<Option<std::fs::File>>> =
  1097→  1097→        Arc::new(Mutex::new(std::fs::File::create(&log_path).ok()));
  1098→  1098→
  1099→  1099→    // 控制台 + 文件双写。
  1100→  1100→    let qlog = {
  1101→  1101→        let app = app.clone();
  1102→  1102→        let log_file = log_file.clone();
  1103→  1103→        move |source: &str, line: &str| {
  1104→  1104→            let _ = app.emit(
  1105→  1105→                "launch://log",
  1106→  1106→                &LaunchLog { source: source.into(), line: line.into() },
  1107→  1107→            );
  1108→  1108→            if let Ok(mut guard) = log_file.lock() {
  1109→  1109→                if let Some(f) = guard.as_mut() {
  1110→  1110→                    let _ = writeln!(f, "[{source}] {line}");
  1111→  1111→                    let _ = f.flush();
  1112→  1112→                }
  1113→  1113→            }
  1114→  1114→        }
  1115→  1115→    };
  1116→  1116→
  1117→  1117→    // ---- JVM 动态库搜索路径（必须先于 smoke test 计算，两处共用） ----
  1118→  1118→    // bionic linker 不认 RUNPATH，bin/java 找 libjli.so 完全依赖 LD_LIBRARY_PATH。
  1119→  1119→    // 顺序（先者优先）：渲染层目录 -> LWJGL 安卓 natives -> 游戏 natives ->
  1120→  1120→    // JRE lib / lib/jli / lib/server / lib/client。
  1121→  1121→    let mut ld_parts: Vec<String> = Vec::new();
  1122→  1122→    #[cfg(target_os = "android")]
  1123→  1123→    let active_renderer = {
  1124→  1124→        let id = renderers::active();
  1125→  1125→        let spec = renderers::spec_for(&id);
  1126→  1126→        if renderers::is_installed(&id) {
  1127→  1127→            ld_parts.push(
  1128→  1128→                renderers::install_dir(&id)
  1129→  1129→                    .to_string_lossy()
  1130→  1130→                    .into_owned(),
  1131→  1131→            );
  1132→  1132→        }
  1133→  1133→        if lwjgl::is_installed() {
  1134→  1134→            ld_parts.push(lwjgl::natives_dir().to_string_lossy().into_owned());
  1135→  1135→        }
  1136→  1136→        qlog(
  1137→  1137→            "quartz",
  1138→  1138→            &format!(
  1139→  1139→                "[quartz] 渲染层：{}（{}）｜ LWJGL 安卓包：{}",
  1140→  1140→                spec.map(|s| s.name).unwrap_or("未知"),
  1141→  1141→                id,
  1142→  1142→                if lwjgl::is_installed() { "已就绪" } else { "未安装" },
  1143→  1143→            ),
  1144→  1144→        );
  1145→  1145→        id
  1146→  1146→    };
  1147→  1147→    #[cfg(not(target_os = "android"))]
  1148→  1148→    let active_renderer = String::new();
  1149→  1149→    ld_parts.push(jre_home.join("lib").display().to_string());
  1150→  1150→    ld_parts.push(jre_home.join("lib/jli").display().to_string());
  1151→  1151→    ld_parts.push(jre_home.join("lib/server").display().to_string());
  1152→  1152→    ld_parts.push(jre_home.join("lib/client").display().to_string());
  1153→  1153→    ld_parts.push(plan.natives_dir.display().to_string());
  1154→  1154→    let ld_library_path = ld_parts.join(":");
  1155→  1155→
  1156→  1156→    // Android 无 /tmp：JVM 的 hsperfdata / 临时文件目录必须显式指向内部可写目录
  1157→  1157→    //（tmpdir 初始化失败是 Pojav 系 JRE 早期 abort 的常见诱因）。
  1158→  1158→    #[cfg(target_os = "android")]
  1159→  1159→    let tmp_dir = {
  1160→  1160→        let d = paths::runtime_dir().join("tmp");
  1161→  1161→        let _ = std::fs::create_dir_all(&d);
  1162→  1162→        d
  1163→  1163→    };
  1164→  1164→    #[cfg(not(target_os = "android"))]
  1165→  1165→    let tmp_dir = std::env::temp_dir();
  1166→  1166→
  1167→  1167→    // ---- 头部：完整命令行 + 环境（诊断核心：一眼看出坏参数/坏路径） ----
  1168→  1168→    if let Ok(mut guard) = log_file.lock() {
  1169→  1169→        if let Some(f) = guard.as_mut() {
  1170→  1170→            let _ = writeln!(f, "==== Quartz launch {version_id} (java {major}) ====");
  1171→  1171→            let _ = writeln!(f, "java: {}", java.display());
  1172→  1172→            let _ = writeln!(f, "cwd: {}", plan.working_dir.display());
  1173→  1173→            let _ = writeln!(f, "JAVA_HOME: {}", jre_home.display());
  1174→  1174→            let _ = writeln!(f, "LD_LIBRARY_PATH: {ld_library_path}");
  1175→  1175→            let _ = writeln!(f, "TMPDIR: {}", tmp_dir.display());
  1176→  1176→            let _ = writeln!(f, "---- jvm args ({} 个) ----", plan.jvm_args.len());
  1177→  1177→            for a in &plan.jvm_args {
  1178→  1178→                let _ = writeln!(f, "  {a}");
  1179→  1179→            }
  1180→  1180→            let _ = writeln!(f, "---- classpath ({} jars) ----", plan.classpath.len());
  1181→  1181→            for c in &plan.classpath {
  1182→  1182→                let _ = writeln!(f, "  {}", c.display());
  1183→  1183→            }
  1184→  1184→            let _ = writeln!(f, "main_class: {}", plan.main_class);
