    89→            }

   123→    override fun parse(app: PluginManager.PluginApp) {
   124→        val metaData = app.appInfo.metaData ?: return
   125→        // v2 优先：同时声明新旧架构的插件只加载 v2 配置
   126→        if (metaData.containsKey(PluginManager.META_PLUGIN_V2)) {
   127→            parseV2(app, metaData)
   128→            return
   129→        }
   130→        if (metaData.getBoolean(PluginManager.META_PLUGIN, false)) {
   131→            parseV1(app, metaData)
   132→        }
   133→    }

   135→    private fun parseV1(app: PluginManager.PluginApp, metaData: Bundle) {
   136→        val rendererString = metaData.getString("renderer") ?: return
   137→        val des = metaData.getString("des") ?: return
   138→        val boatEnvString = metaData.getString("boatEnv") ?: return
   139→        val pojavEnvString = metaData.getString("pojavEnv") ?: return
   140→        val nativeLibraryDir = app.appInfo.nativeLibraryDir
   141→        val renderer = rendererString.split(":")
   142→        val boatEnv = boatEnvString.split(":")
   143→        val pojavEnv = pojavEnvString.split(":")
   144→        val minMCVer = metaData.safeGetString("minMCVer") ?: ""
   145→        val maxMCVer = metaData.safeGetString("maxMCVer") ?: ""
   146→        addRenderer(
   147→            Renderer(
   148→                renderer[0],
   149→                des,
   150→                renderer[1],
   151→                renderer[2],
   152→                nativeLibraryDir,
   153→                boatEnv,
   154→                pojavEnv,
   155→                app.packageName,
   156→                minMCVer,
   157→                maxMCVer,
   158→                source = app.label
   159→            )
   160→        )
   161→    }

   163→    private fun parseV2(app: PluginManager.PluginApp, metaData: Bundle) {
   164→        val configResId = metaData.getInt(PluginManager.META_PLUGIN_V2)
   165→        if (configResId == 0) return
   166→        val entry = runCatching {
   167→            val pm = FCLApp.getAppContext().packageManager
   168→            val resources = pm.getResourcesForApplication(app.appInfo)
   169→            val config = rendererV2Json.decodeFromString<RendererConfigV2>(resources.getString(configResId))
   170→            val titles = mutableMapOf<String, String>()
   171→            config.env.forEach { env ->
   172→                val titleKey = when (env) {
   173→                    is SelectableEnvV2 -> env.title?.key
   174→                    is CustomizableEnvV2 -> env.title?.key
   175→                    is ToggleableEnvV2 -> env.title?.key
   176→                    else -> null
   177→                } ?: return@forEach
   178→                runCatching {
   179→                    val id = metaData.getInt(titleKey)
   180→                    if (id != 0) titles[titleKey] = resources.getString(id)
   181→                }
   182→            }
   183→            V2Entry(config, app.appInfo.nativeLibraryDir, titles, app.label)
   184→        }.getOrNull() ?: return
   185→        v2Entries[app.packageName] = entry
   186→        addRenderer(buildV2Renderer(app.packageName, entry))
   187→    }

   215→    /** 可配置环境变量列表（供配置对话框展示；无 v2 配置的插件返回空） */
   216→    fun getConfigurableEnvs(packageName: String): List<EnvSpec> {
   217→        ensureInit()
   218→        val entry = v2Entries[packageName] ?: return emptyList()
   219→        val prefs = envPrefs[packageName].orEmpty()
   220→        return entry.config.env.mapNotNull { env ->
   221→            when (env) {
   222→                is SelectableEnvV2 -> EnvSpec(
   223→                    env.key,
   224→                    EnvType.SELECTABLE,
   225→                    env.title?.let { entry.titles[it.key] } ?: env.key,
   226→                    buildList {
   227→                        add(env.items.defaultValue)
   228→                        addAll(env.items.values)
   229→                    }.distinct(),
   230→                    env.items.defaultValue,
   231→                    env.check != null,
   232→                    if (env.check == null) true
   233→                    else prefs[env.key + ENABLED_SUFFIX]?.toBooleanStrictOrNull() ?: env.check,
   234→                    prefs[env.key] ?: env.items.defaultValue
   235→                )
   236→
   237→                is CustomizableEnvV2 -> EnvSpec(
   238→                    env.key,
   239→                    EnvType.CUSTOMIZABLE,
   240→                    env.title?.let { entry.titles[it.key] } ?: env.key,
   241→                    emptyList(),
   242→                    env.defaultValue,
   243→                    false,
   244→                    true,
   245→                    prefs[env.key] ?: env.defaultValue.orEmpty()
   246→                )
   247→
   248→                is ToggleableEnvV2 -> EnvSpec(
   249→                    env.key,
   250→                    EnvType.TOGGLEABLE,
   251→                    env.title?.let { entry.titles[it.key] } ?: env.key,
   252→                    emptyList(),
   253→                    null,
   254→                    false,
   255→                    prefs[env.key + ENABLED_SUFFIX]?.toBooleanStrictOrNull() ?: env.toggle,
   256→                    env.value
   257→                )
   258→
   259→                is NormalEnvV2 -> null
   260→            }
   261→        }
   262→    }

   264→    /** 全量保存某插件的环境变量配置：更新内存缓存、异步落盘并重建该插件渲染器 */
   265→    fun updateEnvConfigs(context: Context, packageName: String, values: Map<String, EnvValue>) {
   266→        ensureInit()
   267→        val flat = mutableMapOf<String, String>()
   268→        values.forEach { (key, v) ->
   269→            v.value?.let { flat[key] = it }
   270→            v.enabled?.let { flat[key + ENABLED_SUFFIX] = it.toString() }
   271→        }
   272→        storeEnvPrefs(context, packageName, flat)
   273→        val entry = v2Entries[packageName] ?: return
   274→        val renderer = buildV2Renderer(packageName, entry)
   275→        addRenderer(renderer)
   276→        RendererManager.replaceRenderer(renderer)
   277→    }

   279→    /** 更新内存缓存并异步落盘某键的扁平环境变量配置（插件与内置渲染器共用） */
   280→    private fun storeEnvPrefs(context: Context, key: String, flat: Map<String, String>) {
   281→        envPrefs = envPrefs.toMutableMap().apply { put(key, flat) }
   282→        saveScope.launch {
   283→            runCatching {
   284→                context.appSettingsDataStore.updateData { pref ->
   285→                    pref.copy(plugin = pref.plugin.copy(
   286→                        rendererEnvPrefs = pref.plugin.rendererEnvPrefs.toMutableMap()
   287→                            .apply { put(key, flat) }
   288→                    ))
   289→                }
   290→            }.onFailure { e ->
   291→                Logging.LOG.log(Level.SEVERE, "保存插件环境变量配置失败", e)
   292→            }
   293→        }
   294→    }

   324→    /**

   328→    @JvmStatic
   329→    fun setFpeEmulatorModsPresent(mods: List<LocalModFile>?) {
   330→        val present = mods != null && mods.any { it.isActive && it.id in FPE_EMULATOR_MOD_IDS }
   331→        fpeEmulatorModPresent = present
   332→        if (present) {
   333→            val ids = mods!!.filter { it.isActive && it.id in FPE_EMULATOR_MOD_IDS }
   334→                .joinToString(", ") { it.id }
   335→            Logging.LOG.info("FPE emulator mod detected ($ids), SFPEW compatibility layer disabled")
   336→        }
   337→    }

   350→    /**
   351→     * 该渲染器启动时是否叠加 SFPEW 兼容层（启动链路注入 env 的依据）。
   352→     * 装有自实现 FPE 的模组时整层跳过；
   353→     * 内置渲染器行为已知：默认启用，仅自带 FPE 的内置渲染器硬编码关闭；
   354→     * 外部插件渲染器行为未知：由高级设置里的全局开关控制，默认关闭（用户显式开启）。
   355→     */
   356→    @JvmStatic
   357→    fun isSfpewLayerEnabled(renderer: Renderer): Boolean {
   358→        if (fpeEmulatorModPresent) return false
   359→        if (renderer.path.isEmpty()) {
   360→            return renderer.id !in FPE_CAPABLE_BUILTIN_IDS
   361→        }
   362→        return toggleEnabled(sfpewLayerPrefs(), SFPEW_LAYER_PLUGINS, false)
   363→    }

   495→     * （默认关闭 = 跟随全局序；启用时刻把当前全局序展开为快照落盘，此后不再跟随）。
   496→     * 键名/候选集合/默认序对齐上游 settings.cpp 与 ZL2 #2165。

   498→    private fun multidrawEnvSpecs(prefs: Map<String, String>, context: Context): List<EnvSpec> {
   499→        fun option(value: String): OrderedOption = when (value) {
   500→            "native" -> OrderedOption(

   505→            "unroll" -> OrderedOption("unroll", context.getString(R.string.renderer_config_mg_multidraw_unroll))

   507→            "multiindirect" -> OrderedOption("multiindirect", "MultiIndirect")
   508→            "multibasevertex" -> OrderedOption("multibasevertex", "MultiBaseVertex")
   509→            "multiarrays" -> OrderedOption("multiarrays", "MultiArrays")
   510→            "indirect" -> OrderedOption("indirect", "Indirect")
   511→            "basevertex" -> OrderedOption("basevertex", "BaseVertex")
   512→            else -> OrderedOption(value, value)
   513→        }
   514→
   515→        val globalOrder = MG_MULTIDRAW_ORDERS.getValue("multidrawOrder")
   516→        val globalOrdered = OrderedSpec(options = globalOrder.map(::option), defaultOrder = globalOrder)
   517→        val globalValue = coerceOrder(prefs["multidrawOrder"], globalOrdered)
   518→
   519→        val exceptions = MG_MULTIDRAW_ORDERS.filterKeys { it != "multidrawOrder" }.map { (key, order) ->
   520→            val ordered = OrderedSpec(

   522→                defaultOrder = order,

   524→                nativeValue = MG_MULTIDRAW_NATIVES.getValue(key)

   529→                type = EnvType.ORDERED,
   530→                title = MD_MULTIDRAW_EXCEPTION_TITLES.getValue(key),

   533→                enabled = enabled,

   535→                value = if (enabled) coerceOrder(prefs[key], ordered) else expandOrder(globalValue, ordered),

   594→    /** 生成最终 pojavEnv：可配置项按用户配置取值，DLOPEN 与 dlopenLibPaths 合并为一条 */
   595→    private fun buildPojavEnv(entry: V2Entry, prefs: Map<String, String>): List<String> {
   596→        val dir = entry.nativeLibraryDir
   597→        val result = mutableListOf<String>()
   598→        val dlopen = mutableListOf<String>()
   599→        fun addDlopen(raw: String) {
   600→            raw.split(',').map { it.trim() }.filter { it.isNotEmpty() }.forEach { dlopen.add(it) }
   601→        }
   602→        entry.config.env.forEach { env ->
   603→            when (env) {
   604→                is NormalEnvV2 -> {
   605→                    if (env.key == "DLOPEN") addDlopen(resolveLibValue(env.key, env.value, dir))
   606→                    else result.add("${env.key}=${resolveLibValue(env.key, env.value, dir)}")
   607→                }
   608→
   609→                is SelectableEnvV2 -> {
   610→                    val enabled = if (env.check == null) true
   611→                    else prefs[env.key + ENABLED_SUFFIX]?.toBooleanStrictOrNull() ?: env.check
   612→                    if (enabled) {
   613→                        val value = prefs[env.key] ?: env.items.defaultValue
   614→                        if (value.isNotEmpty()) result.add("${env.key}=$value")
   615→                    }
   616→                }
   617→
   618→                is CustomizableEnvV2 -> {
   619→                    val value = prefs[env.key] ?: env.defaultValue
   620→                    if (!value.isNullOrEmpty()) result.add("${env.key}=$value")
   621→                }
   622→
   623→                is ToggleableEnvV2 -> {
   624→                    val enabled = prefs[env.key + ENABLED_SUFFIX]?.toBooleanStrictOrNull() ?: env.toggle
   625→                    if (enabled) result.add("${env.key}=${resolveLibValue(env.key, env.value, dir)}")
   626→                }
   627→            }
   628→        }
   629→        entry.config.dlopenLibPaths.forEach { addDlopen(resolveLibValue("DLOPEN", it, dir)) }
   630→        if (dlopen.isNotEmpty()) result.add("DLOPEN=${dlopen.distinct().joinToString(",")}")
   631→        return result
   632→    }

   634→    /**
   635→     * 解析环境变量值中的 `**|` 插件库路径：
   636→     * v1 启动链路会对 LIB_MESA_NAME/MESA_LIBRARY/DLOPEN 的值自动拼接插件 nativeLibraryDir，
   637→     * 这些 key 只保留库文件名；其余 key 直接替换为绝对路径。
   638→     */
   639→    private fun resolveLibValue(key: String, raw: String, nativeDir: String): String {
   640→        val joinsDir = key == "LIB_MESA_NAME" || key == "MESA_LIBRARY" || key == "DLOPEN"
   641→        if (raw.startsWith(NATIVE_PREFIX)) {
   642→            val lib = raw.removePrefix(NATIVE_PREFIX)
   643→            return if (joinsDir) lib else "$nativeDir/$lib"
   644→        }
   645→        if (joinsDir && raw.startsWith("/")) return raw.substringAfterLast('/')
   646→        return raw
   647→    }

   692→/** ordered 值规整：剔除未知项、去重保首见、缺失项按默认序补尾（对 config.json 手编值宽容解析） */
   693→fun coerceOrder(raw: String?, spec: RendererPlugin.OrderedSpec): String {
   694→    val known = spec.options.associateBy { it.value }
   695→    val parsed = raw?.split(',')?.mapNotNull { known[it.trim()] }.orEmpty()
   696→    val missing = spec.defaultOrder.mapNotNull { known[it] }
   697→        .filterNot { option -> parsed.any { it.value == option.value } }
   698→    return (parsed + missing).distinctBy { it.value }.joinToString(",") { it.value }
   699→}

   701→/** ordered 继承展开：父序中伪项 native 替换为本条目真实后端后再规整（候选外的项被剔除、缺项补尾） */
   702→fun expandOrder(parentOrder: String, spec: RendererPlugin.OrderedSpec): String =
   703→    coerceOrder(
   704→        parentOrder.split(',').joinToString(",") { segment ->
   705→            val value = segment.trim()
   706→            if (value == "native") spec.nativeValue ?: value else value
   707→        },
   708→        spec
   709→    )
