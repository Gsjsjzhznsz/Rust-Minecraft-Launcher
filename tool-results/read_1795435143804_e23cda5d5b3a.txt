  1280→/// 解析版本 JSON 并装配 LaunchPlan（版本未安装 / JSON 损坏时报错）。
  1281→fn build_plan_for(version_id: &str) -> Result<(model::VersionJson, jvmops::LaunchPlan, u32), String> {
  1282→    let text = std::fs::read_to_string(paths::version_json(version_id))

  1284→    let version: model::VersionJson =

  1290→        .java_version

  1295→    let ctx = instance::standard_rules_ctx();

  1297→        .libraries

  1304→    // Android：移植版 LWJGL 合并 jar 替换 Mojang org.lwjgl:*（glibc natives 不可用）

  1308→        None
  1309→    };

  1311→        ctx: &ctx,
  1312→        version: &version,
  1313→        allowed_libs: &allowed,
  1314→        libraries_root: &paths::libraries_dir(),
  1315→        natives_root: &natives_root,
  1316→        assets_root: &paths::assets_dir(),
  1317→        versions_root: &paths::versions_dir(),
  1318→        instance_dir: &game_dir,

  1320→            .map_err(|e| e.to_string())?

  1322→        memory_mb: 1024,
  1323→        java_major: major,
  1324→        java_path: None,
  1325→        window: None,
  1326→        lwjgl_classes_jar: lwjgl_jar,
  1327→    })

  1329→    Ok((version, plan, major))
  1330→}

  1332→/// Android 12+ tagged-pointer 兼容层 shim 落盘（源码：scripts/tagfix/quartz_tagfix.c）。
  1333→///
  1334→/// Android 12（API 31）起 AArch64 堆指针最高字节被系统保留，OpenJDK top-byte
  1335→/// 用途与之冲突 -> scudo CHECK 失败 abort（SIGABRT，退出码 134，典型崩在 GC
  1336→/// 工作线程）。修复 = java 进程早期 mallopt(M_BIONIC_SET_HEAP_TAGGING_LEVEL,
  1337→/// NONE)（Termux openjdk-17 PR#10347 同款方案）；shim 以 include_bytes 编入
  1338→/// 壳层，启动前写到 JVM pack 私有目录，经 LD_PRELOAD 注入 java 进程。
  1339→#[cfg(target_os = "android")]
  1340→fn ensure_tagfix_lib() -> Option<PathBuf> {
  1341→    const SHIM: &[u8] = include_bytes!("../tagfix/libqztagfix.so");
  1342→    let dst = paths::runtime_dir().join("libqztagfix.so");
  1343→    if let Ok(meta) = std::fs::metadata(&dst) {
  1344→        if meta.len() == SHIM.len() as u64 {
  1345→            return Some(dst);
  1346→        }
  1347→    }
  1348→    if std::fs::write(&dst, SHIM).is_ok() {
  1349→        use std::os::unix::fs::PermissionsExt;
  1350→        let _ = std::fs::set_permissions(&dst, std::fs::Permissions::from_mode(0o644));
  1351→        return Some(dst);
  1352→    }
  1353→    None
  1354→}

  1356→/// 拉起 JVM 子进程并托管日志/退出（`launch://log` / `launch://exit` 事件）。
  1357→///
  1358→/// 前置：JVM pack 已安装（Android 内部目录，targetSdk 28 允许 exec）。
  1359→/// smoke test 先跑 `java -version` 把 JRE 版本写进控制台，失败时给出明确报错。
  1360→fn spawn_game(
  1361→    version_id: String,
  1362→    plan: jvmops::LaunchPlan,
  1363→    major: u32,
  1364→    game_pid: Arc<Mutex<Option<u32>>>,
  1365→    app: AppHandle,
  1366→) -> Result<(), String> {
  1367→    use std::io::BufRead;
  1368→    use std::io::Write;
  1369→    use std::process::{Command, Stdio};
  1370→
  1371→    let java = runtime::java_bin(major);
  1372→    let jre_home = runtime::install_dir(major);
  1373→    let game_dir = paths::minecraft_dir();
  1374→
  1375→    // ---- 持久化启动日志（完整证据链）----
  1376→    // stdout/stderr/quartz 行 + 完整命令行/环境 全部落盘 <game_root>/logs/launch-*.log。
  1377→    // 原生崩溃（SIGABRT 等）的 JVM 报告只写 hs_err 文件 / logcat，不会进 stderr ——
  1378→    // 因此异常退出时自动摘录 hs_err 到本日志，下次运行即可拿到全部诊断信息。
  1379→    let logs_dir = paths::game_root().join("logs");
  1380→    let _ = std::fs::create_dir_all(&logs_dir);
  1381→    let log_path = logs_dir.join(format!(
  1382→        "launch-{}.log",
  1383→        std::time::SystemTime::now()
  1384→            .duration_since(std::time::UNIX_EPOCH)
  1385→            .map(|d| d.as_secs())
  1386→            .unwrap_or(0)
  1387→    ));
  1388→    let log_file: Arc<Mutex<Option<std::fs::File>>> =
  1389→        Arc::new(Mutex::new(std::fs::File::create(&log_path).ok()));
  1390→
  1391→    // 控制台 + 文件双写。
  1392→    let qlog = {
  1393→        let app = app.clone();
  1394→        let log_file = log_file.clone();
  1395→        move |source: &str, line: &str| {
  1396→            let _ = app.emit(
  1397→                "launch://log",
  1398→                &LaunchLog { source: source.into(), line: line.into() },
  1399→            );
  1400→            if let Ok(mut guard) = log_file.lock() {
  1401→                if let Some(f) = guard.as_mut() {
  1402→                    let _ = writeln!(f, "[{source}] {line}");
  1403→                    let _ = f.flush();
  1404→                }
  1405→            }
  1406→        }
  1407→    };
  1408→
  1409→    // ---- JVM 动态库搜索路径（必须先于 smoke test 计算，两处共用） ----
  1410→    // bionic linker 不认 RUNPATH，bin/java 找 libjli.so 完全依赖 LD_LIBRARY_PATH。
  1411→    // 顺序（先者优先）：渲染层目录 -> LWJGL 安卓 natives -> 游戏 natives ->
  1412→    // JRE lib / lib/jli / lib/server / lib/client。
  1413→    let mut ld_parts: Vec<String> = Vec::new();
  1414→    #[cfg(target_os = "android")]
  1415→    let active_renderer = {
  1416→        let id = renderers::active();
  1417→        let spec = renderers::spec_for(&id);
  1418→        if renderers::is_installed(&id) {
  1419→            ld_parts.push(
  1420→                renderers::install_dir(&id)
  1421→                    .to_string_lossy()
  1422→                    .into_owned(),
  1423→            );
  1424→        }
  1425→        if lwjgl::is_installed() {
  1426→            ld_parts.push(lwjgl::natives_dir().to_string_lossy().into_owned());
  1427→        }
  1428→        qlog(
  1429→            "quartz",
  1430→            &format!(
  1431→                "[quartz] 渲染层：{}（{}）｜ LWJGL 安卓包：{}",
  1432→                spec.map(|s| s.name).unwrap_or("未知"),
  1433→                id,
  1434→                if lwjgl::is_installed() { "已就绪" } else { "未安装" },
  1435→            ),
  1436→        );
  1437→        id
  1438→    };
  1439→    #[cfg(not(target_os = "android"))]
  1440→    let active_renderer = String::new();
  1441→    ld_parts.push(jre_home.join("lib").display().to_string());
  1442→    ld_parts.push(jre_home.join("lib/jli").display().to_string());
  1443→    ld_parts.push(jre_home.join("lib/server").display().to_string());
  1444→    ld_parts.push(jre_home.join("lib/client").display().to_string());
  1445→    ld_parts.push(plan.natives_dir.display().to_string());
  1446→    let ld_library_path = ld_parts.join(":");
  1447→
  1448→    // Android 无 /tmp：JVM 的 hsperfdata / 临时文件目录必须显式指向内部可写目录
  1449→    //（tmpdir 初始化失败是 Pojav 系 JRE 早期 abort 的常见诱因）。
  1450→    #[cfg(target_os = "android")]
  1451→    let tmp_dir = {
  1452→        let d = paths::runtime_dir().join("tmp");
  1453→        let _ = std::fs::create_dir_all(&d);
  1454→        d
  1455→    };
  1456→    #[cfg(not(target_os = "android"))]
  1457→    let tmp_dir = std::env::temp_dir();
  1458→
  1459→    // ---- Android JVM 参数包：java.library.path 私有目录优先 + 渲染层接线 ----
  1460→    // FUSE（/storage/emulated/0）挂载为 noexec，System.loadLibrary/dlopen 无法从
  1461→    // 公共目录装载 native：顺序必须是 LWJGL natives（含 pojavexec/jnidispatch）->
  1462→    // 渲染层 -> 游戏 natives（FUSE，仅兜底）。附加在 plan 参数之后（同名 -D 后者生效）。
  1463→    #[cfg(target_os = "android")]
  1464→    let (jlp, android_jvm_extra) = {
  1465→        let jlp = format!(
  1466→            "{}:{}:{}",
  1467→            lwjgl::natives_dir().display(),
  1468→            renderers::install_dir(&active_renderer).display(),
  1469→            plan.natives_dir.display(),
  1470→        );
  1471→        let gl_lib = renderers::spec_for(&active_renderer)
  1472→            .map(|s| s.gl_lib)
  1473→            .unwrap_or("libgl4es_114.so");
  1474→        let extra = vec![
  1475→            format!("-Djava.library.path={jlp}"),
  1476→            // liblwjgl_opengl 按该属性 dlopen 渲染器 GL 库；PojavRendererInit 也据它识别渲染层
  1477→            format!("-Dorg.lwjgl.opengl.libname={gl_lib}"),
  1478→            // tmpdir 显式内部化 + 关闭 hsperfdata（部分安卓构建在 /tmp 缺失时初始化 abort）
  1479→            format!("-Djava.io.tmpdir={}", tmp_dir.display()),
  1480→            "-XX:-UsePerfData".to_string(),
  1481→            // JVM 崩溃报告定向到日志目录（默认写 CWD，FUSE 上难定位）
  1482→            format!("-XX:ErrorFile={}", logs_dir.join("hs_err_%p.log").display()),
  1483→            // 早期初始化诊断：abort 若发生在 JVM 初始化阶段，stderr 会留下最后一步
  1484→            "-Xlog:init=info:stderr".to_string(),
  1485→        ];
  1486→        (jlp, extra)
  1487→    };
  1488→    #[cfg_attr(not(target_os = "android"), allow(unused_variables))]
  1489→    #[cfg(not(target_os = "android"))]
  1490→    let (jlp, android_jvm_extra): (String, Vec<String>) = (String::new(), Vec::new());
  1491→
  1492→    // ---- Android 12+ tagged-pointer 兼容层（JRE GC 线程 SIGABRT 修复）----
  1493→    // 落盘 shim 并以 LD_PRELOAD 注入（smoke test / 二分自检 / 真实启动共用）。
  1494→    #[cfg(target_os = "android")]
  1495→    let tagfix_lib = ensure_tagfix_lib();
  1496→    #[cfg(target_os = "android")]
  1497→    {
  1498→        match tagfix_lib.as_ref() {
  1499→            Some(p) => qlog(
  1500→                "quartz",
  1501→                &format!(
  1502→                    "[quartz] tagged-pointer 兼容层：{}（LD_PRELOAD 注入）",
  1503→                    p.display()
  1504→                ),
  1505→            ),
  1506→            None => qlog("quartz", "[quartz] tagged-pointer 兼容层落盘失败（跳过注入）"),
  1507→        }
  1508→    }
  1509→
  1510→    // ---- 头部：完整命令行 + 环境（诊断核心：一眼看出坏参数/坏路径） ----
  1511→    if let Ok(mut guard) = log_file.lock() {
  1512→        if let Some(f) = guard.as_mut() {
  1513→            let _ = writeln!(f, "==== Quartz launch {version_id} (java {major}) ====");
  1514→            let _ = writeln!(f, "java: {}", java.display());
  1515→            let _ = writeln!(f, "cwd: {}", plan.working_dir.display());
  1516→            let _ = writeln!(f, "JAVA_HOME: {}", jre_home.display());
  1517→            let _ = writeln!(f, "LD_LIBRARY_PATH: {ld_library_path}");
  1518→            let _ = writeln!(f, "TMPDIR: {}", tmp_dir.display());
  1519→            #[cfg(target_os = "android")]
  1520→            {
  1521→                let _ = writeln!(f, "java.library.path: {jlp}");
  1522→                let gl_lib = renderers::spec_for(&active_renderer)
  1523→                    .map(|s| s.gl_lib)
  1524→                    .unwrap_or("libgl4es_114.so");
  1525→                let _ = writeln!(f, "org.lwjgl.opengl.libname: {gl_lib}");
  1526→                let _ = writeln!(
  1527→                    f,
  1528→                    "LD_PRELOAD: {}",
  1529→                    tagfix_lib
  1530→                        .as_ref()
  1531→                        .map(|p| p.display().to_string())

  1795→            "launch://exit",

  1798→                version: version_id,

  1801→        );
  1802→    });
  1803→    Ok(())
  1804→}

  1806→/// ExitStatus -> 退出码：正常退出取 code()；信号死亡取 128+signal。
  1807→fn exit_status_code(st: &std::process::ExitStatus) -> i32 {
  1808→    #[cfg(unix)]
  1809→    {
  1810→        use std::os::unix::process::ExitStatusExt;
  1811→        st.code()
  1812→            .or_else(|| st.signal().map(|sig| 128 + sig))
  1813→            .unwrap_or(-1)
  1814→    }
  1815→    #[cfg(not(unix))]
  1816→    {
  1817→        st.code().unwrap_or(-1)
  1818→    }
  1819→}

  1821→/// 原生层崩溃（SIGABRT 等）的证据摘录。
  1822→///
  1823→/// bionic 的 abort 不写 stderr、JVM 不写 hs_err，唯一记录在 logd：
  1824→/// crash 缓冲（Fatal signal / Abort message）与 main 缓冲（同 uid 可读）。
  1825→/// 另附运行时组件清单，验证 pack 落位与 so 齐全。
  1826→#[cfg(target_os = "android")]
  1827→fn dump_crash_evidence(
  1828→    pid: Option<u32>,
  1829→    qlog: &impl Fn(&str, &str),
  1830→    log_file: &Arc<Mutex<Option<std::fs::File>>>,
  1831→) {
  1832→    use std::io::Write;
  1833→    let write_log = |text: &str| {
  1834→        if let Ok(mut guard) = log_file.lock() {
  1835→            if let Some(f) = guard.as_mut() {
  1836→                let _ = writeln!(f, "{text}");
  1837→                let _ = f.flush();
  1838→            }
  1839→        }
  1840→    };
  1841→    // ---- logcat：crash 缓冲优先，main 缓冲兜底（各限 30 行防刷屏） ----
  1842→    for buf in ["crash", "main"] {
  1843→        let out = std::process::Command::new("logcat")
  1844→            .args(["-d", "-b", buf, "-t", "300"])
  1845→            .output();
  1846→        match out {
  1847→            Ok(out) => {
  1848→                let text = String::from_utf8_lossy(&out.stdout);
  1849→                let mut kept = 0usize;
  1850→                for line in text.lines().rev() {
  1851→                    if kept >= 30 {
  1852→                        break;
  1853→                    }
  1854→                    let hit = match pid {
  1855→                        // threadtime 格式：date time PID TID level tag: msg
  1856→                        Some(p) => line
  1857→                            .split_whitespace()
  1858→                            .nth(2)
  1859→                            .is_some_and(|f| f == p.to_string()),
  1860→                        None => {
  1861→                            line.contains("Fatal signal")
  1862→                                || line.contains("Abort message")
  1863→                                || line.contains("SIGABRT")
  1864→                                || line.contains("SIGSEGV")
  1865→                                || line.contains("SIGBUS")
  1866→                                || line.contains("CheckJNI")
  1867→                        }
  1868→                    };
  1869→                    if hit {
  1870→                        qlog("logcat", line);
  1871→                        write_log(&format!("[logcat-{buf}] {line}"));
  1872→                        kept += 1;
  1873→                    }
  1874→                }
  1875→                if kept == 0 {
  1876→                    qlog(
  1877→                        "quartz",
  1878→                        &format!("[quartz] logcat {buf} 缓冲无匹配记录（可能已被滚动或无权限）"),
  1879→                    );
  1880→                }
  1881→            }
  1882→            Err(e) => {
  1883→                qlog("quartz", &format!("[quartz] logcat 不可用（{e}）"));
  1884→            }
  1885→        }
  1886→    }
  1887→    // ---- 组件清单：确认运行时 pack 实际落位 ----
  1888→    let dirs: [(&str, PathBuf); 4] = [
  1889→        ("lwjgl3/natives", lwjgl::natives_dir()),
  1890→        (
  1891→            "renderers/active",
  1892→            renderers::install_dir(&renderers::active()),
  1893→        ),
  1894→        ("jre lib/server", runtime::install_dir(17).join("lib/server")),
  1895→        ("game natives", paths::minecraft_dir().join("natives")),
  1896→    ];
  1897→    for (label, dir) in dirs {
  1898→        let mut names: Vec<String> = Vec::new();
  1899→        if let Ok(rd) = std::fs::read_dir(&dir) {
  1900→            for e in rd.flatten().take(40) {
  1901→                let size = e.metadata().map(|m| m.len()).unwrap_or(0);
  1902→                names.push(format!("{}({})", e.file_name().to_string_lossy(), size));
  1903→            }
  1904→        }
  1905→        let line = if names.is_empty() {
  1906→            format!("[quartz] 组件清单 {label}：空或不可读（{}）", dir.display())
  1907→        } else {
  1908→            format!("[quartz] 组件清单 {label}：{}", names.join(", "))
  1909→        };
  1910→        qlog("quartz", &line);
  1911→        write_log(&line);
  1912→    }
  1913→}

  1915→/// Android 启动前置：LWJGL 安卓 pack 缺失时阻塞在线下载（进度写启动控制台）。

  1946→/// Android 启动前置：当前渲染层缺失时阻塞在线下载。
  1947→#[cfg(target_os = "android")]
  1948→fn ensure_renderer(app: &AppHandle) -> Result<(), String> {
  1949→    let active = renderers::active();
  1950→    if renderers::is_installed(&active) {
  1951→        return Ok(());
  1952→    }
  1953→    let has_source = renderers::spec_for(&active)
  1954→        .map(|s| s.source.is_some())

  1956→    if !has_source {
  1957→        return Err(format!("所选渲染层 {active} 暂无下载源，请在设置页更换"));
  1958→    }
  1959→    let _ = app.emit(
  1960→        "launch://log",
  1961→        &LaunchLog {
  1962→            source: "quartz".into(),
  1963→            line: format!("[quartz] 首次使用渲染层 {active}：在线获取…"),
  1964→        },
  1965→    );
  1966→    let (tx, mut rx) = tokio::sync::mpsc::channel::<DownloadProgress>(64);
  1967→    let app2 = app.clone();
  1968→    tauri::async_runtime::spawn(async move {
  1969→        while let Some(ev) = rx.recv().await {
  1970→            let line = if ev.stage == "error" {
  1971→                format!("[quartz] 渲染层下载错误：{}", ev.error.unwrap_or_default())
  1972→            } else {
  1973→                format!("[quartz] 渲染层下载 {}%", ev.percent as u32)
  1974→            };
  1975→            let _ = app2.emit("launch://log", &LaunchLog { source: "quartz".into(), line });
  1976→        }
  1977→    });
