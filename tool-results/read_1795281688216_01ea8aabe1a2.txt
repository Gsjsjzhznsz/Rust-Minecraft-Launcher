     1→//! Quartz Tauri 壳层。
     2→//!
     3→//! 架构契约（方案 A）：UI（WebView/JS）只调用本文件注册的 command；
     4→//! command 是 quartz-core 能力的薄封装，不含业务逻辑。
     5→//!
     6→//! 命令契约（v3 蓝图 + M0 启动接入）：
     7→//! - `offline_login(name)`        离线登录：MD5("OfflinePlayer:"+name) -> accounts.json
     8→//! - `get_account()`              当前选中账号
     9→//! - `list_accounts()`            全部账号 + 选中标记
    10→//! - `select_account(uuid)`       切换选中账号
    11→//! - `remove_account(uuid)`       删除账号
    12→//! - `list_versions()`            真实 manifest 版本列表（release 优先）
    13→//! - `list_installed_versions()`  本地 versions/ 已安装版本
    14→//! - `download_game(version_id)`  后台 spawn 完整下载管线，
    15→//!                                经 `download://progress` 事件上报 {stage,current,total,percent,file}
    16→//! - `get_game_dir()`             标准游戏根目录绝对路径（UI 常显）
    17→//! - `get_env_info()`             存储诊断：根目录 + 可写性 + 存储模式 + 完整文件权限
    18→//! - `get_permission_state()`     存储权限状态 {granted}
    19→//! - `request_storage_permission()` 触发系统权限弹窗（授权后经 focus/visibility 重检）
    20→//! - `reinit_storage()`           权限变化后重解析存储根 + 后台迁移旧数据
    21→//!                                （`quartz://migrate` 事件上报迁移进度）
    22→//! - `launch_preflight(version_id)` 启动前自检（版本/账号/Java 运行时）
    23→//! - `get_runtime_status(java_major)` JVM pack 状态（installed/version/path）
    24→//! - `download_runtime(java_major)` 下载 JVM 运行时 pack（`runtime://progress` 事件）
    25→//! - `launch_game(version_id)`    M0 已接入：JRE pack + LaunchPlan 子进程启动，
    26→//!                                `launch://log` / `launch://exit` 事件实时上报
    27→//! - `stop_game()`                终止游戏进程（SIGKILL）

    52→#[cfg(target_os = "android")]
    53→fn with_jni<T>(
    54→    f: impl FnOnce(&mut jni::JNIEnv, &jni::objects::JObject) -> Option<T>,
    55→) -> Option<T> {
    56→    std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
    57→        let ctx = ndk_context::android_context();
    58→        let vm = unsafe { jni::JavaVM::from_raw(ctx.vm().cast()) }.ok()?;
    59→        let mut env = vm.attach_current_thread().ok()?;
    60→        let context = unsafe { jni::objects::JObject::from_raw(ctx.context().cast()) };
    61→        f(&mut env, &context)
    62→    }))
    63→    .ok()
    64→    .flatten()
    65→}

    67→#[cfg(target_os = "android")]
    68→fn jni_file_path(env: &mut jni::JNIEnv, file: &jni::objects::JObject) -> Option<PathBuf> {
    69→    let path = env
    70→        .call_method(file, "getAbsolutePath", "()Ljava/lang/String;", &[])
    71→        .ok()?
    72→        .l()
    73→        .ok()?;
    74→    let s: String = env.get_string((&path).into()).ok()?.into();
    75→    if s.is_empty() {
    76→        None
    77→    } else {
    78→        Some(PathBuf::from(s))
    79→    }
    80→}

    82→#[cfg(target_os = "android")]
    83→fn jni_external_files_dir(env: &mut jni::JNIEnv, context: &jni::objects::JObject) -> Option<PathBuf> {
    84→    use jni::objects::JValue;
    85→    let file = env
    86→        .call_method(
    87→            context,
    88→            "getExternalFilesDir",
    89→            "(Ljava/lang/String;)Ljava/io/File;",
    90→            &[JValue::Object(&jni::objects::JObject::null())],
    91→        )
    92→        .ok()?
    93→        .l()
    94→        .ok()?;
    95→    if file.is_null() {
    96→        return None;
    97→    }
    98→    jni_file_path(env, &file)
    99→}

   115→// 存储权限（完整文件权限）
   116→// ---------------------------------------------------------------------------
   117→//
   118→// 设计：APK 固定 targetSdk 28（Android 10 前的存储模型 + 允许 exec() 应用目录
   119→// 内的 JVM —— Pojav/Termux 同款方案）。READ/WRITE_EXTERNAL_STORAGE 经运行时
   120→// 弹窗授权后，应用对 /storage/emulated/0 拥有完整读写权限，游戏文件即可放在
   121→// 文件管理器可见的标准目录 /storage/emulated/0/Quartz/files。

   123→/// Android 端 WRITE_EXTERNAL_STORAGE 是否已授权（checkSelfPermission == GRANTED）。
   124→#[cfg(target_os = "android")]
   125→fn android_storage_permission_granted() -> bool {
   126→    use jni::objects::JValue;
   127→    with_jni(|env, ctx| {
   128→        let perm = env
   129→            .new_string("android.permission.WRITE_EXTERNAL_STORAGE")
   130→            .ok()?;
   131→        let granted = env
   132→            .call_method(
   133→                ctx,
   134→                "checkSelfPermission",
   135→                "(Ljava/lang/String;)I",
   136→                &[JValue::Object(&perm.into())],
   137→            )
   138→            .ok()?
   139→            .i()
   140→            .ok()?;
   141→        // PackageManager.PERMISSION_GRANTED == 0
   142→        Some(granted == 0)
   143→    })
   144→    .unwrap_or(false)
   145→}

   147→/// 存储权限是否已授权（桌面端恒真）。
   148→fn storage_permission_granted() -> bool {
   149→    #[cfg(target_os = "android")]
   150→    {
   151→        android_storage_permission_granted()
   152→    }
   153→    #[cfg(not(target_os = "android"))]
   154→    {
   155→        true
   156→    }
   157→}

   165→/// 触发系统权限弹窗（WRITE/READ_EXTERNAL_STORAGE）。
   166→/// 结果不通过本命令返回：UI 在 focus/visibilitychange 时重新查询状态。
   167→#[tauri::command]
   168→fn request_storage_permission() -> Result<bool, String> {
   169→    #[cfg(target_os = "android")]
   170→    {
   171→        use jni::objects::JValue;
   172→        with_jni(|env, ctx| {
   173→            let write = env.new_string("android.permission.WRITE_EXTERNAL_STORAGE").ok()?;
   174→            let read = env.new_string("android.permission.READ_EXTERNAL_STORAGE").ok()?;
   175→            let write_obj: jni::objects::JObject = write.into();
   176→            let arr = env
   177→                .new_object_array(2, "java/lang/String", write_obj)
   178→                .ok()?;
   179→            let read_obj: jni::objects::JObject = read.into();
   180→            env.set_object_array_element(&arr, 1, read_obj).ok()?;
   181→            env.call_method(
   182→                ctx,
   183→                "requestPermissions",
   184→                "([Ljava/lang/String;I)V",
   185→                &[JValue::Object(&arr), JValue::Int(42)],
   186→            )
   187→            .ok()?;
   188→            Some(())
   189→        });
   190→        Ok(true)
   191→    }
   192→    #[cfg(not(target_os = "android"))]
   193→    {
   194→        Ok(true)
   195→    }
   196→}

   202→/// 存储诊断信息（启动时确定，UI 经 get_env_info 展示）。

   204→pub struct StorageDiag {

   211→static STORAGE_DIAG: Mutex<Option<StorageDiag>> = Mutex::new(None);

   213→/// Android 平台解析真实可写目录；ndk-context 未初始化等异常一律回退硬编码路径。
   214→///
   215→/// 优先级（与授权状态联动）：
   216→/// 1. 已授权完整文件权限 -> 公共标准目录 /storage/emulated/0/Quartz/files（可见、卸载不丢）
   217→/// 2. 应用外部私有目录（getExternalFilesDir，可浏览，卸载丢失）
   218→/// 3. 应用内部私有目录（getFilesDir，必定可写）
   219→#[cfg(target_os = "android")]
   220→fn resolve_android_root() -> (PathBuf, &'static str) {
   221→    if storage_permission_granted() {
   222→        let public = PathBuf::from("/storage/emulated/0/Quartz/files");
   223→        if std::fs::create_dir_all(&public).is_ok() {
   224→            return (public, "public");
   225→        }
   226→        eprintln!("[quartz] 公共标准目录创建失败，回退应用私有目录");
   227→    }
   228→    let result = with_jni(|env, context| {
   229→        // 外部私有目录（用户可用文件管理器查看 .minecraft）
   230→        if let Some(p) = jni_external_files_dir(env, context) {
   231→            return Some((p, "external"));
   232→        }
   233→        // 内部私有目录（必定可写）
   234→        jni_internal_files_dir(env, context).map(|p| (p, "internal"))
   235→    });
   236→    result.unwrap_or_else(|| {
   237→        (
   238→            PathBuf::from(format!(
   239→                "/storage/emulated/0/Android/data/{}/files",
   240→                paths::ANDROID_PACKAGE_ID
   241→            )),
   242→            "fallback",
   243→        )
   244→    })
   245→}

   247→/// Android 内部私有目录（Context.getFilesDir()，任何设备必定可写；exec 可用）。
   248→#[cfg(target_os = "android")]
   249→fn resolve_android_internal() -> Option<PathBuf> {
   250→    with_jni(|env, context| jni_internal_files_dir(env, context))
   251→}

   253→/// 初始化游戏根目录与 JVM pack 目录并创建标准布局；返回诊断信息。
   254→fn init_storage() -> StorageDiag {
   255→    #[cfg(target_os = "android")]
   256→    let (root, mode) = resolve_android_root();
   257→    #[cfg(not(target_os = "android"))]
   258→    let (root, mode) = (paths::game_root(), "desktop");
   259→
   260→    let _ = paths::set_runtime_root(root.clone());
   261→    let mut layout = paths::ensure_layout();
   262→
   263→    // 外部目录不可写（EACCES / os error 13）时回退内部私有目录，
   264→    // 保证离线登录与下载在任何设备上永远可用。
   265→    #[cfg(target_os = "android")]
   266→    let mut mode = mode;
   267→    #[cfg(target_os = "android")]
   268→    if layout.is_err() && mode != "internal" {
   269→        eprintln!(
   270→            "[quartz] 当前根目录不可写（{:?}），回退内部存储",
   271→            layout.as_ref().err()
   272→        );
   273→        if let Some(internal) = resolve_android_internal() {
   274→            let _ = paths::set_runtime_root(internal.clone());
   275→            layout = paths::ensure_layout();
   276→            mode = "internal";
   277→        }
   278→    }
   279→
   280→    // JVM pack 必须放在可 exec 的内部目录（FUSE noexec + targetSdk>=29 禁 exec 的约束）
   281→    #[cfg(target_os = "android")]
   282→    if let Some(internal) = resolve_android_internal() {
   283→        paths::set_runtime_jres_root(internal.join("jres"));
   284→    }
   285→
   286→    let diag = StorageDiag {
   287→        mode,
   288→        root: paths::game_root(),
   289→        layout_ok: layout.is_ok(),
   290→        layout_error: layout.err().map(|e| format!("{e}")),
   291→    };
   292→    eprintln!(
   293→        "[quartz] 存储初始化：mode={} root={} jres={}",
   294→        diag.mode,
   295→        diag.root.display(),
   296→        paths::runtime_dir().display()
   297→    );
   298→    diag
   299→}

   311→/// 递归收集「源存在且目标缺失/不一致」的文件（不覆盖目标已有的同名同尺寸文件）。
   312→fn collect_migrate_files(src: &Path, dst: &Path, out: &mut Vec<(PathBuf, PathBuf, u64)>) {
   313→    let Ok(entries) = std::fs::read_dir(src) else {
   314→        return;
   315→    };
   316→    for entry in entries.flatten() {
   317→        let sp = entry.path();
   318→        let dp = dst.join(entry.file_name());
   319→        if sp.is_dir() {
   320→            collect_migrate_files(&sp, &dp, out);
   321→        } else if sp.is_file() {
   322→            let size = entry.metadata().map(|m| m.len()).unwrap_or(0);
   323→            let dst_size = std::fs::metadata(&dp).map(|m| m.len()).unwrap_or(u64::MAX);
   324→            if dst_size != size {
   325→                out.push((sp, dp, size));
   326→            }
   327→        }
   328→    }
   329→}

   331→/// 后台迁移：把旧根（私有目录）的数据搬到新根（公共标准目录），
   332→/// 经 `quartz://migrate` 事件上报 {percent,done,error}。
   333→fn migrate_root(from: PathBuf, to: PathBuf, app: AppHandle) {
   334→    let mut plan: Vec<(PathBuf, PathBuf, u64)> = Vec::new();
   335→    collect_migrate_files(&from.join(".minecraft"), &to.join(".minecraft"), &mut plan);
   336→    let old_accounts = from.join("accounts.json");
   337→    let new_accounts = to.join("accounts.json");
   338→    let migrate_accounts = old_accounts.exists() && !new_accounts.exists();
   339→    if migrate_accounts {
   340→        let size = std::fs::metadata(&old_accounts).map(|m| m.len()).unwrap_or(0);
   341→        plan.push((old_accounts, new_accounts, size));
   342→    }
   343→
   344→    if plan.is_empty() {
   345→        let _ = app.emit(
   346→            "quartz://migrate",
   347→            &json!({ "percent": 100.0, "done": true, "error": null }),
   348→        );
   349→        return;
   350→    }
   351→
   352→    let total: u64 = plan.iter().map(|(_, _, s)| *s).sum::<u64>().max(1);
   353→    eprintln!("[quartz] 开始迁移存储：{} 个文件（{} 字节）", plan.len(), total);
   354→    let mut copied: u64 = 0;
   355→    let mut last = std::time::Instant::now() - std::time::Duration::from_secs(1);
   356→    for (src, dst, size) in plan {
   357→        if let Some(parent) = dst.parent() {
   358→            if std::fs::create_dir_all(parent).is_err() {
   359→                let _ = app.emit(
   360→                    "quartz://migrate",
   361→                    &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false,
   362→                             "error": format!("无法创建目录 {}", parent.display()) }),
   363→                );
   364→                return;
   365→            }
   366→        }
   367→        if std::fs::copy(&src, &dst).is_err() {
   368→            let _ = app.emit(
   369→                "quartz://migrate",
   370→                &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false,
   371→                         "error": format!("复制失败：{} -> {}", src.display(), dst.display()) }),
   372→            );
   373→            return;
   374→        }
   375→        copied += size;
   376→        let now = std::time::Instant::now();
   377→        if now.duration_since(last) >= std::time::Duration::from_millis(150) {
   378→            last = now;
   379→            let _ = app.emit(
   380→                "quartz://migrate",
   381→                &json!({ "percent": (copied as f64 / total as f64) * 100.0, "done": false, "error": null }),
   382→            );
   383→        }
   384→    }
   385→    eprintln!("[quartz] 存储迁移完成");
   386→    let _ = app.emit(
   387→        "quartz://migrate",
   388→        &json!({ "percent": 100.0, "done": true, "error": null }),
   389→    );
   390→}

   392→/// 权限变化后重解析存储根；根变化时后台迁移旧数据（幂等，跳过目标已有文件）。
   393→#[tauri::command]
   394→fn reinit_storage(app: AppHandle) -> Result<serde_json::Value, String> {
   395→    let old_root = paths::game_root();
   396→    let diag = init_storage();
   397→    *STORAGE_DIAG.lock().map_err(|_| "内部状态锁异常")? = Some(diag.clone());
   398→    let new_root = paths::game_root();
   399→    if old_root != new_root && old_root.exists() {
   400→        let app2 = app.clone();
   401→        std::thread::spawn(move || migrate_root(old_root, new_root, app2));
   402→    }
   403→    Ok(storage_json(&diag))
   404→}

   434→/// 渲染器建议：按平台默认能力 + 手动覆盖。
   435→#[tauri::command]
   436→fn renderer_advice(mc: String, caps: Option<serde_json::Value>) -> serde_json::Value {
   437→    let hal = platform::default_hal();
   438→    let os = hal.os().bus_name();
   439→    let caps = caps
   440→        .and_then(|v| serde_json::from_value::<DeviceCaps>(v).ok())
   441→        .unwrap_or_else(|| DeviceCaps::for_os(os));
   442→    let required = renderbus::required_profile(&mc);
   443→    match renderbus::select(os, host_arch(), required, &caps) {
   444→        Some(p) => json!({
   445→            "id": p.id, "backend": p.backend, "provides": p.provides,
   446→            "note": p.note, "required": required,
   447→        }),
   448→        None => json!({ "id": null, "required": required,
   449→            "reason": "设备无满足 GL 语义的加速路径（性能红线）" }),
   450→    }
   451→}

   453→/// 生成启动计划（离线演示：内嵌 1.20.4 样例 + 离线账号）。
   454→#[tauri::command]
   455→fn plan_offline(player: String, memory_mb: u32) -> Result<serde_json::Value, String> {
   456→    let version: model::VersionJson =
   457→        serde_json::from_str(quartz_core::fixtures::VERSION_1_20_4)
   458→            .map_err(|e| e.to_string())?;
   459→    let hal = platform::default_hal();
   460→    let ctx = RuntimeCtx::new("linux", host_arch());
   461→    let allowed: Vec<&model::Library> = version
   462→        .libraries
   463→        .iter()
   464→        .filter(|l| meta::library_allowed(l, &ctx))
   465→        .collect();
   466→    let account = auth::offline_account(&player);

   572→/// 拉取真实 version_manifest_v2 并排序：release -> snapshot -> old_beta -> old_alpha。
   573→#[tauri::command]
   574→async fn list_versions() -> Result<Vec<VersionInfo>, String> {
   575→    let client = meta::http_client();
   576→    let manifest = meta::fetch_version_manifest(&client)
   577→        .await
   578→        .map_err(|e| format!("版本清单拉取失败：{e}"))?;
   579→    const KIND_ORDER: [&str; 4] = ["release", "snapshot", "old_beta", "old_alpha"];
   580→    let mut out = Vec::with_capacity(manifest.versions.len());
   581→    for kind in KIND_ORDER {
   582→        for v in manifest.versions.iter().filter(|v| v.kind == kind) {
   583→            out.push(VersionInfo {
   584→                id: v.id.clone(),
   585→                version_type: v.kind.clone(),
   586→                url: v.url.clone(),
   587→            });
   588→        }
   589→    }
   590→    for v in manifest
   591→        .versions
   592→        .iter()
   593→        .filter(|v| !KIND_ORDER.contains(&v.kind.as_str()))
   594→    {
   595→        out.push(VersionInfo {
   596→            id: v.id.clone(),
   597→            version_type: v.kind.clone(),
   598→            url: v.url.clone(),
   599→        });
   600→    }
   601→    Ok(out)
   602→}

   625→/// 后台启动完整下载管线（8 并发 + SHA-1 校验 + 3 次重试 + 原子落盘），
   626→/// 进度经 `download://progress` 事件上报 {stage,current,total,percent,file,done,error}。
   627→#[tauri::command]
   628→async fn download_game(
   629→    version_id: String,
   630→    app: AppHandle,
   631→    state: State<'_, AppState>,
   632→) -> Result<(), String> {
   633→    {
   634→        let mut busy = state.downloading.lock().map_err(|_| "内部状态锁异常")?;
   635→        if *busy {
   636→            return Err("已有下载任务正在进行，请等待完成后再试".to_string());
   637→        }
   638→        *busy = true;
   639→    }
   640→    let busy = state.downloading.clone();
   641→    let app = app.clone();
   642→
   643→    tauri::async_runtime::spawn(async move {
   644→        let (tx, mut rx) = tokio::sync::mpsc::channel::<instance::DownloadProgress>(128);
   645→        let forwarder_app = app.clone();
   646→        let forwarder = tokio::spawn(async move {
   647→            while let Some(ev) = rx.recv().await {
   648→                let _ = forwarder_app.emit("download://progress", &ev);
   649→            }
   650→        });
   651→
   652→        let client = meta::http_client();
   653→        let result = instance::install_standard(&client, &version_id, tx).await;
   654→        if let Err(e) = &result {
   655→            // 失败也走事件通道（UI 红字报错 + 可重试），绝不静默。
   656→            let _ = app.emit(
   657→                "download://progress",
   658→                &instance::DownloadProgress {
   659→                    stage: "error".to_string(),
   660→                    current: 0,
   661→                    total: 0,
   662→                    percent: 0.0,
   663→                    file: String::new(),
   664→                    done: false,
   665→                    error: Some(e.to_string()),
   666→                },
   667→            );
   668→        }
   669→        drop(result);
   670→        if let Ok(mut busy) = busy.lock() {
   671→            *busy = false;
   672→        }
   673→        // forwarder 在 tx 关闭后自然退出
   674→        let _ = forwarder.await;
   675→    });
   676→
   677→    Ok(())
   678→}

   701→/// 后台下载 JVM 运行时 pack，进度经 `runtime://progress` 上报（与下载游戏事件同构）。
   702→#[tauri::command]
   703→async fn download_runtime(java_major: u32, app: AppHandle, state: State<'_, AppState>) -> Result<(), String> {
   704→    if state
   705→        .runtime_busy
   706→        .swap(true, std::sync::atomic::Ordering::SeqCst)
   707→    {
   708→        return Err("已有运行时下载任务正在进行".to_string());
   709→    }
   710→    let busy = state.runtime_busy.clone();
   711→    let app = app.clone();
   712→
   713→    tauri::async_runtime::spawn(async move {
   714→        let (tx, mut rx) = tokio::sync::mpsc::channel::<DownloadProgress>(64);
   715→        let forwarder_app = app.clone();
   716→        let forwarder = tokio::spawn(async move {
   717→            while let Some(ev) = rx.recv().await {
   718→                let _ = forwarder_app.emit("runtime://progress", &ev);
   719→            }
   720→        });
   721→
   722→        let client = meta::http_client();
   723→        let result = runtime::download_and_install(&client, java_major, host_arch(), tx).await;
   724→        if let Err(e) = &result {
   725→            let _ = app.emit(
   726→                "runtime://progress",
   727→                &DownloadProgress {
   728→                    stage: "error".to_string(),
   729→                    current: 0,
   730→                    total: 0,
   731→                    percent: 0.0,
   732→                    file: String::new(),
   733→                    done: false,
   734→                    error: Some(e.to_string()),
   735→                },
   736→            );
   737→        }
   738→        drop(result);
   739→        busy.store(false, std::sync::atomic::Ordering::SeqCst);
   740→        let _ = forwarder.await;
   741→    });
   742→
   743→    Ok(())
   744→}

   790→/// 解析版本 JSON 并装配 LaunchPlan（版本未安装 / JSON 损坏时报错）。
   791→fn build_plan_for(version_id: &str) -> Result<(model::VersionJson, jvmops::LaunchPlan, u32), String> {
   792→    let text = std::fs::read_to_string(paths::version_json(version_id))
   793→        .map_err(|_| format!("版本 {version_id} 未安装（缺少版本 JSON），请先在「下载游戏」页完成下载"))?;
   794→    let version: model::VersionJson =
   795→        serde_json::from_str(&text).map_err(|e| format!("版本 JSON 解析失败：{e}"))?;
   796→    if !paths::version_jar(version_id).exists() {
   797→        return Err(format!("版本 {version_id} 的 client jar 缺失，请重新下载"));
   798→    }
   799→    let major = version
   800→        .java_version
   801→        .as_ref()
   802→        .map(|j| j.major_version)
   803→        .unwrap_or(17);
   804→
   805→    let ctx = instance::standard_rules_ctx();
   806→    let allowed: Vec<&model::Library> = version
   807→        .libraries
   808→        .iter()
   809→        .filter(|l| rules::evaluate(&l.rules, &ctx))
   810→        .collect();
   811→
   812→    let game_dir = paths::minecraft_dir();
   813→    let natives_root = game_dir.join("natives");
   814→    let plan = jvmops::build_launch_plan(&jvmops::PlanInputs {
   815→        ctx: &ctx,
   816→        version: &version,
   817→        allowed_libs: &allowed,
   818→        libraries_root: &paths::libraries_dir(),
   819→        natives_root: &natives_root,
   820→        assets_root: &paths::assets_dir(),
   821→        versions_root: &paths::versions_dir(),
   822→        instance_dir: &game_dir,
   823→        account: &auth::active_account()
   824→            .map_err(|e| e.to_string())?
   825→            .ok_or("尚未登录，请先在「账户」页添加账号")?,
   826→        memory_mb: 1024,
   827→        java_major: major,
   828→        java_path: None,
   829→        window: None,
   830→    })
   831→    .map_err(|e| format!("启动计划装配失败：{e}"))?;
   832→    Ok((version, plan, major))
   833→}

   835→/// 拉起 JVM 子进程并托管日志/退出（`launch://log` / `launch://exit` 事件）。
   836→///
   837→/// 前置：JVM pack 已安装（Android 内部目录，targetSdk 28 允许 exec）。
   838→/// smoke test 先跑 `java -version` 把 JRE 版本写进控制台，失败时给出明确报错。
   839→fn spawn_game(
   840→    version_id: String,
   841→    plan: jvmops::LaunchPlan,
   842→    major: u32,
   843→    game_pid: Arc<Mutex<Option<u32>>>,
   844→    app: AppHandle,
   845→) -> Result<(), String> {
   846→    use std::io::BufRead;
   847→    use std::process::{Command, Stdio};
   848→
   849→    let java = runtime::java_bin(major);
   850→    let jre_home = runtime::install_dir(major);
   851→    let game_dir = paths::minecraft_dir();
   852→
   853→    // ---- smoke test：JRE 可执行性 ----
   854→    let _ = app.emit(
   855→        "launch://log",
   856→        &LaunchLog {
   857→            source: "quartz".into(),
   858→            line: format!("[quartz] JRE {major}：{}", jre_home.display()),
   859→        },
   860→    );
   861→    match Command::new(&java).arg("-version").output() {
   862→        Ok(out) => {
   863→            let mut text = String::from_utf8_lossy(&out.stdout).into_owned();
   864→            text.push_str(&String::from_utf8_lossy(&out.stderr));
   865→            for line in text.lines().filter(|l| !l.trim().is_empty()).take(4) {
   866→                let _ = app.emit(
   867→                    "launch://log",
   868→                    &LaunchLog {
   869→                        source: "quartz".into(),
   870→                        line: format!("[quartz] {line}"),
   871→                    },
   872→                );
   873→            }
   874→        }
   875→        Err(e) => {
   876→            return Err(format!(
   877→                "无法执行 JVM（{e}）。JVM pack 需位于可执行目录：{}",
   878→                jre_home.display()
   879→            ))
   880→        }
   881→    }
   882→
   883→    // ---- 组装命令：jvm_args + main_class + game_args ----
   884→    let ld_library_path = format!(
   885→        "{}:{}:{}",
   886→        jre_home.join("lib").display(),
   887→        jre_home.join("lib/server").display(),
   888→        plan.natives_dir.display()
   889→    );
   890→    let mut cmd = Command::new(&java);
   891→    cmd.args(&plan.jvm_args)
   892→        .arg(&plan.main_class)
   893→        .args(&plan.game_args)
   894→        .current_dir(&plan.working_dir)
   895→        .env("HOME", &game_dir)
   896→        .env("JAVA_HOME", &jre_home)
   897→        .env("LD_LIBRARY_PATH", &ld_library_path)
   898→        .stdin(Stdio::null())
   899→        .stdout(Stdio::piped())
   900→        .stderr(Stdio::piped());
   901→
   902→    let mut child = cmd
   903→        .spawn()
   904→        .map_err(|e| format!("游戏进程启动失败：{e}"))?;
   905→    let pid = child.id();
   906→    if let Ok(mut slot) = game_pid.lock() {
   907→        *slot = Some(pid);
   908→    }
   909→    {
   910→        let app2 = app.clone();
   911→        let _ = app2.emit(
   912→            "launch://log",
   913→            &LaunchLog {
   914→                source: "quartz".into(),
   915→                line: format!("[quartz] 游戏进程已启动 (pid {pid})，工作目录：{}", plan.working_dir.display()),
   916→            },
   917→        );
   918→    }
   919→
   920→    // ---- 日志泵：stdout / stderr -> launch://log ----
   921→    if let Some(stdout) = child.stdout.take() {
   922→        let app2 = app.clone();
   923→        std::thread::spawn(move || {
   924→            for line in std::io::BufReader::new(stdout).lines().map_while(Result::ok) {
   925→                let _ = app2.emit("launch://log", &LaunchLog { source: "stdout".into(), line });
   926→            }
   927→        });
   928→    }
   929→    if let Some(stderr) = child.stderr.take() {
   930→        let app2 = app.clone();
   931→        std::thread::spawn(move || {
   932→            for line in std::io::BufReader::new(stderr).lines().map_while(Result::ok) {
   933→                let _ = app2.emit("launch://log", &LaunchLog { source: "stderr".into(), line });
   934→            }
   935→        });
   936→    }
   937→
   938→    // ---- 退出看护：wait -> launch://exit -> 清理 pid 槽位 ----
   939→    let game_pid_exit = game_pid;
   940→    std::thread::spawn(move || {
   941→        let code = child.wait().map(|s| s.code().unwrap_or(-1)).unwrap_or(-1);
   942→        if let Ok(mut slot) = game_pid_exit.lock() {
   943→            // stop_game 已 take 时这里已是 None，避免覆盖后续新进程
   944→            if slot.is_some() {
   945→                *slot = None;
   946→            }
   947→        }
   948→        let _ = app.emit(
   949→            "launch://exit",
   950→            &LaunchExit { code, version: version_id },
   951→        );
   952→    });
   953→    Ok(())
   954→}
   955→
   956→/// 真实启动：版本 JSON -> LaunchPlan -> JVM 子进程（M0 已接入）。
   957→/// JVM pack 缺失时返回 `RUNTIME_MISSING:<major>`，UI 自动触发运行时下载。
   958→#[tauri::command]
   959→fn launch_game(version_id: String, app: AppHandle, state: State<'_, AppState>) -> Result<(), String> {
   960→    if state.game_pid.lock().map_err(|_| "内部状态锁异常")?.is_some() {
   961→        return Err("已有游戏进程在运行，请先停止后再启动".to_string());
   962→    }
   963→    let (_version, plan, major) = build_plan_for(&version_id)?;
   964→    if !runtime::is_installed(major) {
   965→        return Err(format!("RUNTIME_MISSING:{major}"));
   966→    }
   967→    std::fs::create_dir_all(&plan.working_dir).ok();
   968→    std::fs::create_dir_all(&plan.natives_dir).ok();
   969→    spawn_game(version_id.clone(), plan, major, state.game_pid.clone(), app.clone())?;
   970→    Ok(())
   971→}

   973→/// 终止运行中的游戏进程（SIGKILL）。
   974→#[tauri::command]
   975→fn stop_game(state: State<'_, AppState>) -> Result<bool, String> {
   976→    let pid = state
   977→        .game_pid
   978→        .lock()
