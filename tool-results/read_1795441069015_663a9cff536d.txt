     1→//! Quartz 游戏显示桥核心（Rust 重构 Pojav C 桥 libpojavexec.so）。
     2→//!
     3→//! 契约来源：gladiolus 官方 `lwjgl-glfw-classes.jar` 字节码（javap 反编译核对）。
     4→//! 所有 JNI 导出符号严格使用 `Java_<类路径>_<方法>` 命名（JVM 按名绑定）。
     5→//!
     6→//! 与 Pojav C 版的关键差异（v0.1.5 修复 gladiolus 发布包 jar↔so 版本错位崩溃）：
     7→//! - 不做任何 RegisterNatives（旧 so 的 JNI_OnLoad 注册 jar 不存在的
     8→//!   nativeSetUseInputStackQueue -> RegisterNatives NoSuchMethodError -> 退出码 255）
     9→//! - 输入事件常驻队列模式（新 jar 的 glfwPollEvents 恒走 Start/Pump/StopPumping）
    10→//! - ART/OpenJDK 双 JVM 经显式 C ABI init 传递（不依赖 POJAV_ENVIRON 指针技巧）；
    11→//!   本库仅被游戏侧 OpenJDK dlopen，壳层（ART）经 dlsym 调 C ABI
    12→

    51→/// JNI 导出函数内 panic 会 abort 整个进程（游戏与壳层同进程）——统一吞掉并留痕。
    52→/// 宏体内整体 unsafe：这些导出函数本质全是 FFI 边界，unsafe 是默认语境。
    53→macro_rules! guard {
    54→    ($body:block) => {
    55→        let _ = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| unsafe {
    56→            $body
    57→        }));
    58→    };
    59→}

    78→unsafe fn jstring_to_rust(env: EnvPtr, s: jstring) -> Option<String> {
    79→    if s.is_null() {
    80→        return None;
    81→    }
    82→    let tbl = env_tbl(env);
    83→    let chars = (tbl.GetStringUTFChars.unwrap())(env, s, ptr::null_mut());
    84→    if chars.is_null() {
    85→        return None;
    86→    }
    87→    let text = CStr::from_ptr(chars).to_string_lossy().into_owned();
    88→    (tbl.ReleaseStringUTFChars.unwrap())(env, s, chars);
    89→    Some(text)
    90→}

    92→unsafe fn rust_to_jstring(env: EnvPtr, text: &str) -> Option<jstring> {
    93→    let cs = CString::new(text).ok()?;
    94→    let tbl = env_tbl(env);
    95→    let out = (tbl.NewStringUTF.unwrap())(env, cs.as_ptr());
    96→    if env_tbl(env).ExceptionCheck.unwrap()(env) != 0 {
    97→        env_tbl(env).ExceptionClear.unwrap()(env);
    98→        return None;
    99→    }
   100→    Some(out)
   101→}

   103→/// 线程附着的 OpenJDK（runtime）env：GetEnv -> EDETACHED 则 Attach。
   104→unsafe fn runtime_env() -> Option<EnvPtr> {
   105→    let s = state();
   106→    let vm = s.rt_vm.load(Ordering::Acquire);
   107→    if vm.is_null() {
   108→        return None;
   109→    }
   110→    let iface = vm_tbl(vm);
   111→    let mut env: EnvPtr = ptr::null_mut();
   112→    let rc = (iface.GetEnv.unwrap())(vm, &mut env as *mut EnvPtr as *mut *mut c_void, JNI_VERSION_1_4);
   113→    if rc == JNI_OK && !env.is_null() {
   114→        return Some(env);
   115→    }
   116→    if rc == jni_sys::JNI_EDETACHED {
   117→        let rc = (iface.AttachCurrentThread.unwrap())(
   118→            vm,
   119→            &mut env as *mut EnvPtr as *mut *mut c_void,
   120→            ptr::null_mut(),
   121→        );
   122→        if rc == JNI_OK && !env.is_null() {
   123→            return Some(env);
   124→        }
   125→    }
   126→    None
   127→}
   128→
   129→/// 线程附着的 ART（壳层）env。
   130→unsafe fn art_env() -> Option<EnvPtr> {
   131→    let s = state();
   132→    let vm = s.art_vm.load(Ordering::Acquire);
   133→    if vm.is_null() {
   134→        return None;
   135→    }
   136→    let iface = vm_tbl(vm);
   137→    let mut env: EnvPtr = ptr::null_mut();
   138→    let rc = (iface.GetEnv.unwrap())(vm, &mut env as *mut EnvPtr as *mut *mut c_void, JNI_VERSION_1_4);
   139→    if rc == JNI_OK && !env.is_null() {
   140→        return Some(env);
   141→    }
   142→    if rc == jni_sys::JNI_EDETACHED {
   143→        let rc = (iface.AttachCurrentThread.unwrap())(
   144→            vm,
   145→            &mut env as *mut EnvPtr as *mut *mut c_void,
   146→            ptr::null_mut(),
   147→        );
   148→        if rc == JNI_OK && !env.is_null() {
   149→            return Some(env);
   150→        }
   151→    }
   152→    None
   153→}

   371→    /// dlopen 渲染层 EGL 实现并解析符号（Pojav egl_loader.c 的 dlsym_EGL 语义）。
   372→    pub fn load_api() -> Option<EglApi> {
   373→        extern "C" {
   374→            fn dlopen(filename: *const c_char, flag: c_int) -> *mut c_void;
   375→            fn dlsym(handle: *mut c_void, symbol: *const c_char) -> *mut c_void;
   376→        }
   377→        const RTLD_LAZY: c_int = 1;
   378→        const RTLD_LOCAL: c_int = 0;
   379→
   380→        let configured = std::env::var("POJAVEXEC_EGL").unwrap_or_default();
   381→        let path = if configured.is_empty() { "libEGL.so".to_string() } else { configured };
   382→        let cpath = CString::new(path.clone()).ok()?;
   383→        let handle = unsafe { dlopen(cpath.as_ptr(), RTLD_LAZY | RTLD_LOCAL) };
   384→        if handle.is_null() {
   385→            log(&format!("EGL 库加载失败：{path}"));
   386→            return None;
   387→        }
   388→        let mut api = EglApi {
   389→            make_current: None,
   390→            destroy_surface: None,
   391→            destroy_context: None,
   392→            terminate: None,
   393→            get_display: None,
   394→            initialize: None,
   395→            choose_config: None,
   396→            get_config_attrib: None,
   397→            bind_api: None,
   398→            create_context: None,
   399→            create_window_surface: None,
   400→            create_pbuffer_surface: None,
   401→            swap_buffers: None,
   402→            swap_interval: None,
   403→            query_surface: None,
   404→            get_error: None,
   405→        };
   406→        macro_rules! get {
   407→            ($field:ident, $name:literal, $ty:ty) => {{
   408→                let cname = CString::new($name).unwrap();
   409→                let p = unsafe { dlsym(handle, cname.as_ptr()) };
   410→                if p.is_null() {
   411→                    log(&concat!("EGL 符号缺失：", $name));
   412→                    return None;
   413→                }
   414→                api.$field = Some(unsafe { std::mem::transmute::<*mut c_void, $ty>(p) });
   415→            }};
   416→        }
   417→        get!(make_current, "eglMakeCurrent", EglMakeCurrentFn);
   418→        get!(destroy_surface, "eglDestroySurface", EglDestroyFn);
   419→        get!(destroy_context, "eglDestroyContext", EglDestroyFn);
   420→        get!(terminate, "eglTerminate", EglTerminateFn);
   421→        get!(get_display, "eglGetDisplay", EglDisplayFn);
   422→        get!(initialize, "eglInitialize", EglInitializeFn);
   423→        get!(choose_config, "eglChooseConfig", EglChooseConfigFn);

   431→        get!(query_surface, "eglQuerySurface", EglQuerySurfaceFn);
   432→        get!(get_error, "eglGetError", EglGetErrorFn);
   433→        Some(api)
   434→    }

   461→// ---------------------------------------------------------------------------
   462→// GLFW Java 侧回调（internalChangeMonitorSize / internalWindowSizeChanged）
   463→// ---------------------------------------------------------------------------
   464→
   465→unsafe fn call_glfw_void(method_slot: &AtomicPtr<c_void>, label: &str, args: &[jvalue]) {
   466→    let Some(env) = (unsafe { runtime_env() }) else { return };
   467→    unsafe {
   468→        let cls = state().rt_glfw_class.load(Ordering::Acquire) as jclass;
   469→        let mid = method_slot.load(Ordering::Acquire) as jmethodID;
   470→        if cls.is_null() || mid.is_null() {
   471→            return;
   472→        }
   473→        let tbl = env_tbl(env);
   474→        let Some(call) = tbl.CallStaticVoidMethodA else { return };
   475→        if tbl.ExceptionCheck.unwrap()(env) != 0 {
   476→            tbl.ExceptionClear.unwrap()(env);
   477→        }
   478→        call(env, cls, mid, args.as_ptr());
   479→        if tbl.ExceptionCheck.unwrap()(env) != 0 {
   480→            tbl.ExceptionClear.unwrap()(env);
   481→            log(&format!("GLFW 回调异常已清除（{label}）"));
   482→        }
   483→    }
   484→}

   486→fn update_monitor_size_call() {
   487→    let s = state();
   488→    let w = s.saved_w.load(Ordering::Acquire);
   489→    let h = s.saved_h.load(Ordering::Acquire);
   490→    let args = [jvalue { i: w }, jvalue { i: h }];
   491→    unsafe {
   492→        call_glfw_void(&s.m_monitor_size, "internalChangeMonitorSize", &args);
   493→    }
   494→}

   499→        call_glfw_void(&state().m_window_size, "internalWindowSizeChanged", &args);
   500→    }
   501→}

   507→/// `int pojavInit(void)` —— glfwInit。
   508→#[no_mangle]
   509→pub extern "C" fn pojav_init() -> c_int {
   510→    let mut ok = false;
   511→    guard!({
   512→        let s = state();
   513→        let window = s.window.load(Ordering::Acquire);
   514→        let (mut w, mut h) = (0, 0);
   515→        if !window.is_null() {
   516→            ANativeWindow_acquire(window);
   517→            w = ANativeWindow_getWidth(window);
   518→            h = ANativeWindow_getHeight(window);
   519→            s.saved_w.store(w, Ordering::Release);
   520→            s.saved_h.store(h, Ordering::Release);
   521→            ANativeWindow_setBuffersGeometry(window, w, h, WINDOW_FORMAT_RGBX_8888);
   522→            s.upd_monitor.store(true, Ordering::Release);
   523→            s.monitor_consumed.store(false, Ordering::Release);
   524→        } else {
   525→            log("pojavInit：ANativeWindow 尚未就绪（等待壳层 Surface）");
   526→        }
   527→        let renderer_env = std::env::var("POJAV_RENDERER").unwrap_or_default();
   528→        if renderer_env.starts_with("vulkan_zink") {
   529→            s.renderer.store(RENDERER_VK_ZINK, Ordering::Release);
   530→            log("渲染桥：Zink 路径待 M2（Mesa pack 未接入）");
   531→            return;
   532→        }
   533→        s.renderer.store(RENDERER_GL4ES, Ordering::Release);
   534→        if let Some(api) = egl::load_api() {
   535→            let display = (api.get_display.unwrap())(egl::EGL_DEFAULT_DISPLAY);
   536→            if display != egl::EGL_NO_DISPLAY
   537→                && (api.initialize.unwrap())(display, ptr::null_mut(), ptr::null_mut()) == egl::EGL_TRUE
   538→            {
   539→                *EGL_STATE.lock().unwrap() = Some((api, display as usize));
   540→                log("EGL 桥就绪（渲染层实现库）");
   541→                ok = true;
   542→            } else {
   543→                let err = (api.get_error.unwrap())();
   544→                log(&format!("EGL 初始化失败：{err:#06x}"));
   545→            }
   546→        }
   547→        log(&format!("pojavInit：窗口 {w}x{h}"));
   548→    });
   549→    if ok { 1 } else { 1 } // 与 Pojav 一致恒返 1（失败细节已写日志，MC 会再报错）
   550→}

   552→/// `void* pojavCreateContext(void* share)` —— nglfwCreateContext。
   553→#[no_mangle]
   554→pub extern "C" fn pojav_create_context(share: *mut c_void) -> *mut c_void {
   555→    let mut result: *mut c_void = ptr::null_mut();
   556→    guard!({
   557→        let s = state();
   558→        let guard = EGL_STATE.lock().unwrap();
   559→        let Some((api, display)) = guard.as_ref() else {
   560→            log("pojavCreateContext：EGL 未初始化");
   561→            return;
   562→        };
   563→        let display = *display as egl::EGLDisplay;
   564→        let attributes: [i32; 15] = [
   565→            egl::EGL_BLUE_SIZE, 8,
   566→            egl::EGL_GREEN_SIZE, 8,
   567→            egl::EGL_RED_SIZE, 8,
   568→            egl::EGL_ALPHA_SIZE, 8,
   569→            egl::EGL_DEPTH_SIZE, 24,
   570→            egl::EGL_SURFACE_TYPE, egl::EGL_WINDOW_BIT | egl::EGL_PBUFFER_BIT,
   571→            egl::EGL_RENDERABLE_TYPE, egl::EGL_OPENGL_ES2_BIT,
   572→            egl::EGL_NONE,
   573→        ];
   574→        let mut num_configs: i32 = 0;
   575→        if (api.choose_config.unwrap())(display, attributes.as_ptr(), ptr::null_mut(), 0, &mut num_configs)
   576→            != egl::EGL_TRUE
   577→            || num_configs == 0
   578→        {
   579→            let err = (api.get_error.unwrap())();
   580→            log(&format!("eglChooseConfig 失败：{err:#06x}（configs={num_configs}）"));
   581→            return;
   582→        }
   583→        let mut config: egl::EGLConfig = ptr::null_mut();
   584→        let mut got: i32 = 0;
   585→        (api.choose_config.unwrap())(display, attributes.as_ptr(), &mut config, 1, &mut got);
   586→        let mut format: i32 = 0;
   587→        (api.get_config_attrib.unwrap())(display, config, egl::EGL_NATIVE_VISUAL_ID, &mut format);
   588→
   589→        let renderer_env = std::env::var("POJAV_RENDERER").unwrap_or_default();
   590→        let bind = if renderer_env.contains("desktopgl") {
   591→            (api.bind_api.unwrap())(egl::EGL_OPENGL_API)
   592→        } else {
   593→            (api.bind_api.unwrap())(egl::EGL_OPENGL_ES_API)
   594→        };
   595→        if bind != egl::EGL_TRUE {
   596→            log("eglBindAPI 失败");
   597→        }
   598→
   599→        // GLES 主版本：LIBGL_ES 显式指定（mobileglues 默认 3，gl4es 默认 2）
   600→        let es = std::env::var("LIBGL_ES")
   601→            .ok()
   602→            .and_then(|v| v.trim().parse::<i32>().ok())
   603→            .filter(|v| (1..=3).contains(v))
   604→            .unwrap_or(if renderer_env.contains("mobileglues") { 3 } else { 2 });
   605→        let ctx_attrs = [egl::EGL_CONTEXT_CLIENT_VERSION, es, egl::EGL_NONE];
   606→        let share_ctx = if share.is_null() {
   607→            egl::EGL_NO_CONTEXT
   608→        } else {
   609→            let windows = s.windows.lock().unwrap();
   610→            match windows.get(&(share as usize)) {
   611→                Some(w) => w.context.get(),
   612→                None => egl::EGL_NO_CONTEXT,
   613→            }
   614→        };
   615→        let context = (api.create_context.unwrap())(display, config, share_ctx, ctx_attrs.as_ptr());
   616→        if context == egl::EGL_NO_CONTEXT {
   617→            let err = (api.get_error.unwrap())();
   618→            log(&format!("eglCreateContext 失败：{err:#06x}（client_version={es}）"));
   619→            return;
   620→        }
   621→        let bundle = Arc::new(RenderWindow {
   622→            state: std::cell::Cell::new(egl::STATE_ALIVE),
   623→            config: std::cell::Cell::new(config),
   624→            context: std::cell::Cell::new(context),
   625→            surface: std::cell::Cell::new(ptr::null_mut()),
   626→            format: std::cell::Cell::new(format),
   627→            native_surface: std::cell::Cell::new(ptr::null_mut()),
   628→            new_native_surface: std::cell::Cell::new(ptr::null_mut()),
   629→        });
   630→        let handle = Arc::as_ptr(&bundle) as usize;
   631→        s.windows.lock().unwrap().insert(handle, bundle);
   632→        log(&format!("pojavCreateContext：窗口 {handle:#x}（GLES{es}）"));
   633→        result = handle as *mut c_void;
   634→    });
   635→    result
   636→}

   649→/// Pojav gl_bridge.c gl_swap_surface：换绑新 Surface 或 1x1 pbuffer 兜底。
   650→unsafe fn gl_swap_surface(bundle: &Arc<RenderWindow>, display: egl::EGLDisplay, api: &egl::EglApi) {
   651→    let old_native = bundle.native_surface.get();
   652→    if !old_native.is_null() {
   653→        ANativeWindow_release(old_native);
   654→        bundle.native_surface.set(ptr::null_mut());
   655→    }
   656→    if !bundle.surface.get().is_null() {
   657→        (api.destroy_surface.unwrap())(display, bundle.surface.get());
   658→        bundle.surface.set(ptr::null_mut());
   659→    }
   660→    let new_native = bundle.new_native_surface.get();
   661→    if !new_native.is_null() {
   662→        bundle.new_native_surface.set(ptr::null_mut());
   663→        bundle.native_surface.set(new_native);
   664→        ANativeWindow_acquire(new_native);
   665→        ANativeWindow_setBuffersGeometry(new_native, 0, 0, bundle.format.get());
   666→        let surface =
   667→            (api.create_window_surface.unwrap())(display, bundle.config.get(), new_native as *mut c_void, ptr::null());
   668→        bundle.surface.set(surface);
   669→        if surface.is_null() {
   670→            let err = (api.get_error.unwrap())();
   671→            log(&format!("eglCreateWindowSurface 失败：{err:#06x}"));
   672→        }
   673→    } else {
   674→        log("无新 Surface，切 1x1 pbuffer 兜底");
   675→        let pbuf_attrs = [egl::EGL_WIDTH, 1, egl::EGL_HEIGHT, 1, egl::EGL_NONE];
   676→        bundle
   677→            .surface
   678→            .set((api.create_pbuffer_surface.unwrap())(display, bundle.config.get(), pbuf_attrs.as_ptr()));
   679→    }
   680→}

   682→/// `void pojavMakeCurrent(void* window)`。
   683→#[no_mangle]
   684→pub extern "C" fn pojav_make_current(window: *mut c_void) {
   685→    guard!({
   686→        let s = state();
   687→        let handle = window as usize;
   688→        if window.is_null() {
   689→            let guard = EGL_STATE.lock().unwrap();
   690→            if let Some((api, display)) = guard.as_ref() {
   691→                let display = *display as egl::EGLDisplay;
   692→                (api.make_current.unwrap())(display, egl::EGL_NO_SURFACE, egl::EGL_NO_SURFACE, egl::EGL_NO_CONTEXT);
   693→            }
   694→            CURRENT_WINDOW.with(|c| c.set(0));
   695→            return;
   696→        }
   697→        let Some(bundle) = s.windows.lock().unwrap().get(&handle).cloned() else { return };
   698→        let guard = EGL_STATE.lock().unwrap();
   699→        let Some((api, display)) = guard.as_ref() else { return };
   700→        let display = *display as egl::EGLDisplay;
   701→        let mut first_main = false;
   702→        {
   703→            let mut main_slot = s.main_window.lock().unwrap();
   704→            if main_slot.is_none() {
   705→                *main_slot = Some(handle);
   706→                first_main = true;
   707→                bundle.new_native_surface.set(s.window.load(Ordering::Acquire));
   708→                log(&format!("主窗口 bundle = {handle:#x}"));
   709→            }
   710→        }
   711→        if bundle.surface.get().is_null() {
   712→            gl_swap_surface(&bundle, display, api);
   713→        }
   714→        if (api.make_current.unwrap())(display, bundle.surface.get(), bundle.surface.get(), bundle.context.get())
   715→            == egl::EGL_TRUE
   716→        {
   717→            CURRENT_WINDOW.with(|c| c.set(handle));
   718→        } else {
   719→            let err = (api.get_error.unwrap())();
   720→            log(&format!("eglMakeCurrent 失败：{err:#06x}（首帧={first_main}）"));
   721→            if first_main {
   722→                bundle.new_native_surface.set(ptr::null_mut());
   723→                gl_swap_surface(&bundle, display, api);
   724→                s.main_window.lock().unwrap().take();
   725→            }
   726→        }
   727→    });
   728→}

   730→/// `void pojavSwapBuffers(void* window)`。
   731→#[no_mangle]
   732→pub extern "C" fn pojav_swap_buffers(_window: *mut c_void) {
   733→    guard!({
   734→        let guard = EGL_STATE.lock().unwrap();
   735→        let Some((api, display)) = guard.as_ref() else { return };
   736→        let display = *display as egl::EGLDisplay;
   737→        let handle = CURRENT_WINDOW.with(|c| c.get());
   738→        if handle == 0 {
   739→            return;
   740→        }
   741→        let Some(bundle) = state().windows.lock().unwrap().get(&handle).cloned() else { return };
   742→        if bundle.state.get() == egl::STATE_NEW_WINDOW {
   743→            (api.make_current.unwrap())(display, egl::EGL_NO_SURFACE, egl::EGL_NO_SURFACE, egl::EGL_NO_CONTEXT);
   744→            gl_swap_surface(&bundle, display, api);
   745→            (api.make_current.unwrap())(display, bundle.surface.get(), bundle.surface.get(), bundle.context.get());
   746→            bundle.state.set(egl::STATE_ALIVE);
   747→        }
   748→        if !bundle.surface.get().is_null()
   749→            && (api.swap_buffers.unwrap())(display, bundle.surface.get()) == 0
   750→            && (api.get_error.unwrap())() == egl::EGL_BAD_SURFACE
   751→        {
   752→            log("EGL surface 失效（surfaceChanged），等待新窗口");
   753→            (api.make_current.unwrap())(display, egl::EGL_NO_SURFACE, egl::EGL_NO_SURFACE, egl::EGL_NO_CONTEXT);
   754→            bundle.new_native_surface.set(ptr::null_mut());
   755→            gl_swap_surface(&bundle, display, api);
   756→            (api.make_current.unwrap())(display, bundle.surface.get(), bundle.surface.get(), bundle.context.get());
   757→        }
   758→    });
   759→}

   781→/// `void pojavTerminate(void)`。
   782→#[no_mangle]
   783→pub extern "C" fn pojav_terminate() {
   784→    guard!({
   785→        let s = state();
   786→        s.input_ready.store(false, Ordering::Release);
   787→        let guard = EGL_STATE.lock().unwrap();
   788→        if let Some((api, display)) = guard.as_ref() {
   789→            let display = *display as egl::EGLDisplay;
   790→            (api.make_current.unwrap())(display, egl::EGL_NO_SURFACE, egl::EGL_NO_SURFACE, egl::EGL_NO_CONTEXT);
   791→            let mut windows = s.windows.lock().unwrap();
   792→            for (_, bundle) in windows.iter() {
   793→                if !bundle.surface.get().is_null() {
   794→                    (api.destroy_surface.unwrap())(display, bundle.surface.get());
   795→                }
   796→                (api.destroy_context.unwrap())(display, bundle.context.get());
   797→                let native = bundle.native_surface.get();
   798→                if !native.is_null() {
   799→                    ANativeWindow_release(native);
   800→                }
   801→            }
   802→            windows.clear();
   803→            s.main_window.lock().unwrap().take();
   804→            (api.terminate.unwrap())(display);
   805→        }
   806→        drop(guard);
   807→        *EGL_STATE.lock().unwrap() = None;
   808→        log("pojavTerminate 完成");
   809→    });
   810→}

   812→/// `void pojavStartPumping(void)`。
   813→#[no_mangle]
   814→pub extern "C" fn pojav_start_pumping() {
   815→    let s = state();
   816→    {
   817→        let cursor = *s.cursor.lock().unwrap();
   818→        let mut last = s.cursor_last.lock().unwrap();
   819→        if *last != cursor {
   820→            *last = cursor;
   821→            s.upd_mouse.store(true, Ordering::Release);
   822→        }
   823→    }
   824→    if s.upd_monitor.load(Ordering::Acquire) {
   825→        update_monitor_size_call();
   826→        s.monitor_consumed.store(true, Ordering::Release);
   827→    }
   828→    let mut queue = s.queue.lock().unwrap();
   829→    let mut pump = s.pump_buf.lock().unwrap();
   830→    pump.clear();
   831→    std::mem::swap(&mut *queue, &mut *pump);
   832→}

   834→/// `void pojavPumpEvents(void* window)` —— 对每个窗口派发本帧事件。
   835→#[no_mangle]
   836→pub extern "C" fn pojav_pump_events(window: *mut c_void) {
   837→    guard!({
   838→        let s = state();
   839→        if s.upd_mouse.load(Ordering::Acquire) {
   840→            let cb = s.cb_cursor_pos.load(Ordering::Acquire);
   841→            if cb != 0 {
   842→                let cursor = *s.cursor.lock().unwrap();
   843→                let f: unsafe extern "system" fn(*mut c_void, f64, f64) = std::mem::transmute(cb);
   844→                unsafe { f(window, cursor.0, cursor.1) };
   845→            }
   846→        }
   847→        if s.upd_monitor.load(Ordering::Acquire) {
   848→            update_window_size_call(window as usize);
   849→        }
   850→        let events: Vec<Event> = std::mem::take(&mut *s.pump_buf.lock().unwrap());
   851→        for ev in events {
   852→            match ev.typ {
   853→                EVENT_TYPE_CHAR => {
   854→                    let cb = s.cb_char.load(Ordering::Acquire);
   855→                    if cb != 0 {
   856→                        let f: unsafe extern "system" fn(*mut c_void, u32) = std::mem::transmute(cb);
   857→                        unsafe { f(window, ev.i1 as u32) };
   858→                    }
   859→                }
   860→                EVENT_TYPE_CHAR_MODS => {
   861→                    let cb = s.cb_char_mods.load(Ordering::Acquire);
   862→                    if cb != 0 {
   863→                        let f: unsafe extern "system" fn(*mut c_void, u32, i32) = std::mem::transmute(cb);
   864→                        unsafe { f(window, ev.i1 as u32, ev.i2) };
   865→                    }
   866→                }
   867→                EVENT_TYPE_CURSOR_ENTER => {
   868→                    let cb = s.cb_cursor_enter.load(Ordering::Acquire);
   869→                    if cb != 0 {
   870→                        let f: unsafe extern "system" fn(*mut c_void, i32) = std::mem::transmute(cb);
   871→                        unsafe { f(window, ev.i1) };
   872→                    }
   873→                }
   874→                EVENT_TYPE_KEY => {
   875→                    let cb = s.cb_key.load(Ordering::Acquire);
   876→                    if cb != 0 {
   877→                        let f: unsafe extern "system" fn(*mut c_void, i32, i32, i32, i32) =
   878→                            std::mem::transmute(cb);
   879→                        unsafe { f(window, ev.i1, ev.i2, ev.i3, ev.i4) };
   880→                    }
   881→                }
   882→                EVENT_TYPE_MOUSE_BUTTON => {
   883→                    let cb = s.cb_mouse.load(Ordering::Acquire);
   884→                    if cb != 0 {
   885→                        let f: unsafe extern "system" fn(*mut c_void, i32, i32, i32) =
   886→                            std::mem::transmute(cb);
   887→                        unsafe { f(window, ev.i1, ev.i2, ev.i3) };
   888→                    }
   889→                }
   890→                EVENT_TYPE_SCROLL => {
   891→                    let cb = s.cb_scroll.load(Ordering::Acquire);
   892→                    if cb != 0 {
   893→                        let f: unsafe extern "system" fn(*mut c_void, f64, f64) =
   894→                            std::mem::transmute(cb);
   895→                        unsafe { f(window, ev.dx, ev.dy) };
   896→                    }
   897→                }
   898→                _ => {}
   899→            }
   900→        }
   901→    });
   902→}

   904→/// `void pojavStopPumping(void)`。
   905→#[no_mangle]
   906→pub extern "C" fn pojav_stop_pumping() {
   907→    let s = state();
   908→    s.pump_buf.lock().unwrap().clear();
   909→    s.upd_mouse.store(false, Ordering::Release);
   910→    if s.upd_monitor.load(Ordering::Acquire) && s.monitor_consumed.load(Ordering::Acquire) {
   911→        s.upd_monitor.store(false, Ordering::Release);
   912→        s.monitor_consumed.store(false, Ordering::Release);
   913→    }
   914→}
   915→
   916→// ---------------------------------------------------------------------------
   917→// 输入派发（Pojav critical_send_* 语义，队列模式恒开）
   918→// ---------------------------------------------------------------------------
   919→
   920→fn dispatch_event(ev: Event) {
   921→    let s = state();
   922→    match ev.typ {
   923→        EVENT_TYPE_CURSOR_POS => {
   924→            // Pojav 语义：isCursorEntered 联动（无 enter 回调时 grab 下自行置位）
   925→            if !s.cursor_entered.load(Ordering::Acquire) {
   926→                if s.cb_cursor_enter.load(Ordering::Acquire) != 0 {
   927→                    s.cursor_entered.store(true, Ordering::Release);
   928→                    s.queue
   929→                        .lock()
   930→                        .unwrap()
   931→                        .push(Event::ints(EVENT_TYPE_CURSOR_ENTER, 1, 0, 0, 0));
   932→                } else if s.grabbing.load(Ordering::Acquire) {
   933→                    s.cursor_entered.store(true, Ordering::Release);
   934→                }
   935→            }
   936→            *s.cursor.lock().unwrap() = (ev.dx, ev.dy);
   937→        }
   938→        EVENT_TYPE_KEY => {
   939→            let buf = s.key_buf.load(Ordering::Acquire);
   940→            if !buf.is_null() {
   941→                let key = ev.i1 as i64;
   942→                if key >= 32 {
   943→                    let idx = (key - 32) as usize;
   944→                    if idx < KEY_BUF_SIZE {
   945→                        unsafe { *buf.add(idx) = ev.i3 as u8 };
   946→                    }
   947→                }
   948→            }
   949→            s.queue.lock().unwrap().push(ev);
   950→        }
   951→        EVENT_TYPE_MOUSE_BUTTON => {
   952→            let buf = s.mouse_buf.load(Ordering::Acquire);
   953→            if !buf.is_null() {
   954→                let btn = ev.i1.clamp(0, MOUSE_BUF_SIZE as i32 - 1) as usize;
   955→                unsafe { *buf.add(btn) = ev.i2 as u8 };
   956→            }
   957→            s.queue.lock().unwrap().push(ev);
   958→        }
   959→        EVENT_TYPE_WINDOW_SIZE | EVENT_TYPE_FRAMEBUFFER_SIZE => {
   960→            s.saved_w.store(ev.i1, Ordering::Release);
   961→            s.saved_h.store(ev.i2, Ordering::Release);
   962→            s.upd_monitor.store(true, Ordering::Release);
   963→            s.monitor_consumed.store(false, Ordering::Release);
   964→        }
   965→        _ => {
   966→            s.queue.lock().unwrap().push(ev);
   967→        }
   968→    }
   969→}

   971→/// 解析 nativeSendData 的字符串载荷（壳层 Kotlin 与本桥的约定协议）：
   972→/// 空格分隔 token，整型/浮点按事件类型解析。
   973→fn parse_send_data(typ: i32, data: &str) -> Option<Event> {
   974→    let mut it = data.split_whitespace();
   975→    let mut next_i = |it: &mut std::str::SplitWhitespace| it.next().and_then(|t| t.parse::<i64>().ok());
   976→    let mut next_f = |it: &mut std::str::SplitWhitespace| it.next().and_then(|t| t.parse::<f64>().ok());
   977→    match typ {
   978→        EVENT_TYPE_CHAR => Some(Event::ints(EVENT_TYPE_CHAR, next_i(&mut it)? as i32, 0, 0, 0)),
   979→        EVENT_TYPE_CHAR_MODS => Some(Event::ints(
   980→            EVENT_TYPE_CHAR_MODS,
   981→            next_i(&mut it)? as i32,
   982→            next_i(&mut it)? as i32,
   983→            0,
   984→            0,
   985→        )),
   986→        EVENT_TYPE_CURSOR_ENTER => Some(Event::ints(EVENT_TYPE_CURSOR_ENTER, next_i(&mut it)? as i32, 0, 0, 0)),
   987→        EVENT_TYPE_CURSOR_POS => {
   988→            let dx = next_f(&mut it)?;
   989→            let dy = next_f(&mut it)?;
   990→            Some(Event { typ: EVENT_TYPE_CURSOR_POS, i1: 0, i2: 0, i3: 0, i4: 0, dx, dy })
   991→        }
   992→        EVENT_TYPE_KEY => Some(Event::ints(
   993→            EVENT_TYPE_KEY,
   994→            next_i(&mut it)? as i32,
   995→            next_i(&mut it)? as i32,
   996→            next_i(&mut it)? as i32,
   997→            next_i(&mut it)? as i32,
   998→        )),
   999→        EVENT_TYPE_MOUSE_BUTTON => Some(Event::ints(
  1000→            EVENT_TYPE_MOUSE_BUTTON,
  1001→            next_i(&mut it)? as i32,
  1002→            next_i(&mut it)? as i32,
  1003→            next_i(&mut it)? as i32,
  1004→            0,
  1005→        )),
  1006→        EVENT_TYPE_SCROLL => {
  1007→            let dx = next_f(&mut it)?;
  1008→            let dy = next_f(&mut it)?;
  1009→            Some(Event { typ: EVENT_TYPE_SCROLL, i1: 0, i2: 0, i3: 0, i4: 0, dx, dy })
  1010→        }
  1011→        EVENT_TYPE_WINDOW_SIZE | EVENT_TYPE_FRAMEBUFFER_SIZE => Some(Event::ints(
  1012→            typ,
  1013→            next_i(&mut it)? as i32,
  1014→            next_i(&mut it)? as i32,
  1015→            0,
  1016→            0,
  1017→        )),
  1018→        _ => None,
  1019→    }
  1020→}

  1053→/// 剪贴板：经 ART 侧 QuartzBridge.accessAndroidClipboard(I,String)String 转发。
  1054→#[no_mangle]
  1055→pub unsafe extern "system" fn Java_org_lwjgl_glfw_CallbackBridge_nativeClipboard(
  1056→    env: *mut JNIEnv,
  1057→    _class: jclass,
  1058→    action: jint,
  1059→    src: jbyteArray,
  1060→) -> jstring {
  1061→    let mut result: jstring = ptr::null_mut();

  1064→        let Some(art) = (unsafe { art_env() }) else {

  1084→                        if let Some(js) = rust_to_jstring(art, &text) {
  1085→                            payload = js;
  1086→                        }
  1087→                    }
  1088→                    (tbl.ReleaseByteArrayElements.unwrap())(env, src, elems, 0);
  1089→                }
  1090→            }
  1091→            let args = [jvalue { l: payload as jobject }];
  1092→            let out = (atbl.CallStaticObjectMethodA.unwrap())(art, art_cls, mid, args.as_ptr());
  1093→            if atbl.ExceptionCheck.unwrap()(art) != 0 {
  1094→                atbl.ExceptionClear.unwrap()(art);
  1095→                return;
  1096→            }
  1097→            // ART 返回 String -> runtime env jstring
  1098→            if let Some(text) = (unsafe { jstring_to_rust(art, out) }) {
  1099→                if let Some(js) = (unsafe { rust_to_jstring(env, &text) }) {
  1100→                    result = js;
  1101→                }
  1102→            }
  1103→        }
  1104→    });

  1375→                if let Some(bundle) = s.windows.lock().unwrap().get(&handle).cloned() {
  1376→                    bundle.new_native_surface.set(window);
  1377→                    bundle.state.set(egl::STATE_NEW_WINDOW);
  1378→                }
  1379→            }
  1380→            if !old.is_null() && old != window {
  1381→                ANativeWindow_release(old);
  1382→            }
  1383→            rc = 1;
  1384→        }
  1385→    });
  1386→    rc
  1387→}
