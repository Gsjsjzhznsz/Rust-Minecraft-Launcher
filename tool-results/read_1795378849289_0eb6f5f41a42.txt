    66→function bytes(n: number): string {
    67→  if (!n) return "0 B";
    68→  const units = ["B", "KB", "MB", "GB"];
    69→  let v = n;
    70→  let i = 0;
    71→  while (v >= 1024 && i < units.length - 1) {
    72→    v /= 1024;
    73→    i += 1;
    74→  }
    75→  return `${v.toFixed(i === 0 ? 0 : 1)} ${units[i]}`;
    76→}

   121→function avatarStyle(uuid: string): string {
   122→  let h = 0;
   123→  for (const c of uuid) h = (h * 31 + c.charCodeAt(0)) >>> 0;
   124→  const hue = h % 360;
   125→  return `background: linear-gradient(135deg, hsl(${hue} 70% 62%), hsl(${(hue + 50) % 360} 70% 45%))`;
   126→}

   175→function fillHomeSelect() {
   176→  const sel = el<HTMLSelectElement>("version-select");
   177→  const prev = storage.selected;
   178→  sel.innerHTML = "";
   179→  if (!versions || versions.length === 0) {
   180→    const opt = document.createElement("option");
   181→    opt.value = "";
   182→    opt.textContent = installed.size > 0 ? "官方清单未获取，可先用已安装版本" : "版本清单未获取，请前往「下载游戏」页";
   183→    sel.appendChild(opt);
   184→    for (const v of installed) {
   185→      const o = document.createElement("option");
   186→      o.value = v;
   187→      o.textContent = `${v}（已安装）`;
   188→      sel.appendChild(o);
   189→    }
   190→    sel.disabled = false;
   191→    updateInstallBadge();
   192→    return;
   193→  }
   194→  for (const kind of KIND_ORDER) {
   195→    const list = versions.filter((v) => v.type === kind);
   196→    if (list.length === 0) continue;
   197→    const group = document.createElement("optgroup");
   198→    group.label = KIND_LABELS[kind];
   199→    for (const v of list) {
   200→      const opt = document.createElement("option");
   201→      opt.value = v.id;
   202→      opt.textContent = v.id;
   203→      group.appendChild(opt);
   204→    }
   205→    sel.appendChild(group);
   206→  }
   207→  sel.disabled = false;
   208→  restoreSelected(sel, prev);
   209→  updateInstallBadge();
   210→}

   212→function restoreSelected(sel: HTMLSelectElement, prev: string) {
   213→  if (prev && versions?.some((v) => v.id === prev)) {
   214→    sel.value = prev;
   215→  } else if (prev && installed.has(prev)) {
   216→    sel.value = prev;
   217→  } else {
   218→    sel.value = versions?.[0]?.id ?? "";
   219→  }
   220→  storage.selected = sel.value;
   221→}

   254→function renderVersionList() {
   255→  const wrap = el<HTMLDivElement>("version-list");
   256→  wrap.innerHTML = "";
   257→  const note = document.createElement("div");
   258→  note.className = "list-note";
   259→  note.id = "version-list-note";
   260→  wrap.appendChild(note);
   261→
   262→  if (!versions) {
   263→    note.textContent = el<HTMLDivElement>("version-list-note").textContent || "版本清单未加载";
   264→    return;
   265→  }
   266→
   267→  const text = searchText.trim().toLowerCase();
   268→  const filtered = versions.filter(
   269→    (v) => (kindFilter === "all" || v.type === kindFilter) && (!text || v.id.toLowerCase().includes(text)),
   270→  );
   271→  if (filtered.length === 0) {
   272→    note.textContent = "没有匹配的版本";
   273→    hide("load-more-btn");
   274→    return;
   275→  }
   276→
   277→  shownCount = Math.min(shownCount, filtered.length);
   278→  for (const v of filtered.slice(0, shownCount)) {
   279→    wrap.appendChild(versionRow(v));
   280→  }
   281→
   282→  const more = el<HTMLButtonElement>("load-more-btn");
   283→  if (shownCount < filtered.length) {
   284→    more.textContent = `加载更多（${shownCount} / ${filtered.length}）`;
   285→    show("load-more-btn");
   286→  } else {
   287→    hide("load-more-btn");
   288→  }
   289→}

   324→let listening = false;
   325→async function ensureProgressListener() {
   326→  if (listening) return;
   327→  listening = true;
   328→  await listen<DownloadProgress>("download://progress", (event) => onProgress(event.payload));
   329→  await listen<DownloadProgress>("runtime://progress", (event) => onRuntimeProgress(event.payload));
   330→  await listen<DownloadProgress>("renderer://progress", (event) => onRendererProgress(event.payload));
   331→  await listen<DownloadProgress>("lwjgl://progress", (event) => onLwjglProgress(event.payload));
   332→  await listen<LaunchLog>("launch://log", (event) => appendLog(event.payload.source, event.payload.line));
   333→  await listen<LaunchExit>("launch://exit", (event) => {
   334→    hide("stop-btn");
   335→    if (event.payload.code === 0) {
   336→      setLaunchBadge("已退出（0）", "dim");
   337→      toast("游戏已正常退出", "ok");
   338→    } else {
   339→      setLaunchBadge(`已退出（${event.payload.code}）`, "err");
   340→      toast(`游戏已退出（代码 ${event.payload.code}），详情见控制台`, "err");
   341→    }
   342→    appendLog("quartz", `[quartz] 进程退出，代码 ${event.payload.code}`);
   343→  });
   344→  await listen<MigrateMsg>("quartz://migrate", (event) => {
   345→    const p = event.payload;
   346→    if (p.error) {
   347→      toast(`存储迁移失败：${p.error}`, "err");
   348→      appendLog("quartz", `[quartz] 存储迁移失败：${p.error}`);
   349→      return;
   350→    }
   351→    if (p.done) {
   352→      toast("旧游戏数据已迁移到标准目录", "ok");
   353→      appendLog("quartz", "[quartz] 存储迁移完成");
   354→      refreshInstalled();
   355→      loadEnvAndMeta();
   356→    }
   357→  });
   358→}

   360→function progressEls(prefix: string) {
   361→  return {
   362→    fill: el<HTMLDivElement>(`${prefix}-progress-fill`),
   363→    text: el<HTMLSpanElement>(`${prefix}-progress-text`),
   364→    stage: el<HTMLSpanElement>(`${prefix}-progress-stage`),
   365→    file: el<HTMLDivElement>(`${prefix}-progress-file`),
   366→    status: el<HTMLDivElement>(`${prefix}-progress-status`),
   367→  };
   368→}
   369→
   370→function onProgress(p: DownloadProgress) {
   371→  for (const prefix of ["home", "dl"]) {
   372→    const ui = progressEls(prefix);
   373→    if (p.stage === "error") {
   374→      alertIn(`${prefix}-progress-status`, "alert-error", `下载失败（可重试，已下载文件会自动跳过）：${p.error ?? "未知错误"}`);
   375→    } else if (p.stage === "done") {
   376→      ui.fill.style.width = "100%";
   377→      ui.text.textContent = "100%";
   378→      ui.stage.textContent = "完成";
   379→      ui.file.textContent = "";
   380→      alertIn(`${prefix}-progress-status`, "alert-ok", "下载完成！游戏文件已写入标准 .minecraft 目录。");
   381→    } else {
   382→      alertOut(`${prefix}-progress-status`);
   383→      const percent = Math.max(0, Math.min(100, p.percent || 0));
   384→      ui.fill.style.width = `${percent.toFixed(1)}%`;
   385→      ui.text.textContent = `${percent.toFixed(1)}%（${bytes(p.current)} / ${bytes(p.total)}）`;
   386→      ui.stage.textContent = STAGE_LABELS[p.stage] ?? p.stage;
   387→      ui.file.textContent = p.file ?? "";
   388→    }
   389→  }
   390→
   391→  if (p.stage === "done") {
   392→    busy = false;
   393→    refreshInstalled();
   394→    toast("下载完成", "ok");
   395→  } else if (p.stage === "error") {
   396→    busy = false;
   397→    toast("下载失败", "err");
   398→  }
   399→  setBusyUI(busy);
   400→}

   408→async function startDownload(versionId: string) {
   409→  if (!versionId) {
   410→    toast("请先选择版本", "err");
   411→    return;
   412→  }
   413→  if (busy) {
   414→    toast("已有下载任务进行中", "err");
   415→    return;
   416→  }
   417→  busy = true;
   418→  setBusyUI(true);
   419→  storage.selected = versionId;
   420→  el<HTMLSelectElement>("version-select").value = versionId;
   421→  updateInstallBadge();
   422→
   423→  for (const prefix of ["home", "dl"]) {
   424→    const ui = progressEls(prefix);
   425→    alertOut(`${prefix}-progress-status`);
   426→    ui.fill.style.width = "0%";
   427→    ui.text.textContent = "0%";
   428→    ui.stage.textContent = "准备中";
   429→    ui.file.textContent = "";
   430→  }
   431→  show("home-progress");
   432→  show("dl-progress");
   433→  await ensureProgressListener();
   434→  try {
   435→    await invoke("download_game", { versionId });
   436→  } catch (e) {
   437→    busy = false;
   438→    setBusyUI(false);
   439→    for (const prefix of ["home", "dl"]) {
   440→      alertIn(`${prefix}-progress-status`, "alert-error", `下载启动失败：${String(e)}`);
   441→    }
   442→    toast("下载启动失败", "err");
   443→  }
   444→}

   520→  }

   522→  delBtn.className = "btn btn-sm btn-danger";
   523→  delBtn.textContent = "删除";
   524→  delBtn.addEventListener("click", async () => {

   526→    if (delBtn.dataset.confirm !== "1") {
   527→      delBtn.dataset.confirm = "1";
   528→      delBtn.textContent = "确认删除？";

   534→    }

   536→      await invoke("remove_account", { uuid: a.uuid });

   539→    } catch (e) {
   540→      toast(`删除失败：${String(e)}`, "err");
   541→    }
   542→  });
   543→  actions.appendChild(delBtn);
   544→  card.appendChild(actions);
   545→  return card;
   546→}

   548→async function addAccount() {
   549→  const input = el<HTMLInputElement>("new-account-name");
   550→  const name = input.value.trim();
   551→  alertOut("account-error");
   552→  if (!name) {

   554→    return;
   555→  }
   556→  const btn = el<HTMLButtonElement>("add-account-btn");
   557→  btn.disabled = true;

   974→        toast(String(e2), "err");
   975→        setLaunchBadge("启动失败", "err");
   976→      }
   977→      return;
   978→    }
   979→    setLaunchBadge("启动失败", "err");
   980→    hide("stop-btn");
   981→    appendLog("quartz", `[quartz] 启动失败：${msg}`);
   982→    toast(msg, "err");

   986→async function launch() {
   987→  const versionId = el<HTMLSelectElement>("version-select").value;
   988→  if (!versionId) {
   989→    toast("请先选择版本", "err");
   990→    return;
   991→  }
   992→  if (!installed.has(versionId)) {
   993→    toast("该版本尚未下载，请先在「下载游戏」页获取", "err");
   994→    return;
   995→  }
   996→  try {
   997→    const pf = await invoke<Preflight>("launch_preflight", { versionId });
   998→    if (!pf.accountOk) {
   999→      toast("请先在「账户」页登录", "err");
  1000→      return;
  1001→    }
  1002→    if (!pf.versionInstalled) {
  1003→      toast("版本文件缺失，请重新下载", "err");
  1004→      return;
  1005→    }
  1006→    if (!pf.runtimeInstalled) {
  1007→      if (!pf.runtimeSpecAvailable) {
  1008→        toast(`该版本需要 Java ${pf.javaMajor}，运行时下载源暂未收录`, "err");
  1009→        return;
  1010→      }
  1011→      appendLog("quartz", `[quartz] 需要 Java ${pf.javaMajor} 运行时，开始自动下载…`);
  1012→      pendingLaunchVersion = versionId;
  1013→      showRtProgress();
  1014→      await invoke("download_runtime", { javaMajor: pf.javaMajor });
  1015→      return; // 完成事件回调里继续 doLaunch
  1016→    }
  1017→    await doLaunch(versionId);
  1018→  } catch (e) {
  1019→    toast(String(e), "err");
  1020→  }
  1021→}
