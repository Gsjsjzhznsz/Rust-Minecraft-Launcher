    66→function bytes(n: number): string {
    67→  if (!n) return "0 B";
    68→  const units = ["B", "KB", "MB", "GB"];
    69→  let v = n;
    70→  let i = 0;
    71→  while (v >= 1024 && i < units.length - 1) {
    72→    v /= 1024;
    73→    i += 1;
    74→  }
    75→  return `${v.toFixed(i === 0 ? 0 : 1)} ${units[i]}`;
    76→}

   121→function avatarStyle(uuid: string): string {
   122→  let h = 0;
   123→  for (const c of uuid) h = (h * 31 + c.charCodeAt(0)) >>> 0;
   124→  const hue = h % 360;
   125→  return `background: linear-gradient(135deg, hsl(${hue} 70% 62%), hsl(${(hue + 50) % 360} 70% 45%))`;
   126→}

   175→function fillHomeSelect() {
   176→  const sel = el<HTMLSelectElement>("version-select");
   177→  const prev = storage.selected;
   178→  sel.innerHTML = "";
   179→  if (!versions || versions.length === 0) {
   180→    const opt = document.createElement("option");
   181→    opt.value = "";
   182→    opt.textContent = installed.size > 0 ? "官方清单未获取，可先用已安装版本" : "版本清单未获取，请前往「下载游戏」页";
   183→    sel.appendChild(opt);
   184→    for (const v of installed) {
   185→      const o = document.createElement("option");
   186→      o.value = v;
   187→      o.textContent = `${v}（已安装）`;
   188→      sel.appendChild(o);
   189→    }
   190→    sel.disabled = false;
   191→    updateInstallBadge();
   192→    return;
   193→  }
   194→  for (const kind of KIND_ORDER) {
   195→    const list = versions.filter((v) => v.type === kind);
   196→    if (list.length === 0) continue;
   197→    const group = document.createElement("optgroup");
   198→    group.label = KIND_LABELS[kind];
   199→    for (const v of list) {
   200→      const opt = document.createElement("option");
   201→      opt.value = v.id;
   202→      opt.textContent = v.id;
   203→      group.appendChild(opt);
   204→    }
   205→    sel.appendChild(group);
   206→  }
   207→  sel.disabled = false;
   208→  restoreSelected(sel, prev);
   209→  updateInstallBadge();
   210→}

   212→function restoreSelected(sel: HTMLSelectElement, prev: string) {
   213→  if (prev && versions?.some((v) => v.id === prev)) {
   214→    sel.value = prev;
   215→  } else if (prev && installed.has(prev)) {
   216→    sel.value = prev;
   217→  } else {
   218→    sel.value = versions?.[0]?.id ?? "";
   219→  }
   220→  storage.selected = sel.value;
   221→}

   254→function renderVersionList() {
   255→  const wrap = el<HTMLDivElement>("version-list");
   256→  wrap.innerHTML = "";
   257→  const note = document.createElement("div");
   258→  note.className = "list-note";
   259→  note.id = "version-list-note";
   260→  wrap.appendChild(note);
   261→
   262→  if (!versions) {
   263→    note.textContent = el<HTMLDivElement>("version-list-note").textContent || "版本清单未加载";
   264→    return;
   265→  }
   266→
   267→  const text = searchText.trim().toLowerCase();
   268→  const filtered = versions.filter(
   269→    (v) => (kindFilter === "all" || v.type === kindFilter) && (!text || v.id.toLowerCase().includes(text)),
   270→  );
   271→  if (filtered.length === 0) {
   272→    note.textContent = "没有匹配的版本";
   273→    hide("load-more-btn");
   274→    return;
   275→  }
   276→
   277→  shownCount = Math.min(shownCount, filtered.length);
   278→  for (const v of filtered.slice(0, shownCount)) {
   279→    wrap.appendChild(versionRow(v));
   280→  }
   281→
   282→  const more = el<HTMLButtonElement>("load-more-btn");
   283→  if (shownCount < filtered.length) {
   284→    more.textContent = `加载更多（${shownCount} / ${filtered.length}）`;
   285→    show("load-more-btn");
   286→  } else {
   287→    hide("load-more-btn");
   288→  }
   289→}

   323→/* ---------- 下载 ---------- */
   324→let listening = false;
   325→async function ensureProgressListener() {
   326→  if (listening) return;
   327→  listening = true;
   328→  await listen<DownloadProgress>("download://progress", (event) => onProgress(event.payload));
   329→  await listen<DownloadProgress>("runtime://progress", (event) => onRuntimeProgress(event.payload));
   330→  await listen<LaunchLog>("launch://log", (event) => appendLog(event.payload.source, event.payload.line));
   331→  await listen<LaunchExit>("launch://exit", (event) => {
   332→    hide("stop-btn");
   333→    if (event.payload.code === 0) {
   334→      setLaunchBadge("已退出（0）", "dim");
   335→      toast("游戏已正常退出", "ok");
   336→    } else {
   337→      setLaunchBadge(`已退出（${event.payload.code}）`, "err");
   338→      toast(`游戏已退出（代码 ${event.payload.code}），详情见控制台`, "err");
   339→    }
   340→    appendLog("quartz", `[quartz] 进程退出，代码 ${event.payload.code}`);
   341→  });
   342→  await listen<MigrateMsg>("quartz://migrate", (event) => {
   343→    const p = event.payload;
   344→    if (p.error) {
   345→      toast(`存储迁移失败：${p.error}`, "err");
   346→      appendLog("quartz", `[quartz] 存储迁移失败：${p.error}`);
   347→      return;
   348→    }
   349→    if (p.done) {
   350→      toast("旧游戏数据已迁移到标准目录", "ok");
   351→      appendLog("quartz", "[quartz] 存储迁移完成");
   352→      refreshInstalled();
   353→      loadEnvAndMeta();
   354→    }
   355→  });
   356→}

   358→function progressEls(prefix: string) {
   359→  return {
   360→    fill: el<HTMLDivElement>(`${prefix}-progress-fill`),
   361→    text: el<HTMLSpanElement>(`${prefix}-progress-text`),
   362→    stage: el<HTMLSpanElement>(`${prefix}-progress-stage`),
   363→    file: el<HTMLDivElement>(`${prefix}-progress-file`),
   364→    status: el<HTMLDivElement>(`${prefix}-progress-status`),
   365→  };
   366→}
   367→
   368→function onProgress(p: DownloadProgress) {
   369→  for (const prefix of ["home", "dl"]) {
   370→    const ui = progressEls(prefix);
   371→    if (p.stage === "error") {
   372→      alertIn(`${prefix}-progress-status`, "alert-error", `下载失败（可重试，已下载文件会自动跳过）：${p.error ?? "未知错误"}`);
   373→    } else if (p.stage === "done") {
   374→      ui.fill.style.width = "100%";
   375→      ui.text.textContent = "100%";
   376→      ui.stage.textContent = "完成";
   377→      ui.file.textContent = "";
   378→      alertIn(`${prefix}-progress-status`, "alert-ok", "下载完成！游戏文件已写入标准 .minecraft 目录。");
   379→    } else {
   380→      alertOut(`${prefix}-progress-status`);
   381→      const percent = Math.max(0, Math.min(100, p.percent || 0));
   382→      ui.fill.style.width = `${percent.toFixed(1)}%`;
   383→      ui.text.textContent = `${percent.toFixed(1)}%（${bytes(p.current)} / ${bytes(p.total)}）`;
   384→      ui.stage.textContent = STAGE_LABELS[p.stage] ?? p.stage;
   385→      ui.file.textContent = p.file ?? "";
   386→    }
   387→  }
   388→
   389→  if (p.stage === "done") {
   390→    busy = false;
   391→    refreshInstalled();
   392→    toast("下载完成", "ok");
   393→  } else if (p.stage === "error") {
   394→    busy = false;
   395→    toast("下载失败", "err");
   396→  }
   397→  setBusyUI(busy);
   398→}

   406→async function startDownload(versionId: string) {
   407→  if (!versionId) {
   408→    toast("请先选择版本", "err");
   409→    return;
   410→  }
   411→  if (busy) {
   412→    toast("已有下载任务进行中", "err");
   413→    return;
   414→  }
   415→  busy = true;
   416→  setBusyUI(true);
   417→  storage.selected = versionId;
   418→  el<HTMLSelectElement>("version-select").value = versionId;
   419→  updateInstallBadge();
   420→
   421→  for (const prefix of ["home", "dl"]) {
   422→    const ui = progressEls(prefix);
   423→    alertOut(`${prefix}-progress-status`);
   424→    ui.fill.style.width = "0%";
   425→    ui.text.textContent = "0%";
   426→    ui.stage.textContent = "准备中";
   427→    ui.file.textContent = "";
   428→  }
   429→  show("home-progress");
   430→  show("dl-progress");
   431→  await ensureProgressListener();
   432→  try {
   433→    await invoke("download_game", { versionId });
   434→  } catch (e) {
   435→    busy = false;
   436→    setBusyUI(false);
   437→    for (const prefix of ["home", "dl"]) {
   438→      alertIn(`${prefix}-progress-status`, "alert-error", `下载启动失败：${String(e)}`);
   439→    }
   440→    toast("下载启动失败", "err");
   441→  }
   442→}

   511→        await loadAccounts();
   512→        toast(`已切换到 ${a.username}`, "ok");

   514→        toast(`切换失败：${String(e)}`, "err");

   520→  delBtn.className = "btn btn-sm btn-danger";
   521→  delBtn.textContent = "删除";
   522→  delBtn.addEventListener("click", async () => {
   523→    // 两段式确认，避免误删
   524→    if (delBtn.dataset.confirm !== "1") {
   525→      delBtn.dataset.confirm = "1";
   526→      delBtn.textContent = "确认删除？";

   528→        delBtn.dataset.confirm = "";

   532→    }

   534→      await invoke("remove_account", { uuid: a.uuid });

   536→      await loadAccounts();
   537→    } catch (e) {
   538→      toast(`删除失败：${String(e)}`, "err");
   539→    }
   540→  });
   541→  actions.appendChild(delBtn);
   542→  card.appendChild(actions);
   543→  return card;
   544→}

   546→async function addAccount() {
   547→  const input = el<HTMLInputElement>("new-account-name");
   548→  const name = input.value.trim();
   549→  alertOut("account-error");
   550→  if (!name) {
   551→    alertIn("account-error", "alert-error", "请输入用户名");
   552→    return;
   553→  }
   554→  const btn = el<HTMLButtonElement>("add-account-btn");
   555→  btn.disabled = true;
   556→  try {
   557→    const acc = await invoke<Account>("offline_login", { name });
   558→    input.value = "";
   559→    toast(`已添加账号 ${acc.username}`, "ok");

   644→function appendLog(source: string, line: string) {
   645→  const box = el<HTMLDivElement>("console-box");
   646→  // 清掉占位行
   647→  const first = box.firstElementChild;
   648→  if (first && first.classList.contains("dim") && box.children.length === 1) first.remove();
   649→  const node = document.createElement("div");
   650→  node.className = consoleLineClass(source);
   651→  node.textContent = line;
   652→  box.appendChild(node);
   653→  while (box.children.length > CONSOLE_MAX_LINES) box.removeChild(box.firstElementChild!);
   654→  box.scrollTop = box.scrollHeight;
   655→}

   738→function showRtProgress() {
   739→  runtimeDownloading = true;
   740→  show("rt-progress");
   741→  const fill = el<HTMLDivElement>("rt-progress-fill");
   742→  fill.style.width = "0%";

   745→  setText("rt-progress-file", "");

   747→function onRuntimeProgress(p: DownloadProgress) {
   748→  if (!runtimeDownloading) return;
   749→  if (p.stage === "error") {
   750→    runtimeDownloading = false;
   751→    hide("rt-progress");
   752→    appendLog("quartz", `[quartz] 运行时下载失败：${p.error ?? "未知错误"}`);
   753→    toast("运行时下载失败，可重试", "err");
   754→    pendingLaunchVersion = null;
   755→    return;
   756→  }
   757→  if (p.stage === "done") {
   758→    runtimeDownloading = false;
   759→    hide("rt-progress");
   760→    appendLog("quartz", `[quartz] Java 运行时就绪：${p.file}`);
   761→    toast("Java 运行时就绪", "ok");
   762→    refreshRuntimeBadge();
   763→    const v = pendingLaunchVersion;
   764→    pendingLaunchVersion = null;
   765→    if (v) doLaunch(v);
   766→    return;
   767→  }
   768→  const fill = el<HTMLDivElement>("rt-progress-fill");
   769→  setText("rt-progress-stage", p.stage === "runtime-download" ? "下载" : "解包");
   770→  setText("rt-progress-file", p.file ?? "");
   771→  if (p.total > 0) {
   772→    fill.style.width = `${Math.min(100, p.percent).toFixed(1)}%`;
   773→    setText("rt-progress-text", `${p.percent.toFixed(1)}%（${bytes(p.current)} / ${bytes(p.total)}）`);
   774→  } else {
   775→    fill.style.width = "100%";
   776→    setText("rt-progress-text", String(p.current));
   777→  }
   778→}

   780→/* ---------- 启动游戏（M0 真实启动链路） ---------- */
   781→async function doLaunch(versionId: string) {
   782→  try {
   783→    setLaunchBadge("启动中…", "warn");
   784→    appendLog("quartz", `[quartz] 启动 Minecraft ${versionId}…`);
   785→    await invoke("launch_game", { versionId });
   786→    setLaunchBadge("运行中", "ok");
   787→    show("stop-btn");
   788→  } catch (e) {
   789→    const msg = String(e);
   790→    if (msg.startsWith("RUNTIME_MISSING:")) {
   791→      const major = Number(msg.split(":")[1] || 17);
   792→      appendLog("quartz", `[quartz] 需要 Java ${major} 运行时，开始自动下载…`);
   793→      showRtProgress();
   794→      try {
   795→        await invoke("download_runtime", { javaMajor: major });
   796→      } catch (e2) {
   797→        runtimeDownloading = false;
   798→        hide("rt-progress");
   799→        toast(String(e2), "err");
   800→        setLaunchBadge("启动失败", "err");
   801→      }
   802→      return;
   803→    }
   804→    setLaunchBadge("启动失败", "err");
   805→    hide("stop-btn");
   806→    appendLog("quartz", `[quartz] 启动失败：${msg}`);
   807→    toast(msg, "err");
   808→  }
   809→}

   811→async function launch() {
   812→  const versionId = el<HTMLSelectElement>("version-select").value;
   813→  if (!versionId) {
   814→    toast("请先选择版本", "err");
   815→    return;
   816→  }
   817→  if (!installed.has(versionId)) {
   818→    toast("该版本尚未下载，请先在「下载游戏」页获取", "err");
   819→    return;
   820→  }
   821→  try {
   822→    const pf = await invoke<Preflight>("launch_preflight", { versionId });
   823→    if (!pf.accountOk) {
   824→      toast("请先在「账户」页登录", "err");
   825→      return;
   826→    }
   827→    if (!pf.versionInstalled) {
   828→      toast("版本文件缺失，请重新下载", "err");
   829→      return;
   830→    }
   831→    if (!pf.runtimeInstalled) {
   832→      if (!pf.runtimeSpecAvailable) {
   833→        toast(`该版本需要 Java ${pf.javaMajor}，运行时下载源暂未收录`, "err");
   834→        return;
   835→      }
   836→      appendLog("quartz", `[quartz] 需要 Java ${pf.javaMajor} 运行时，开始自动下载…`);
   837→      pendingLaunchVersion = versionId;
   838→      showRtProgress();
   839→      await invoke("download_runtime", { javaMajor: pf.javaMajor });
   840→      return; // 完成事件回调里继续 doLaunch
   841→    }
   842→    await doLaunch(versionId);
   843→  } catch (e) {
   844→    toast(String(e), "err");
   845→  }
   846→}
